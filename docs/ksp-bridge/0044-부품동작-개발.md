# 부품동작 개발 (2단계: 부속 상위 제어 + action group)

이슈: [#15](https://github.com/SeongSikChae/KSP1-BRIDGE-MCP/issues/15)
방안: [0040-부품동작-방안.md](0040-부품동작-방안.md) · 설계: [0043-부품동작-설계.md](0043-부품동작-설계.md)

- 순서: 1 → 2 → 3 → 4 → 5 → 6. 단위마다 구현 → 검증 → 커밋 여부를 확인한다.
- Logic 단위(1~4)는 단위 1에서 한 번 `Routes.Version = "0.2.43"`으로 올린다.

- Logic 빌드: `dotnet build src/KspBridge.Logic`(GameData `KspBridge/PluginData`로 배포). App: `dotnet build src/KspBridge.App.Tools`, `dotnet test tests/KspBridge.App.Tests`.

```csharp
/// <summary>CanActivateField for an active-vessel part (design 0041 PAW rules).</summary>
private static bool Shown(BaseField f) => f.guiActive && !(f.advancedTweakable && !GameSettings.ADVANCED_TWEAKABLES);

/// <summary>
/// The flight control a player can change the value with (design 0043 set_vessel_part_field step 5), or null; the field type
/// must also be one PartSettings.Convert takes, so editable never promises a value set_vessel_part_field refuses.
/// </summary>
private static UI_Control EditableControl(BaseField f) =>
    f.uiControlFlight is UI_Control c && !(c is UI_Label) && !(c is UI_ProgressBar) && c.controlEnabled && Convertible(f.FieldInfo.FieldType) ? c : null;

private static bool Convertible(Type t) =>
    t == typeof(bool) || t == typeof(string) || t.IsEnum || t == typeof(float) || t == typeof(double) || t == typeof(int);

/// <summary>UIPartAction* value change: requireFullControl → TWEAKABLES_FULLONLY, else TWEAKABLES_ANYCONTROL.</summary>
private static ControlTypes FieldLock(UI_Control c) => c.requireFullControl ? ControlTypes.TWEAKABLES_FULLONLY : ControlTypes.TWEAKABLES_ANYCONTROL;

/// <summary>UIPartActionResource.Setup: the flow button exists.</summary>
private static bool FlowToggleable(PartResource r) =>
    r.maxAmount != 0 && r.isVisible && r.info.resourceFlowMode != ResourceFlowMode.NO_FLOW && !r.hideFlow;   // window row + flow button (완료 전 검토 R1)
```

```csharp
private static void AddRange(Dictionary<string, object> d, BaseField f, UI_Control c)
{
    switch (c)
    {
        case UI_FloatRange r: d["min"] = r.minValue; d["max"] = r.maxValue; d["step"] = r.stepIncrement; break;
        case UI_FloatEdit e: d["min"] = e.minValue; d["max"] = e.maxValue; break;
        case UI_ScaleEdit s when s.intervals != null && s.intervals.Length > 0:
            d["min"] = s.intervals[0]; d["max"] = s.intervals[s.intervals.Length - 1]; break;
        case UI_Cycle y when y.stateNames != null: Options(d, f, y.stateNames, y.stateNames); break;
        case UI_ChooseOption o when o.options != null: Options(d, f, o.options, o.display ?? o.options); break;
    }
}

/// <summary>options: titles the window shows; a string field also gets option_values, the values set_vessel_part_field takes.</summary>
private static void Options(Dictionary<string, object> d, BaseField f, string[] values, string[] titles)
{
    d["options"] = titles.Select(t => (object)Localizer.Format(t)).ToList();
    if (f.FieldInfo.FieldType == typeof(string))
    {
        d["option_values"] = values.Cast<object>().ToList();
    }
}
```

**대상**

- `FlightHandler.Routes`: 라우트 추가

**코드 초안**

```csharp
/// <summary>
/// A flight part action window value change (UIPartActionFieldItem.SetFieldValue, scene Flight): counterparts first when the control
/// affects them in flight, then the part itself with onFieldChanged (old value), only when something changes.
/// </summary>
public static object SetField(Vessel v, RequestBody b)
{
    uint id = PartIdFromBody(b);
    string module = b.GetString("module");
    string name = b.GetString("field");
    object raw = b.GetRaw("value") ?? throw ApiException.BadRequest("'value' is required");
    int? moduleIndex = b.Has("module_index") ? ModuleIndex(b) : (int?)null;
    Part p = v.parts.FirstOrDefault(x => x.persistentId == id) ?? throw PartNotFound(id);
    PartModule target = FindModule(p, module, moduleIndex, "field_not_editable", "part-level fields", out int? index);
    BaseFieldList fields = target == null ? p.Fields : target.Fields;
    BaseField f = fields[name];
    if (f == null || !Shown(f))
    {
        throw new ApiException(404, "field_not_found", name + " is not a field the part action window shows for " + module + " on "
            + p.partInfo.name + " (fields: " + string.Join(", ", fields.Cast<BaseField>().Where(Shown).Select(x => x.name).ToArray()) + ")");
    }
    UI_Control c = EditableControl(f)
        ?? throw new ApiException(409, "field_not_editable", module + "." + name + " is shown but cannot be changed in flight");
    if (!Unlocked(FieldLock(c)))
    {
        throw new ApiException(409, "field_locked", module + "." + name + " cannot be changed now: input_locked (" + FieldLock(c) + ")");
    }
    object value = PartSettings.Convert(f, raw);
    PartSettings.CheckRange(f, c, value);

    object old = f.GetValue(f.host);
    bool changed = !Equals(old, value);
    List<Part> group = new List<Part> { p };
    List<object> written = new List<object>();   // counterpart ids whose value was set
    bool own = false;
    try
    {
        if ((c.affectSymCounterparts & UI_Scene.Flight) != UI_Scene.None)
        {
            foreach (Part s in p.symmetryCounterparts.Where(s => s != null))
            {
                if (target == null)
                {
                    changed |= !Equals(s.Fields.GetValue(name), value);
                    s.Fields.SetValue(name, value);   // part-level: value only, no events (SetSymCounterpartValue)
                    written.Add((double)s.persistentId);
                }
                else
                {
                    PartModule sm = PartSettings.CounterpartModule(s, index.Value, target);
                    if (sm == null)
                    {
                        continue;
                    }
                    changed |= !Equals(sm.Fields.GetValue(name), value);
                    sm.Fields.SetValue(name, value);
                    written.Add((double)s.persistentId);
                    BaseField sf = sm.Fields[name];
                    sf?.uiControlEditor?.onSymmetryFieldChanged?.Invoke(f, value);
                    sf?.uiControlFlight?.onSymmetryFieldChanged?.Invoke(f, value);
                }
                group.Add(s);
            }
        }
        if (changed)
        {
            f.SetValue(value, f.host);
            own = true;
            c.onFieldChanged?.Invoke(f, old);
            Log("part_field", ..., ...);   // {part_id, part, module, index, field, old, value, counterparts: written, error: null}
        }
    }
    catch (Exception ex)   // a game callback failed part way (design 0043 step 8): say what already changed
    {
        string error = ex.GetType().Name + ": " + ex.Message;
        Log("part_field", ..., ...);   // same data with own and error
        throw new ApiException(500, "field_failed", error + " (value already set on counterparts [" + string.Join(", ", written.Select(x => x.ToString()).ToArray())
            + "]" + (own ? " and on the part itself" : "") + "; read get_vessel_parts before retrying)");
    }
    finally
    {
        foreach (Part s in group)
        {
            MonoUtilities.RefreshPartContextWindow(s);
        }
    }
    return new Dictionary<string, object>
    {
        ["part_id"] = (double)id, ["part"] = p.partInfo.name, ["module"] = module, ["index"] = index, ["field"] = name,
        ["old"] = FieldValue(old), ["value"] = FieldValue(f.GetValue(f.host)), ["changed"] = changed,
        ["values"] = group.Select(s => (object)new Dictionary<string, object>
        {
            ["part_id"] = (double)s.persistentId,
            ["value"] = FieldValue(target == null ? s.Fields.GetValue(name) : (s == p ? target : PartSettings.CounterpartModule(s, index.Value, target))?.Fields.GetValue(name)),
        }).ToList(),
    };
}
```

- 대칭 부품 루프는 `PartSettings.SetField`와 같은 구조다(그쪽은 editor 전용이라 그대로 두고 호출하지 않는다).
- `Log`: `FlightEvents.Add(type, Coordinator.Running ? Coordinator.StepIndex : -1, data)`와 `UnityEngine.Debug.Log("[KspBridge] " + line)`. 1단계 `Record`는 바꾸지 않는다.
- 라우트: `yield return new Route("POST", "/api/flight/part_field", (m, b, q) => Flight.PartStates.SetField(Guard.ActiveVessel(), b));`

**검증**
- 시험 기체(단위 1)의 엔진에서 `thrustPercentage` 50 → `values`는 그 엔진 하나(`affectSymCounterparts`가 Editor만이라 비행 중 대칭 전파 없음, 게임과 같음), `get_vessel_parts` `thrust_limit` 50, `get_flight_events`에 `part_field` 1건(`error: null`). 대칭 전파는 조명 `blinkRate`(Flight 포함)로 확인한다. (처음 기대값 "values 모든 부품 50"은 실측에서 틀린 것으로 확인해 고쳤다.)
- 같은 값을 다시 보내면 `changed: false`, 이벤트 없음.
- 400: `value` 없음, 범위 밖(150), 타입 불일치(문자열). 404 `field_not_found`(없는 field, 메시지에 목록). 409 `field_not_editable`(`editable: false`인 표시 field, 예: 엔진 상태 문자열 field).
- `module: "Part"`에 `module_index`를 주면 400 "(part-level fields)". 시험 기체에 편집 가능한 부품 수준 field가 있으면 그 값도 한 번 바꿔 본다. 없으면 기록한다.
- `debug_input_lock`으로 `TWEAKABLES_ANYCONTROL`(0x1000) 잠금 → 409 `field_locked`, `get_vessel_parts` 해당 행 `editable: false`. 잠금 해제(mask null) 뒤 다시 `editable: true`.
- `field_failed`(500)는 콜백 예외를 일으킬 stock 경로가 없어 실측하지 않는다. 코드 확인으로 대신하고 미확인으로 기록한다.

**선행 단위**: 1

```csharp
/// <summary>The resource row's flow button (UIPartActionResource.FlowBtnToggled): this part only, no symmetry.</summary>
public static object SetFlow(Vessel v, RequestBody b)
{
    uint id = PartIdFromBody(b);
    string name = b.GetString("resource");
    bool enabled = b.GetBool("enabled");
    Part p = v.parts.FirstOrDefault(x => x.persistentId == id) ?? throw PartNotFound(id);
    PartResource r = p.Resources[name] ?? throw new ApiException(404, "resource_not_found", name + " is not in " + p.partInfo.name
        + " (resources: " + string.Join(", ", p.Resources.Cast<PartResource>().Select(x => x.resourceName).ToArray()) + ")");
    if (!FlowToggleable(r))
    {
        throw new ApiException(409, "flow_not_toggleable", name + " in " + p.partInfo.name + " has no flow button (no-flow resource or hidden)");
    }
    if (!Unlocked(ControlTypes.TWEAKABLES_ANYCONTROL))
    {
        throw new ApiException(409, "flow_locked", "the flow of " + name + " cannot be changed now: input_locked (TWEAKABLES_ANYCONTROL)");
    }
    bool before = r.flowState;
    r.flowState = enabled;
    if (before != enabled)
    {
        Log("resource_flow", ..., ...);   // {part_id, part, resource, before, flow_enabled}
    }
    MonoUtilities.RefreshPartContextWindow(p);
    return new Dictionary<string, object> { ["part_id"] = (double)id, ["part"] = p.partInfo.name, ["resource"] = name, ["before"] = before, ["flow_enabled"] = r.flowState };
}
```

**검증**
- 연료 탱크 `LiquidFuel` false → 응답 `flow_enabled: false`, `get_vessel_parts` 같은 값, `resource_flow` 이벤트 1건. true로 되돌림.
- 404 `resource_not_found`, 409 `flow_not_toggleable`(단위 1에서 `flow_toggleable: false`로 나온 자원), 400(`enabled` 누락).
- `debug_input_lock` `TWEAKABLES_ANYCONTROL`(0x1000) → 409 `flow_locked`, 해제 뒤 정상.
- 비행 PAW는 자원 행 control로 null을 넘기므로(설계 0043) `requireFullControl`은 보지 않는다.

**코드 초안**

```csharp
/// <summary>get_action_groups and set_action_group (design 0043). Main thread only.</summary>
public static class ActionGroups
{
    private static readonly KSPActionGroup[] Settable =
    {
        KSPActionGroup.Custom01, KSPActionGroup.Custom02, KSPActionGroup.Custom03, KSPActionGroup.Custom04, KSPActionGroup.Custom05,
        KSPActionGroup.Custom06, KSPActionGroup.Custom07, KSPActionGroup.Custom08, KSPActionGroup.Custom09, KSPActionGroup.Custom10,
        KSPActionGroup.Light, KSPActionGroup.Gear, KSPActionGroup.Brakes, KSPActionGroup.Abort,
    };

    /// <summary>The lock FlightInputHandler checks before the group's key.</summary>
    private static ControlTypes KeyLock(KSPActionGroup g)
    {
        switch (g)
        {
            case KSPActionGroup.Light: return ControlTypes.GROUP_LIGHTS;
            case KSPActionGroup.Gear: return ControlTypes.GROUP_GEARS;
            case KSPActionGroup.Brakes: return ControlTypes.GROUP_BRAKES;
            case KSPActionGroup.Abort: return ControlTypes.GROUP_ABORT;
            default: return ControlTypes.CUSTOM_ACTION_GROUPS;
        }
    }

    public static object List(Vessel v)
    {
        List<object> rows = new List<object>();
        for (int i = 0; i < BaseAction.ActionGroupsLength; i++)   // KSPActionGroup bits without None and REPLACEWITHDEFAULT
        {
            KSPActionGroup g = (KSPActionGroup)(1 << i);
            bool settable = Settable.Contains(g);
            rows.Add(new Dictionary<string, object>
            {
                ["group"] = g.ToString(),
                ["on"] = v.ActionGroups[g],
                ["cooldown"] = v.ActionGroups.CooldownTimeRemaining(g),
                ["settable"] = settable,
                ["locked"] = settable ? (object)(!PartStates.Unlocked(KeyLock(g)) || ManualControl.Overridden(v, g)) : null,
                ["actions"] = Actions(v, g),
            });
        }
        return new Dictionary<string, object>
        {
            ["vessel"] = v.vesselName, ["id"] = (double)v.persistentId,
            ["override"] = GameSettings.ADDITIONAL_ACTION_GROUPS ? v.GroupOverride : 0,
            ["groups"] = rows,
        };
    }

    /// <summary>The actions ActionGroupList.ToggleGroup would invoke for the group, chosen the same way.</summary>
    private static List<object> Actions(Vessel v, KSPActionGroup g)
    {
        int n = GameSettings.ADDITIONAL_ACTION_GROUPS ? v.GroupOverride : 0;
        bool def = n > 0 && n <= v.OverrideDefault.Length && v.OverrideDefault[n - 1];
        return BaseAction.CreateActionList(v.parts, g, n, def, true).Select(a =>
        {
            Part p = a.listParent.part;
            PartModule m = a.listParent.module;
            return (object)new Dictionary<string, object>
            {
                ["part_id"] = (double)p.persistentId, ["part"] = p.partInfo.name,
                ["module"] = m == null ? PartStates.PartLevel : m.moduleName,
                ["index"] = m == null ? null : (object)p.Modules.IndexOf(m),
                ["action"] = a.name, ["title"] = a.guiName,
            };
        }).ToList();
    }

    public static object Set(Vessel v, RequestBody b)
    {
        if (v.isEVA)
        {
            throw new ApiException(409, "eva_not_supported", "EVA kerbals are not supported");
        }
        KSPActionGroup g = ParseGroup(b.GetString("group"));
        string action = b.GetString("action");
        if (action != "toggle" && action != "on" && action != "off")
        {
            throw ApiException.BadRequest("action must be toggle, on or off");
        }
        bool before = v.ActionGroups[g];
        bool want = action == "toggle" ? !before : action == "on";
        if (want != before && !PartStates.Unlocked(KeyLock(g)))   // already holding: changed false even when locked (완료 전 검토 R1)
        {
            throw new ApiException(409, "group_locked", g + " is input_locked (" + KeyLock(g) + (v.IsControllable ? "" : "; no control") + ")");
        }
        List<object> actions = Actions(v, g);
        uint id = v.persistentId;
        if (want != before)
        {
            try
            {
                ManualControl.SetGroupChecked(v, g, want);   // 409 group_locked: override or cooldown, group left as it was
            }
            catch (ApiException)
            {
                throw;
            }
            catch (Exception ex)   // an action threw inside ToggleGroup: the group already flipped, later actions did not run, no cooldown
            {
                string error = ex.GetType().Name + ": " + ex.Message;
                PartStates.Log("action_group", ..., ...);   // {group, action, before, on: current, changed, actions, error}
                throw new ApiException(500, "action_failed", error + " (" + g + " is now " + (v.ActionGroups[g] ? "on" : "off")
                    + "; read get_action_groups before retrying)");
            }
            PartStates.Log("action_group", ..., ...);   // {group, action, before, on, changed: true, actions, error: null}
        }
        return new Dictionary<string, object>
        {
            ["group"] = g.ToString(), ["action"] = action, ["before"] = before, ["on"] = v.ActionGroups[g], ["changed"] = want != before,
            ["cooldown"] = v.ActionGroups.CooldownTimeRemaining(g), ["actions"] = want != before ? actions : new List<object>(),
            ["vessel_changed"] = FlightGlobals.ActiveVessel == null || FlightGlobals.ActiveVessel.persistentId != id,
        };
    }

    /// <summary>Case-insensitive; SAS, RCS and Stage point to their own tools.</summary>
    private static KSPActionGroup ParseGroup(string name)
    {
        KSPActionGroup g = Settable.FirstOrDefault(x => string.Equals(x.ToString(), name, StringComparison.OrdinalIgnoreCase));
        if (g != KSPActionGroup.None)
        {
            return g;
        }
        switch (name.ToUpperInvariant())
        {
            case "SAS": throw ApiException.BadRequest("use set_sas for the SAS group");
            case "RCS": throw ApiException.BadRequest("use set_rcs for the RCS group");
            case "STAGE": throw ApiException.BadRequest("Stage is not set here (staging)");
        }
        throw ApiException.BadRequest("group must be one of " + string.Join(", ", Settable.Select(x => x.ToString()).ToArray()));
    }
}
```

- override·쿨다운 409, 조종 불가 기체의 `group_locked` "; no control" 문구, `ambiguous_module`, 계획 실행 중 분리(`vessel_changed`), `action_failed`·`field_failed`(500), `flow_not_toggleable`(흐름 버튼 없는 자원이 시험 기체에 없음), Custom 그룹 실행(배정된 craft 없음).

### 완료 전 검토 반영

- `flow_toggleable`·`set_resource_flow`에 비행 PAW 자원 행 조건(`maxAmount != 0`, `isVisible`)을 더했다. 시험 기체 탱크 LiquidFuel·Oxidizer는 그대로 `flow_toggleable: true`, LiquidFuel 흐름 변경 정상.
- `UI_ColorPicker`를 편집 가능 컨트롤에서 뺐다. 조명 `colorChanger` `editable: false`, `set_vessel_part_field` 409 `field_not_editable`.
- `set_action_group`: 이미 같은 값이면 잠금 검사 전에 `changed: false`로 끝낸다. Light `off`(이미 off) → `changed: false`. 잠금이 걸린 상태의 같은 값 호출은 `debug.cfg`를 다시 켜지 않아 실측하지 않았다(코드 순서로 확인).
- 도구 설명: `get_flight_events`의 새 기록 3종 설명 위치를 고쳤고, `invoke_part_event` `module_state`가 `get_vessel_parts`와 같은 키를 가진다는 것과 `set_action_group` 잠금 중 같은 값 동작을 적었다.
