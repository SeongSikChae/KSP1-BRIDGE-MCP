에디터 내부 부착 로직(`attachPart`, `createSymmetry`, `checkAttach`)은 private이며 FSM 상태에 묶여 있다. 부착은 공개 API로 직접 구성하고(아래 U5), 동작은 게임 안에서 확인한다.

`EditorLogic.RootPart`는 읽기 전용이고, 루트는 private `rootPart` 필드와 FSM 상태(`st_podSelect` = 빈 기체)로 관리된다. 그래서 루트는 직접 만들지 않고 게임의 불러오기 경로(`EditorLogic.LoadShipFromFile`)로 넣는다(U5).

`BaseField`(`BaseField<KSPField>`)는 `FieldInfo`, `SetValue(object, object)`, `GetValue(object)`, `guiActiveEditor`, `uiControlEditor`가 공개다. Mod의 메인 스레드 제한은 `HttpServer.MainThreadTimeout` = 5초다.

## 설계 보완 (개발 중 확인)

- **Δv 상황**: 스테이지 앱의 천체·고도 선택(`DeltaVAppSituation.SelectedBody`)은 읽기 전용 UI 상태라 도구가 정할 수 없다. 그래서 `check_craft`는 내장 `VesselDeltaV`의 스테이지 구성·질량·진공/해수면 값을 그대로 쓴다. `body`·`altitude`를 주면 그 상황의 TWR·Δv를 내장 스테이지 질량과 엔진 `atmosphereCurve`(압력별 Isp)로 따로 구한다. 설계 결정 5(내장 DeltaV 사용)는 유지하고, 상황별 값만 이렇게 구한다.

## 공통 코드 (각 단위에서 재사용)

```csharp
// Logic/Editor/EditorScene.cs
public static class EditorScene
{
    public static bool Ready =>
        HighLogic.LoadedScene == GameScenes.EDITOR && EditorLogic.fetch != null && EditorLogic.fetch.ship != null
        && !EditorDriver.fetch.restartingEditor;

    public static ShipConstruct Require()
    {
        if (HighLogic.LoadedScene != GameScenes.EDITOR)
            throw new ApiException(409, "not_in_editor", "open the VAB or SPH first (scene: " + HighLogic.LoadedScene + ")");
        if (!Ready)
            throw new ApiException(409, "editor_not_ready", "the editor is still loading; poll get_status until editor.ready");
        return EditorLogic.fetch.ship;
    }

    /// <summary>After a successful edit: one Undo step and editor/DeltaV refresh.</summary>
    public static void Commit(ShipConstruct ship)
    {
        GameEvents.onEditorShipModified.Fire(ship);
        EditorLogic.fetch.SetBackup();
    }
}
```

```csharp
// Logic/Editor/PartIndex.cs
public static Part Find(ShipConstruct ship, uint id) =>
    ship.parts.FirstOrDefault(p => p.persistentId == id)
    ?? throw new ApiException(404, "part_not_found", "no part with id " + id + " in the editor craft");

// 루트 로컬 좌표 (설계: +Y = 위/기수, VAB·SPH 동일)
public static Vector3 Local(ShipConstruct ship, Vector3 world) => ship.parts[0].transform.InverseTransformPoint(world);
```

```csharp
public static Dictionary<string, object> Summary(ShipConstruct ship, Part p) => new Dictionary<string, object>
{
    ["id"] = p.persistentId, ["part"] = p.partInfo.name, ["title"] = p.partInfo.title,
    ["parent"] = p.parent != null ? (object)p.parent.persistentId : null,
    ["attach"] = Attach(p),          // root | node{parent_node, child_node} | surface
    ["symmetry"] = p.symmetryCounterparts.Count == 0 ? null : new Dictionary<string, object>
        { ["mode"] = p.symMethod == SymmetryMethod.Mirror ? "mirror" : "radial",
          ["group"] = new[] { p }.Concat(p.symmetryCounterparts).Select(c => c.persistentId).ToList() },
    ["stage"] = p.hasStagingIcon && p.stagingOn ? (object)p.inverseStage : null,   // U10 중 수정: 스테이징이 꺼진 부품(도킹포트)은 null
    ["position"] = Vec(Local(ship, p.transform.position)),
    ["rotation"] = Vec((Quaternion.Inverse(Root(ship).transform.rotation) * p.transform.rotation).eulerAngles),
};

static object Attach(Part p)
{
    if (p.parent == null) return new Dictionary<string, object> { ["kind"] = "root" };
    AttachNode toParent = p.FindAttachNodeByPart(p.parent);
    if (p.attachMode == AttachModes.SRF_ATTACH || toParent == null) return new Dictionary<string, object> { ["kind"] = "surface" };
    return new Dictionary<string, object> { ["kind"] = "node", ["parent_node"] = p.parent.FindAttachNodeByPart(p)?.id, ["child_node"] = toParent.id };
}
```

```csharp
public static object Add(ShipConstruct ship, AddRequest r)
{
    AvailablePart ap = PartCatalog.RequireUsable(r.Part);         // 없음 404, 잠김 409 part_locked
    Part parent = r.Parent.HasValue ? PartIndex.Find(ship, r.Parent.Value) : null;
    if (parent == null && ship.parts.Count > 0) throw ApiException.BadRequest("parent is required: the craft already has a root");
    // 1) 검사: 노드 존재·비어 있음·대칭 슬롯 — 실패하면 아무것도 만들지 않는다
    Placement[] places = parent == null ? new[] { Placement.Root() } : Plan(ship, ap, parent, r);
    // 2) 생성: 실패하면 만든 부품만 지운다
    List<Part> created = new List<Part>();
    try
    {
        foreach (Placement pl in places) created.Add(Place(ship, ap, parent, pl));
        LinkSymmetry(created, r.Symmetry);
    }
    catch { foreach (Part p in created) EditorLogic.DeletePart(p); throw; }
    EditorScene.Commit(ship);
    return EditResult.Added(ship, created);
}

static Part Spawn(AvailablePart ap)
{
    Part p = UnityEngine.Object.Instantiate(ap.partPrefab);
    p.gameObject.SetActive(true);
    p.name = ap.name;
    p.partInfo = ap;
    p.persistentId = FlightGlobals.CheckPartpersistentId(p.persistentId, p, false, true, 0);
    return p;
}

// 스택: 자식 노드가 부모 노드와 반대 방향을 보도록 회전, 노드 위치를 겹친다
static void AttachStack(ShipConstruct ship, Part parent, AttachNode pn, Part child, AttachNode cn)
{
    Quaternion rot = parent.transform.rotation * Quaternion.FromToRotation(cn.orientation, -pn.orientation);
    Vector3 pos = parent.transform.TransformPoint(pn.position) - rot * cn.position;
    child.transform.SetPositionAndRotation(pos, rot);
    child.attachMode = AttachModes.STACK;
    pn.attachedPart = child;
    cn.attachedPart = parent;
    child.setParent(parent);
    child.onAttach(parent, true);
    ship.Add(child);
    GameEvents.onEditorPartEvent.Fire(ConstructionEventType.PartAttached, child);
}
```

- 루트(R1): `RootPart`가 읽기 전용이고 빈 기체는 FSM `st_podSelect` 상태라 직접 만들지 않는다. 부품 1개짜리 기체를 임시 `.craft`로 써서 게임의 불러오기로 넣는다.

```csharp
static object AddRoot(AvailablePart ap)
{
    Part p = Spawn(ap);
    string path = Path.Combine(TempDir, "root-" + Guid.NewGuid().ToString("N") + ".craft");   // GameData/KspBridge/PluginData/tmp
    try
    {
        ShipConstruct one = new ShipConstruct(EditorLogic.fetch.ship.shipName, "", p) { shipFacility = EditorDriver.editorFacility };
        one.SaveShip().Save(path);
    }
    finally { UnityEngine.Object.Destroy(p.gameObject); }
    EditorLogic.LoadShipFromFile(path);            // 게임이 루트·FSM·Undo 기준점을 처리
    File.Delete(path);
    ShipConstruct ship = EditorScene.Require();
    return EditResult.Added(ship, new List<Part> { ship.parts[0] });   // id는 불러온 뒤의 persistentId
```

### U6. 표면 부착 + 대칭

- **대상**: `PartPlacer`(`AttachSurface`, `Plan`, `LinkSymmetry`)
- **코드 초안**:

```csharp
// 부모 +Y 축 기준 방위각·높이에서 바깥으로 레이를 쏴 표면 점을 찾는다 (부모 로컬)
static bool SurfacePoint(Part parent, float azimuthDeg, float height01, out Vector3 point, out Vector3 normal)
{
    Bounds b = PartBounds(parent);                                   // 부모 로컬 렌더러 경계
    float y = Mathf.Lerp(b.min.y, b.max.y, height01);
    Vector3 dir = Quaternion.AngleAxis(azimuthDeg, Vector3.up) * Vector3.forward;   // 0 = +Z
    Vector3 from = parent.transform.TransformPoint(new Vector3(0, y, 0) + dir * (b.extents.magnitude * 2));
    Ray ray = new Ray(from, -parent.transform.TransformDirection(dir));
    foreach (RaycastHit hit in Physics.RaycastAll(ray, b.extents.magnitude * 4).OrderBy(h => h.distance))
        if (hit.collider.GetComponentInParent<Part>() == parent) { point = hit.point; normal = hit.normal; return true; }
    point = normal = Vector3.zero; return false;
}

static void AttachSurface(ShipConstruct ship, Part parent, Part child, Vector3 point, Vector3 normal, float angleDeg)
{
    AttachNode sn = child.srfAttachNode;
    Quaternion rot = Quaternion.AngleAxis(angleDeg, normal)
                   * Quaternion.LookRotation(normal, parent.transform.up)
                   * Quaternion.Inverse(Quaternion.LookRotation(-sn.orientation, Vector3.up));
    child.transform.SetPositionAndRotation(point - rot * sn.position, rot);
    child.attachMode = AttachModes.SRF_ATTACH;
    sn.attachedPart = parent;
    child.setParent(parent);
    child.onAttach(parent, true);
    ship.Add(child);
    GameEvents.onEditorPartEvent.Fire(ConstructionEventType.PartAttached, child);
}

static void LinkSymmetry(List<Part> parts, SymmetrySpec s)
{
    if (parts.Count < 2) return;
    foreach (Part p in parts)
    {
        p.symMethod = s.Mode == "mirror" ? SymmetryMethod.Mirror : SymmetryMethod.Radial;
        p.symmetryCounterparts = parts.Where(o => o != p).ToList();
    }
}
```

- **구현 중 결정 (U6, 게임 검증 결과)**:
  - 위치·회전은 디컴파일한 에디터 공식을 그대로 쓴다. 표면: `EditorLogic.checkAttach`의 `rotation = LookRotation(hit.normal, parent.rotation*up) * LookRotation(srfAttachNode.orientation, up)`, `position = hit.point - rotation * attRotation * srfAttachNode.position`, `srfAttachNode.srfAttachMeshName = hit.collider.name`. 방사형: `EditorLogic.UpdateSymmetry`처럼 첫 부품을 부모 `up` 축으로 `k×360/n` 회전한 사본. 좌우: 루트 `up`·`forward` 평면 반사 + 루트 기준 x 성분 반전 + `Part.OnWillBeMirrored` 훅 + 표면 노드 `orientation.x != 0`이면 180도. 에디터도 메시 반전(`SetMirror`)은 하지 않는다.
  - 표면 레이: 부모 부품 **자신의** 콜라이더만 쓴다(자식 부품은 transform 자식이라 `GetComponentsInChildren`에 섞인다). 에디터의 부품 콜라이더는 트리거라서 트리거를 제외하지 않고 `Collider.Raycast`를 직접 쓴다.
  - 방위각 기준축은 부모 원점이 아니라 **부모 콜라이더 경계 중심을 지나는 +Y 축**이다(분리기처럼 원점이 몸체 밖에 있는 부품 대응). 원통형 부품에서는 원점 축과 같다. 방위각은 부모 로컬 기준이라 바깥쪽 방향은 `get_craft`의 회전으로 계산한다(예: 탱크 +Z 쪽 방사형 분리기는 로컬 -X가 바깥 → 방위각 270).
  - 대칭 부품은 모두 집기 → 모두 연결 → 모두 놓기 순서로 붙인다. 놓기 전에 `symmetryCounterparts`를 묶어 두어 `StageManager`가 대칭 아이콘을 한 스테이지로 묶는다.
  - 표면을 못 찾으면 `attach_failed` 메시지에 콜라이더 개수·상태를 담는다(원인 파악용).
- 검사 단계에서 부모의 `attachRules.allowSrfAttach`, 자식의 `attachRules.srfAttach`, 모든 대칭 위치의 레이 적중을 먼저 확인한다(하나라도 실패하면 생성 전 409 `attach_failed`).
- radial: `azimuth + 360·k/count`. mirror(SPH 기본): 두 번째는 부모 로컬 X 반전 방위(`360 - azimuth`), 회전은 `SetMirror`로 좌우 반전.
- 스택 부착에 대칭을 주면 400(스택 대칭은 비범위).
- **검증**: 게임: 탱크에 `radialDecoupler` radial 3개, 그 위에 SRB 각각 → 화면에서 120° 간격, `get_craft` symmetry group 3개. SPH 동체에 날개 mirror 2개 → 좌우 대칭 확인. 레이 미적중(높이 밖) 요청은 `part_count` 변화 없음.
- **선행 단위**: U5

### U7. 제거·보정 (`remove_part`, `offset_part`, `rotate_part`)

- **대상**: `PartPlacer`(`Remove`, `Offset`, `Rotate`)
- **코드 초안**:

```csharp
public static object Remove(ShipConstruct ship, uint id)
{
    Part p = PartIndex.Find(ship, id);
    if (p.parent == null) throw ApiException.BadRequest("cannot remove the root part; use new_craft");
    List<uint> removed = new[] { p }.Concat(p.symmetryCounterparts)
        .SelectMany(Subtree).Select(x => x.persistentId).Distinct().ToList();
    foreach (Part s in new[] { p }.Concat(p.symmetryCounterparts).ToList())
        if (s != null && ship.parts.Contains(s)) EditorLogic.DeletePart(s);   // R6: 앞의 DeletePart가 사본을 이미 지웠으면 건너뜀
    EditorScene.Commit(ship);
    return EditResult.Removed(removed);
}

// 루트 로컬 좌표로 받은 변위를 대칭 사본에도 각자의 대칭 프레임으로 적용한다
public static object Offset(ShipConstruct ship, uint id, Vector3 deltaRootLocal)
{
    Part p = PartIndex.Find(ship, id);
    Vector3 world = Root(ship).transform.TransformVector(deltaRootLocal);
    foreach (Part s in SymmetryGroup(p)) MoveSubtree(s, SymmetricVector(p, s, world));
    EditorScene.Commit(ship);
    return EditResult.Changed(ship, SymmetryGroup(p));
}
```

- `MoveSubtree`는 부품과 자식 전체의 `transform.position`을 같이 옮긴다. 회전은 부품 자신의 부착점 기준 `AngleAxis`.
- 루트를 offset하면 기체 전체를 옮긴다(에디터와 같음).
- R6 확인 항목: 대칭 그룹 하나를 `remove_part`했을 때 `EditorLogic.DeletePart`가 사본까지 지우는지 로그로 남기고, 어느 쪽이든 그룹 전체가 한 번씩만 지워지는지 확인한다.
- **구현 중 결정 (U7, 디컴파일·게임 검증 결과)**:
  - 제거(R6): `EditorLogic.DeletePart`는 쓰지 않는다. 내부에서 `displayAttachNodeIcons(ship.parts, SelectedPart, ...)`를 불러 마우스로 든 부품이 없으면 NRE가 나고, 대칭 처리(`deleteSymmetryParts`)도 든 부품 기준이라 사본을 지우지 않는다. 대신 에디터 순서를 재현한다: 집기(`detachPart` → 기체에 남은 사본마다 `PartSymmetryDeleted`·`detachPart`·`deletePart` → `wipeSymmetry` → `onEditorPartPicked` → `PartDetached`) → `DestroySelectedPart`(`PartDeleted` → `deletePart` → `onEditorPartDeleted(null)`) → `Commit` → 승무원 갱신. `deletePart`는 자식부터 `Part.OnDelete()` + `Object.Destroy(gameObject)`다(Destroy가 없으면 목록에서만 빠지고 화면·스테이징 아이콘이 남는다). `detachPart`·`clearAttachNodes`·`wipeSymmetry`(private)는 본문대로 재현했다. 모듈 예외로 중간에 멈추면 기체에서 이미 빠진 부품도 파괴하고, `onEditorPartDeleted`·`Commit`·승무원 갱신은 `finally`에서 실행하며, 그 단계의 예외는 로그만 남겨 원래 예외를 응답한다.
  - 승무원 목록: 에디터는 `SetBackup` 뒤 private `RefreshCrewAssignment(ShipConstruction.ShipConfig, GetPartExistsFilter())`를 부른다(유인 부품 부착 후, 모든 삭제 후). 이것이 없으면 API로 붙인 승무원실이 승무원 탭에 나오지 않는다(U5 누락). `EditorScene.RefreshCrew()`가 두 private 메서드를 리플렉션으로 그대로 호출하고, `add_part`는 `CrewCapacity > 0` 부품이 있을 때, `remove_part`는 항상 부른다. 루트는 게임 불러오기 경로가 처리한다(조종석이 승무원 탭에 나오는 것을 확인). 원본 갱신은 `updateUI: false`인데, 손으로는 승무원 탭을 연 채 편집할 수 없지만 도구는 할 수 있으므로 이어서 `CrewAssignmentDialog.Instance.RefreshCrewLists(ShipManifest, false, true)`로 화면도 다시 그린다(`RefreshCrewDialog`는 열린 화면의 이전 목록을 합치므로 쓰지 않는다). 편집이 확정된 뒤 실행되므로 실패해도 예외를 던지지 않고 로그만 남긴다.
  - 이동: 에디터 offset 도구처럼 reference transform을 옮기고 `attPos`를 갱신한다. 하위 부품은 transform 자식이라 함께 움직인다. 부모 분리 방지 스냅(`TestPartBoundsSeparate`)은 하지 않는다(분리 경고는 U10).
  - 회전: 축은 루트 x/y/z, 피벗은 부모 연결 노드(`FindAttachNodeByPart(parent)`, 루트는 원점), `attRotation` 갱신식은 `onRotateGizmoUpdate`와 같다. 사본별 회전은 루프 전에 계산한다.
  - 대칭 사본: radial은 `s.rotation * inv(p.rotation)`로 벡터를 돌리고, mirror는 루트 right 성분을 반전하며 회전은 반사 축·반대 각도다(`UpdateSymmetry` 배치와 같은 관계). 요청 부품이 그룹의 첫 부품이 아니어도 된다.
  - 라우트: `DELETE /api/editor/parts/{id}`, `POST .../{id}/offset {x,y,z}`, `POST .../{id}/rotate {axis, angle}`, 값은 ±1000 제한. offset/rotate 응답의 `changed`는 대칭 그룹만 담는다(하위 부품은 함께 움직이지만 목록에 없음).
  - 에디터 Undo는 백업에서 기체를 다시 불러와 모든 `persistentId`가 새로 발급된다. 도구 설명에 Undo 후 `get_craft` 재호출을 적었다.
  - 대칭 부모에 붙이면 사본 부모에도 자동 부착하는 에디터 동작은 U6 `add_part`에 없다(별도 항목). 그래서 U7 검증은 분리기 radial 그룹 + 한쪽 SRB, SPH 윙렛 mirror 쌍으로 했다.
  - 알려진 제약(원본과 같음): 하위 트리 안 부품의 대칭 사본이 범위 밖 부모에 있으면 제거·이동 시 그 사본은 따라가지 않는다.
  - 남은 문제: `add_part` 실패 정리(`DiscardAll`)도 `EditorLogic.DeletePart`를 써서 같은 NRE가 날 수 있다(U7 범위 밖).
- **검증**: 게임: U6 SRB 그룹을 `offset y=-0.5` → 3개 모두 이동, 화면 확인. `rotate_part axis=z 10` → 대칭 사본 동일 각도. 루트 `remove_part`는 400. 엔진 제거 후 `get_craft`에서 그 id가 사라짐.
- **선행 단위**: U6

  - 편집 가능 판정: `guiActiveEditor`이고 `uiControlEditor`가 있으며 `UI_Label`·`UI_ProgressBar`가 아니고 `controlEnabled`. 같은 이름 모듈이 여럿이면 그 필드를 보이는 첫 활성 모듈(모두 꺼져 있으면 409 `module_disabled`). 모듈 인덱스 지정은 비범위.
  - 값 변환: JSON 숫자/불/문자열 → `float`(범위 초과 400)·`double`·`int`(정수만)·`bool`·`string`·enum(이름 또는 숫자). 그 밖의 타입은 400. 범위 검사는 배치된 부품의 컨트롤 기준: `UI_FloatRange`·`UI_FloatEdit` min/max, `UI_ScaleEdit` intervals 처음..끝, `UI_ChooseOption.options`·`UI_Cycle.stateNames`(이름 또는 int 인덱스). 슬라이더 step 스냅은 하지 않는다(API는 정확한 값).
  - 자원: PAW 자원 슬라이더처럼 본 부품 `amount = fraction × max`, 사본은 같은 절대량. `isTweakable`이 아닌 자원은 400(PAW도 슬라이더를 만들지 않음), 사본에 자원이 없으면 바꾸기 전에 404.
  - 응답: `get_part`는 프리팹 값만 보여 주므로 응답에 `values`(부품별 설정 후 다시 읽은 필드 값, 자원은 amount/max)를 넣는다.
  - 열린 PAW: 토글 등은 창 생성 때만 값을 읽어 외부 변경이 안 보인다(게임 확인). 변경 후 그룹 부품마다 `MonoUtilities.RefreshPartContextWindow`로 다시 그린다.

```csharp
public static List<Dictionary<string, object>> Run(ShipConstruct ship, ICollection<Part> focus = null)
{
    List<Dictionary<string, object>> issues = new List<Dictionary<string, object>>();
    if (ship.parts.Count == 0) issues.Add(Issue("error", "no_root", null, "the craft has no parts"));
    foreach (Part p in ship.parts)
    {
        if (!PartCatalog.IsUsable(p.partInfo)) issues.Add(Issue("error", "locked_part", p, p.partInfo.name + " is not unlocked"));
        if (p != Root(ship) && p.parent == null) issues.Add(Issue("error", "orphan", p, "not attached to the craft"));
        foreach (AttachNode n in p.attachNodes)
            if (n.attachedPart != null && n.attachedPart.FindAttachNodeByPart(p) == null && n.attachedPart.srfAttachNode?.attachedPart != p)
                issues.Add(Issue("error", "bad_node", p, "node " + n.id + " is linked one way only"));
    }
    issues.AddRange(Overlaps(ship, focus));
    if (!ship.AreAllPartsConnected()) issues.Add(Issue("error", "unsaveable", null, "not all parts are connected"));
    return issues;
}

static IEnumerable<Dictionary<string, object>> Overlaps(ShipConstruct ship, ICollection<Part> focus)
{
    foreach ((Part a, Part b) in Pairs(ship.parts, focus))
    {
        if (a.parent == b || b.parent == a) continue;                       // 직접 붙은 쌍은 제외
        if (!PartColliderBounds(a).Intersects(PartColliderBounds(b))) continue;   // R3: 부품 단위 bounds로 먼저 거름
        float depth = MaxPenetration(a, b);                                 // bounds가 겹치는 콜라이더 쌍만 Physics.ComputePenetration
        if (depth > OverlapThreshold) yield return Issue("warning", "overlap", new[] { a, b }, "overlap", depth);
    }
}
```

- R3: 부품 100개 이상 기체(예: 스톡 기체 불러오기)에서 `check_craft` 응답 시간을 재서 5초 메인 스레드 제한보다 충분히 작은지(1초 이내 목표) 확인한다.
- `OverlapThreshold`는 0.05 m에서 시작하고 U10 검증에서 실제 기체로 조정한 값을 문서에 기록한다.
- 편집 응답의 `warnings`는 `Run(ship, focus: 변경 부품)`의 결과다.

```csharp
public static object Run(ShipConstruct ship, CelestialBody body, double altitude, bool vacuum)
{
    VesselDeltaV dv = ship.vesselDeltaV;
    if (dv == null || !dv.IsReady) { dv?.SetCalcsDirty(true, true); return new Dictionary<string, object> { ["ready"] = false }; }

    double g = body.gravParameter / Math.Pow(body.Radius + altitude, 2);
    double atm = vacuum || !body.atmosphere ? 0 : body.GetPressure(altitude) / 101.325;   // kPa → atm
    List<object> stages = dv.OperatingStageInfo.OrderByDescending(s => s.stage).Select(s =>
    {
        double isp = SituationIsp(s.enginesInStage, atm);              // atmosphereCurve.Evaluate(atm), 추력 가중
        double thrust = s.thrustVac * (isp / Math.Max(s.ispVac, 1e-6)); // 연료 유량 일정 가정(스톡 엔진)
        return (object)new Dictionary<string, object>
        {
            ["stage"] = s.stage, ["mass_start"] = s.startMass, ["mass_end"] = s.endMass,
            ["thrust"] = thrust, ["isp"] = isp, ["twr"] = thrust / (s.startMass * g),
            ["delta_v"] = isp * 9.80665 * Math.Log(s.startMass / Math.Max(s.endMass, 1e-6)),
            ["delta_v_vac"] = s.deltaVinVac, ["delta_v_asl"] = s.deltaVatASL, ["burn_time"] = s.stageBurnTime,
        };
    }).ToList();
    // com: 질량 가중 평균, cot/col: CenterOfThrustQuery / CenterOfLiftQuery 합산 (루트 로컬)
    // ... warnings: twr_below_1(첫 스테이지), col_ahead_of_com(CoL이 CoM보다 +Y), no_control(명령 모듈 없음)
}
```

```csharp
public static object Render(ShipConstruct ship, List<ViewSpec> views, int resolution, bool markers)
{
    Bounds b = CraftBounds(ship);                                    // 모든 부품 렌더러 경계(월드)
    GameObject go = new GameObject("KspBridgeRenderCamera");
    RenderTexture rt = new RenderTexture(resolution, resolution, 24, RenderTextureFormat.ARGB32) { antiAliasing = 4 };
    Texture2D tex = new Texture2D(resolution, resolution, TextureFormat.RGB24, false);
    try
    {
        Camera cam = go.AddComponent<Camera>();
        cam.clearFlags = CameraClearFlags.SolidColor;
        cam.backgroundColor = new Color(0.18f, 0.2f, 0.23f);
        cam.cullingMask = 1 << 0;                                    // 부품 레이어만 (U12에서 확인·조정)
        cam.fieldOfView = 30; cam.targetTexture = rt; cam.enabled = false;
        List<object> images = new List<object>();
        foreach (ViewSpec v in views)
        {
            Vector3 dir = Root(ship).transform.rotation * v.Direction();   // 루트 로컬 기준 방위·고도
            float dist = b.extents.magnitude / Mathf.Sin(cam.fieldOfView * 0.5f * Mathf.Deg2Rad) * 1.1f;
            cam.transform.position = b.center + dir * dist;
            cam.transform.LookAt(b.center, Root(ship).transform.up);
            cam.nearClipPlane = Math.Max(0.01f, dist - b.extents.magnitude * 2); cam.farClipPlane = dist + b.extents.magnitude * 2;
            cam.Render();
            RenderTexture.active = rt;
            tex.ReadPixels(new Rect(0, 0, resolution, resolution), 0, 0);
            if (markers) DrawMarkers(tex, cam, ship);                // CoM 노랑, CoT 빨강, CoL 파랑 점
            images.Add(new Dictionary<string, object> { ["view"] = v.Name, ["width"] = resolution, ["height"] = resolution, ["png"] = Convert.ToBase64String(tex.EncodeToPNG()) });
        }
        return new Dictionary<string, object> { ["images"] = images };
    }
    finally { RenderTexture.active = null; UnityEngine.Object.Destroy(tex); rt.Release(); UnityEngine.Object.Destroy(rt); UnityEngine.Object.Destroy(go); }
}
```

```csharp
public static object Save(ShipConstruct ship, string name, bool overwrite)
{
    if (name.IndexOfAny(Path.GetInvalidFileNameChars()) >= 0) throw ApiException.BadRequest("invalid craft name");
    string path = CraftPath(EditorDriver.editorFacility, name);
    if (File.Exists(path) && !overwrite) throw new ApiException(409, "name_exists", name + " already exists; set overwrite=true");
    List<Dictionary<string, object>> errors = StructureCheck.Run(ship).Where(i => (string)i["severity"] == "error").ToList();
    if (errors.Count > 0) throw new ApiException(409, "unsaveable", "fix structure errors first: " + string.Join(", ", errors.Select(e => e["code"])));
    ship.shipName = name;
    EditorLogic.fetch.shipNameField.text = name;
    ShipConstruction.SaveShipToPath(HighLogic.SaveFolder, EditorDriver.editorFacility, "", name);
    return new Dictionary<string, object> { ["path"] = path, ["warnings"] = StructureCheck.Run(ship).Where(i => (string)i["severity"] == "warning").ToList() };
}

static string CraftPath(EditorFacility f, string name) =>
    Path.Combine(ShipConstruction.GetShipsPathFor(HighLogic.SaveFolder, f), name + ".craft");
```

```csharp
public static object Launch(ShipConstruct ship, string site)
{

    if (!EditorDriver.ValidLaunchSite(site)) throw ApiException.BadRequest("invalid launch site: " + site);
    if (StructureCheck.Run(ship).Any(i => (string)i["severity"] == "error")) throw new ApiException(409, "unsaveable", "fix structure errors first");
    EditorLogic.fetch.launchVessel(site);       // 게임이 저장·크루 배정·씬 전환 처리
    return new Dictionary<string, object> { ["accepted"] = true, ["site"] = site };
}
```

- 게임이 띄우는 확인 대화상자(크루 미배정, 자금 부족 등)는 그대로 둔다. 응답에 "게임 화면 확인 필요"를 쓰지 않고, 결과는 `get_status`의 `scene=FLIGHT`로 확인한다.
- **검증**: 게임: VAB U5 기체 `launch_craft` → `get_status` 폴링으로 `scene=FLIGHT`, `has_active_vessel=true`. SPH 기체는 Runway. 잘못된 site는 400.
- **선행 단위**: U13
