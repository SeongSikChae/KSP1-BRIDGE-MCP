    readonly StringBuilder _sb = new StringBuilder(); bool _needComma;
    public JsonWriter BeginObject() { Sep(); _sb.Append('{'); _needComma = false; return this; }

### U3. Mod — 서버·디스패처·애드온·/api/status
**대상**: `KspBridgeAddon.cs`, `Settings.cs`, `HttpServer.cs`, `MainThreadDispatcher.cs`, `ApiException.cs`

**코드 초안**:
```csharp
[KSPAddon(KSPAddon.Startup.Instantly, true)]
public sealed class KspBridgeAddon : MonoBehaviour {
    HttpServer _server; readonly MainThreadDispatcher _dispatcher = new MainThreadDispatcher();
    void Awake() {
        DontDestroyOnLoad(this);
        var s = Settings.Load(KSPUtil.ApplicationRootPath + "GameData/KspBridge/settings.cfg");
        _server = new HttpServer(s.Port, _dispatcher, Routes.Build());
        try { _server.Start(); Debug.Log($"[KspBridge] listening on 127.0.0.1:{s.Port}"); }
        catch (Exception e) { Debug.LogError("[KspBridge] start failed: " + e); }
    }
    void Update() => _dispatcher.Pump();
    void OnDestroy() => _server?.Stop();
}

public sealed class MainThreadDispatcher {
    readonly ConcurrentQueue<WorkItem> _q = new ConcurrentQueue<WorkItem>();
    public object Invoke(Func<object> fn, TimeSpan timeout) {
        var w = new WorkItem(fn); _q.Enqueue(w);
        if (!w.Done.Wait(timeout)) { w.Cancelled = true; throw new ApiException(504, "timeout", "main thread did not respond in 5s"); }
        if (w.Error != null) throw w.Error; return w.Result;
    }
    public void Pump() { while (_q.TryDequeue(out var w)) { if (w.Cancelled) continue;
        try { w.Result = w.Fn(); } catch (Exception e) { w.Error = e; } finally { w.Done.Set(); } } }
}
// HttpServer: HttpListener("http://127.0.0.1:{port}/"), 수신 스레드 → 요청마다 ThreadPool.QueueUserWorkItem
// 라우트: (method, 정규식 경로) → Func<RequestContext, object>; 핸들러는 dispatcher.Invoke 안에서 실행
// ApiException(status, code, msg) → {"ok":false,"error":{...}}, 그 외 예외 → 500 internal
// Guard.ActiveVessel(): HighLogic.LoadedScene != GameScenes.FLIGHT → 409 not_in_flight,
//                        FlightGlobals.ActiveVessel == null → 409 no_active_vessel
```
**검증**: KSP 실행 → `KSP.log`에 `[KspBridge] listening` / 메인 메뉴에서 `curl http://127.0.0.1:8085/api/status` → `scene:"MAINMENU"`, `/api/vessel` → 409 `not_in_flight`, 없는 경로 → 404
**선행**: U2

```csharp
internal sealed class AppConfiguration {
    [Property(PropertyType.STRING, required: true)] public string ContentRootPath { get; set; } = null!;
    [Property(PropertyType.STRING, required: true)] public string BaseUrl { get; set; } = null!;   // http://127.0.0.1:8085
    [Property(PropertyType.INTEGER, required: true)] public int TimeoutSeconds { get; set; }
}

public sealed class KspBridgeClient(HttpClient http) {
    public async Task<string> SendAsync(HttpMethod method, string path, object? body = null, CancellationToken ct = default) {
        using var req = new HttpRequestMessage(method, path) { Content = body is null ? null : JsonContent.Create(body) };
        HttpResponseMessage res;
        try { res = await http.SendAsync(req, ct); }
        catch (HttpRequestException) { return "error: KSP가 실행 중이 아니거나 KspBridge 모드가 로드되지 않았습니다."; }
        using var doc = JsonDocument.Parse(await res.Content.ReadAsStringAsync(ct));
        var root = doc.RootElement;
        return root.GetProperty("ok").GetBoolean() ? root.GetProperty("data").GetRawText()
            : $"error: {root.GetProperty("error").GetProperty("code").GetString()}: {root.GetProperty("error").GetProperty("message").GetString()}";
    }
}

[McpServerToolType]
public sealed class FlightTools(KspBridgeClient client) {
    [McpServerTool(Name = "set_throttle"), Description("Set main throttle (0.0~1.0) of the active vessel.")]
    public Task<string> SetThrottle([Description("0.0~1.0")] double value, CancellationToken ct)
        => client.SendAsync(HttpMethod.Post, "api/throttle", new { value }, ct);
    // get_status, get_vessel_state, activate_next_stage, set_sas, set_sas_mode, set_rcs, set_action_group,
    // list_maneuver_nodes, add_maneuver_node(ut?, inSeconds?, prograde, normal, radial), remove_maneuver_node(index)
}

// Program.CreateHostApplication 추가분
builder.Services.AddHttpClient<KspBridgeClient>(c => {
    c.BaseAddress = new Uri(appConfiguration.BaseUrl.TrimEnd('/') + "/");
    c.Timeout = TimeSpan.FromSeconds(appConfiguration.TimeoutSeconds); });
builder.Services.AddMcpServer().WithStdioServerTransport().WithTools<FlightTools>();
```

**테스트 (MSTest)** — 가짜 `HttpMessageHandler`로 요청 기록:
- 각 도구 → 메서드·경로·본문 JSON 일치 (`set_throttle(0.5)` → `POST /api/throttle {"value":0.5}` 등)
- `ok:false` 응답 → `error: code: message`
- 연결 실패(`HttpRequestException`) → 안내 메시지
- `add_maneuver_node` 인자 검증

## 검토 반영 (2026-11-07, 독립 검토 후 회귀)

### R1. 디스패처 타임아웃 경쟁·해제·스택
**대상**: `src/KspBridge.Mod/MainThreadDispatcher.cs`

**코드 초안** — 상태를 CAS로 전이해 타임아웃과 실행이 겹치지 않게 한다:
```csharp
const int Pending = 0, Running = 1, Cancelled = 2;
// Invoke
if (!item.Done.Wait(timeout)) {
    if (Interlocked.CompareExchange(ref item.State, Cancelled, Pending) == Pending) {
        item.Done.Dispose(); throw new ApiException(504, "timeout", ...);
    }
    item.Done.Wait();                 // 이미 실행 중 → 결과를 끝까지 받는다
}
item.Done.Dispose();
if (item.Error != null) ExceptionDispatchInfo.Capture(item.Error).Throw();
// Pump
if (Interlocked.CompareExchange(ref item.State, Running, Pending) != Pending) continue;
```
`FailAll`도 같은 CAS(Pending→Cancelled)로 오류를 설정한다.
**검증**: 빌드 경고 0, 게임에서 status·vessel·throttle 정상 응답
**선행**: 없음
