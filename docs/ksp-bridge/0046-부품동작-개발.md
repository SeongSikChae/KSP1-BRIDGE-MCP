```csharp
public static class ResourceTransfer
{
    private sealed class Side { public uint Id; public string Name; public double Amount, Max; public bool Flow; }

    private static bool _active;
    private static int _seq, _id;
    private static string _resource;
    private static int _resId, _n;            // _n: sources with amount > 0 at start (game transferFraction, fixed)
    private static double? _requested;
    private static double _transferred, _lastUT;
    private static double? _startUT, _endUT;
    private static string _reason;
    private static uint _vesselId;
    private static List<Side> _sources = new List<Side>(), _targets = new List<Side>();

    public static object Start(Vessel v, RequestBody b)
    {
        string name = b.GetString("resource");
        List<uint> src = Ids(b, "source_part_ids"), dst = Ids(b, "target_part_ids");
        double? amount = null;
        if (b.Has("amount"))
        {
            double a = b.GetDouble("amount");
            if (!(a > 0) || double.IsInfinity(a)) throw ApiException.BadRequest("amount must be a number > 0");
            amount = a;
        }
        if (src.Concat(dst).Distinct().Count() != src.Count + dst.Count)
            throw ApiException.BadRequest("a part id is listed twice (source and target lists must not overlap)");
        if (_active) throw new ApiException(409, "transfer_running", "transfer " + _id + " is running; stop_resource_transfer first");
        List<PartResource> s = src.Select(id => Row(v, id, name)).ToList(), t = dst.Select(id => Row(v, id, name)).ToList();
        PartResourceDefinition def = s[0].info;
        if (def.resourceTransferMode == ResourceTransferMode.NONE)
            throw new ApiException(409, "not_transferable", name + " cannot be transferred (resourceTransferMode NONE)");
        if (!GameVariables.Instance.UnlockedFuelTransfer(ScenarioUpgradeableFacilities.GetFacilityLevel(SpaceCenterFacility.ResearchAndDevelopment)))
            throw new ApiException(409, "transfer_not_unlocked", "resource transfer needs a higher Research and Development level");
        if (HighLogic.CurrentGame.Parameters.CustomParams<GameParameters.AdvancedParams>().ResourceTransferObeyCrossfeed)
        {
            List<PartResource> all = s.Concat(t).ToList();
            foreach (PartResource a in all)
                foreach (PartResource c in all)
                    if (a != c && !a.part.CanCrossfeed(c.part, def.id))
                        throw new ApiException(409, "no_crossfeed", name + " cannot cross-feed from " + Label(a.part) + " to " + Label(c.part)
                            + " (difficulty setting: resource transfer obeys crossfeed rules)");
        }
        PartResource blocked = s.FirstOrDefault(r => r.flowMode != PartResource.FlowMode.Out && r.flowMode != PartResource.FlowMode.Both)
            ?? t.FirstOrDefault(r => r.flowMode != PartResource.FlowMode.In && r.flowMode != PartResource.FlowMode.Both);
        if (blocked != null)
            throw new ApiException(409, "flow_mode_blocked", name + " in " + Label(blocked.part) + " has flow mode " + blocked.flowMode);
        if (!PartStates.Unlocked(ControlTypes.TWEAKABLES_ANYCONTROL))
            throw new ApiException(409, "transfer_locked", "resource transfer is not possible now: input_locked (TWEAKABLES_ANYCONTROL)");
        int full = s.Count(r => r.amount > 0), open = t.Count(r => r.amount < r.maxAmount);
        if (full == 0 || open == 0)
            throw new ApiException(409, "nothing_to_transfer", full == 0 ? "every source is empty" : "every target is full");
        int n = s.Count == 1 ? open : full;   // Out button (one source) splits over open targets, In button over non-empty sources (design 2)
        // 상태 초기화 → _active = true, _id = ++_seq, _lastUT = _startUT = UT, _vesselId = v.persistentId
        // PartStates.Log("resource_transfer_start", {id, resource, requested, source_part_ids, target_part_ids}, "resource_transfer " + _id + " start ...")
        return Describe(null);
    }

    private static PartResource Row(Vessel v, uint id, string name)
    {
        Part p = v.parts.FirstOrDefault(x => x.persistentId == id) ?? throw PartStates.PartNotFound(id);
        PartResource r = p.Resources[name];
        if (r == null || r.maxAmount == 0 || !r.isVisible) throw PartStates.ResourceNotFound(p, name);   // flight resource row (design 0043)
        return r;
    }

    public static void Tick()
    {
        if (!_active) return;
        try
        {
            Vessel v = FlightGlobals.ActiveVessel;
            if (v == null || v.persistentId != _vesselId) { Stop("vessel_changed"); return; }
            List<PartResource> s = Resolve(v, _sources), t = Resolve(v, _targets);
            if (s == null || t == null) { Stop("part_lost"); return; }
            double ut = Planetarium.GetUniversalTime(), dt = Math.Round(ut - _lastUT, 2);
            _lastUT = ut;
            foreach (PartResource to in t)
            {
                double rate = to.maxAmount / 20.0 / _n * dt;
                foreach (PartResource from in s)
                {
                    double val = Math.Min(Math.Min(rate, from.amount), to.maxAmount - to.amount);
                    if (_requested != null) val = Math.Min(val, _requested.Value - _transferred);
                    if (val <= 0) continue;
                    double fromBefore = from.amount, toBefore = to.amount;
                    from.part.TransferResource(from, -val, to.part);          // take first: nothing is created (design 10)
                    double taken = fromBefore - from.amount;
                    if (taken <= 0) continue;
                    to.part.TransferResource(to, taken, from.part);
                    double moved = to.amount - toBefore;
                    if (taken - moved > 0) from.amount += taken - moved;   // give back what did not fit, directly: TransferResource refuses + on an Out-only source
                    if (Math.Abs((fromBefore - from.amount) - moved) > 1e-9) { Stop("error", "source and target amounts moved differently"); return; }
                    _transferred += moved;
                }
            }
            Snapshot(s, t);
            if (_requested != null && _transferred >= _requested.Value - 1e-9) Stop("done");
            else if (s.All(r => r.amount <= 0)) Stop("source_empty");
            else if (t.All(r => r.amount >= r.maxAmount)) Stop("target_full");
        }
        catch (Exception e)
        {
            UnityEngine.Debug.LogError("[KspBridge] resource transfer: " + e);
            Stop("error", e.Message);
        }
    }

    /// <summary>Never throws (whole body guarded: BridgeLogic.Stop calls it before Coordinator.Stop). Ends a running transfer once.</summary>
    public static void Stop(string reason, string error = null)
    {
        if (!_active) return;
        _active = false;   // first: a throw below never leaves the transfer running
        _reason = reason;
        _endUT = _lastUT;  // replaced by the current UT below when it can be read
        try
        {
            // 비행 장면이면 _endUT = UT, 부품을 찾을 수 있으면 Snapshot (실패하면 마지막으로 읽은 값)
        }
        catch (Exception e) { UnityEngine.Debug.LogError("[KspBridge] resource transfer snapshot: " + e); }
        try
        {
            // PartStates.Log("resource_transfer_end", {id, resource, requested, transferred, reason, error}, "resource_transfer " + _id + " end " + reason ...)
        }
        catch (Exception e) { UnityEngine.Debug.LogError("[KspBridge] resource transfer end log: " + e); }   // the end event is missing only here
    }

    public static object StopRequest()
    {
        bool was = _active;
        Stop("stopped");
        return Describe(was);
    }
}
```

- `Ids(b, key)`: 키가 없거나 배열이 아니면 `b.GetList(key)`가 기존 문구(`'<key>' must be an array`)로 400을 던진다. 빈 배열이면 400 `"<key> must hold at least one part id"`, 원소는 `PartIdFromBody`와 같은 정수 규칙(`"<key> must hold integer part ids (persistentId) 0.." + uint.MaxValue`).
- `Resolve(v, sides)`: 부품마다 `v.parts`에서 id로 찾고 `State != DEAD`이며 자원이 그대로 있으면 그 `PartResource`, 하나라도 없으면 null.
- `Snapshot`: `Side.Amount`·`Max`·`Flow(flowState)`를 갱신한다. `Describe(stopped)`: 설계 상태 응답 키(`ut`는 `HighLogic.LoadedSceneIsFlight`일 때 `Planetarium.GetUniversalTime()`, 아니면 null; `stopped`는 null이 아닐 때만).
- `Status()`: `Describe(null)`. 진행 중이면 응답 전에 부품을 다시 읽어 `Snapshot`한다(못 찾으면 이전 값).

[McpServerTool(Name = "start_resource_transfer", Destructive = true)]
[Description("Start moving one resource between parts of the active vessel (docked vessels included) over time, like the part action window " +
    "Transfer Out (sources) / Transfer In (targets): with one source each target receives max_amount/20 per second split over the targets not full " +
    "at start (game Out button); with several sources each target receives max_amount/20 per second split over the sources not empty at start (In button); " +
    "time warp speeds it up. Flow locks (flow_enabled false) do not stop it, as in the game. amount = total the targets receive; omit to run until " +
    "every source is empty or every target is full. One transfer at a time; allowed while a flight plan runs. Returns the get_resource_transfer state. " +
    "Errors: not_in_flight, no_active_vessel, bad_request, transfer_running, part_not_found, resource_not_found, not_transferable, transfer_not_unlocked " +
    "(R&D level), no_crossfeed (difficulty setting obeys crossfeed), flow_mode_blocked, transfer_locked (input locked or no control), nothing_to_transfer.")]
public Task<string> StartResourceTransfer([Description("Resource name, e.g. LiquidFuel")] string resource,
    [Description("Part ids (persistentId) to take from")] double[] source_part_ids,
    [Description("Part ids (persistentId) to fill")] double[] target_part_ids,
    [Description("Total amount (units) to move; omit for as much as possible")] double? amount = null, CancellationToken ct = default)
    => client.SendAsync(HttpMethod.Post, "api/flight/resource_transfer", new { resource, source_part_ids, target_part_ids, amount }, ct);

    "max_amount, flow_enabled}]}; id null when none ran since the mod loaded.")]
public Task<string> GetResourceTransfer(CancellationToken ct = default)
    => client.SendAsync(HttpMethod.Get, "api/flight/resource_transfer", null, ct);

    => client.SendAsync(HttpMethod.Post, "api/flight/resource_transfer/stop", new { }, ct);   // as the other argument-less POST tools

- 기체 A(`KspBridge-Xfer`, 단위 1에서 만들어 발사대에 올린 것): 편집기에서 Clamp-O-Tron으로 결합한 탱크. 항목 1~10.

  - 목표 기체 `KspBridge-XferT`: 탐사선 코어, 배터리, 태양광 패널, FL-T400(LiquidFuel·Oxidizer 절반, editor `set_resource`), Clamp-O-Tron, 상승 단(엔진·탱크·디커플러).

  - 순서: `get_status` 확인 → 목표 기체 발사, 비행 계획으로 원궤도(약 80 km) → 추적 기체 발사, 같은 궤도면 원궤도 → `set_target` → `plan_maneuver` `rendezvous_phase`·`correct_approach`와 노드 실행 → 비행 계획 approach·dock 단계로 도킹 → 도킹 후 활성 기체에서 목표 FL-T400 → 추적 FL-T200 전달.

### 확인 항목

탱크 상태: 발사 시점은 A 가득, B·C·D 빈 상태다. 각 항목 전에 그 항목에 필요한 상태(원본에 잔량, 대상에 빈 용량)가 아니면 `get_status` 확인 후 `revert_flight`로 발사 시점 상태로 되돌린다(비행 중 자원량 직접 설정은 비범위). 되돌린 뒤 부품 id는 `get_vessel_parts`로 다시 읽는다.

1. 속도(In 규칙): A → B(LiquidFuel, amount 생략). 두 번 조회한 `transferred` 차이를 `ut` 차이로 나눈 값이 B 용량/20 초당(±5%)인지. 측정 뒤 `stop_resource_transfer`로 멈춘다.
2. 속도(Out 규칙): A → C·D(원본 1개, 대상 2개, 두 대상 모두 비운 상태). 대상마다 초당 자기 용량/20/2(±5%)인지. 측정 뒤 멈춘다.
3. `amount`: A → B 20 → `end_reason: done`, `transferred` 20(±1e-6), A·B 잔량 변화가 −20/+20.
4. 원본 여러 개: C·D → A(대상이 찰 때까지) → `target_full` 또는 `source_empty`, 잔량 합 보존.
5. 취소: 긴 전달 중 `stop_resource_transfer` → `stopped: true`, `end_reason: stopped`; 다시 호출 → `stopped: false`.
6. 흐름 잠금: B의 LiquidFuel을 `set_resource_flow` false로 둔 채 A → B 전달이 진행되고 응답 `flow_enabled: false`. 끝나면 true로 되돌린다.
7. 시간 가속: 발사대(LANDED)에서 긴 전달 중 `set_time_warp`로 on rails 가속 → 전달이 계속되고 끝 판정(`done` 또는 `source_empty`·`target_full`)과 잔량 합 보존(±1e-6)이 맞는지. 가속을 1배로 되돌린다.
8. 오류: `transfer_running`(진행 중 두 번째 시작), `nothing_to_transfer`(빈 원본), `not_transferable`(SolidFuel), `part_not_found`, `resource_not_found`(코어에 LiquidFuel), `bad_request`(겹침, 키 없음). `transfer_locked`는 `debug_input_lock`으로 `TWEAKABLES_ANYCONTROL`을 걸어 확인한다(debug 도구는 `debug.cfg`가 꺼져 있으면 사용자 승인을 받아 실측 동안만 켜고 끈다).
9. 중단
   - `part_lost`: B 쪽 Clamp-O-Tron을 `invoke_part_event` Decouple Node/Undock으로 떼어 내는 동안 A → B 전달 → `end_reason: part_lost`.
   - `vessel_changed`: 떼어 낸 B 기체로 `switch_vessel` 하기 전에 C → A 전달을 시작해 두고 전환 → `vessel_changed`.
   - `logic_stop`: 진행 중 `reload_mod` → 다시 읽은 로직의 `get_flight_events`에 `resource_transfer_end` `reason: logic_stop`이 남아 있는지(인계 버퍼), 새 로직 조회는 `id: null`.
   - `scene`: 진행 중 `get_status` 확인 후 `revert_flight` 또는 `leave_flight` → 장면 뒤 `get_resource_transfer`가 그 전달을 `active: false`, `end_reason: scene`으로 돌려주는지(결정 12: 상태는 `reload_mod` 뒤에만 없어진다).
10. 기록: `get_flight_events`에 `resource_transfer_start`·`resource_transfer_end`, KSP.log에 `[KspBridge] resource_transfer` 줄.
11. 실제 도킹 결합: 도킹 후 `get_vessel_parts`로 두 탱크가 한 기체에 있음을 확인하고, 목표 FL-T400 → 추적 FL-T200 LiquidFuel `amount` 30 → `done`, 전후 부품별 잔량이 −30/+30(±1e-6).

### 실측 중 발견(#24로 등록)

- 목표 기체 1회차 발사: 조종면 없는 구성이 공기역학 불안정으로 뒤집혔고 계획은 연료를 다 쓸 때까지 진행했다. 2회차 `tailfin` 조종면 + 짐벌 구성은 상승 중 흔들렸다.
- `approach` 단계가 `rendezvous_phase` 노드 직후 186 km 떨어진 곳에서 바로 `kill` 연소를 해 약 200 m/s를 낭비했다(에이전트가 중단).
- `execute_node`가 노드 약 200 s 전 가속을 멈춘 뒤 회전이 끝나도 다시 가속하지 않았고, 큰 각도 회전 때 자세 오차가 오르내렸다.
