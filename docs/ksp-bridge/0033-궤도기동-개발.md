```csharp
/// <summary>Burn time for dv across stages from VesselDeltaV (current stage down). Short = the stages cannot supply dv:
/// the missing part is timed at the last firing stage's average rate. +∞ only when no stage can fire (the no-thrust case).</summary>
public static double BurnTimeStaged(Vessel v, double dv, out bool shortDv)
{
    shortDv = false;
    VesselDeltaV d = v.VesselDeltaV;
    if (!PerformanceCheck.UpToDate(d))
        return BurnTime(v, dv);   // not computed yet: current engines only
    double left = dv, time = 0, lastRate = 0;
    for (int s = v.currentStage; s >= 0 && left > 0; s--)
    {
        DeltaVStageInfo info = d.GetStage(s);
        if (info == null || info.deltaVActual <= 0 || info.stageBurnTime <= 0) continue;
        double use = Math.Min(left, info.deltaVActual);
        lastRate = info.deltaVActual / info.stageBurnTime;   // m/s per s (단 안 평균)
        time += use / lastRate;
        left -= use;
    }
    if (left <= 1e-3) return time;
    if (lastRate <= 0) return BurnTime(v, dv);   // no stage data: current engines (+∞ = no thrust)
    shortDv = true;
    return time + left / lastRate;
}
```

  - **Δv 부족과 추력 없음 구분**: `BurnTimeStaged`는 Δv가 모자라도 유한한 시간과 `shortDv`를 돌려준다. +∞는 쏠 수 있는 단이 없을 때만 나온다(검토 R1-1-B-2, R1-2-B-1, R1-3-B-1). `circularize`는 지금 `BurnTime`을 그대로 쓴다(방안 결정 10).
  - **`ManualControl` 공개 범위**: `RequireManual`과 `SignX`·`SignY`·`SignZ`·`DeadZone`을 internal로 바꾼다. 노드·target 변경 도구는 수동 조작과 같은 전제(복구 대기 아님, 계획 없음, EVA 아님, 조종 가능)를 쓴다. 의도적이다. 변경 도구는 비행 제어 입력과 같은 등급의 조작이기 때문이다(검토 R1-2-B-12). 조회 도구(`list_vessels`)는 쓰지 않는다.
  - **공통 상수**: `CircularizeMode`의 `LockDv`·`AttitudeGateDeg`를 `internal const`로 바꾸고 `ExecuteNodeMode`가 그 상수를 참조한다. 연소 로직은 공유하지 않는다. `CircularizeMode`에서 바뀌는 것은 상수의 접근 범위뿐이다(설계 결정 19, 검토 R2-2-B-6, R2-1-B-2).

```csharp
public static class ManeuverNodes
{
    public static PatchedConicSolver Solver(Vessel v)
    {
        PatchedConicSolver s = v.patchedConicSolver;
        if (s == null)
            throw new ApiException(409, "nodes_unavailable", "maneuver nodes are not available (tracking station level or no patched conics)");
        return s;
    }

    /// <summary>node.DeltaV is (radial, normal, prograde) in KSP's node frame.</summary>
    public static Dictionary<string, object> Describe(Vessel v, ManeuverNode n)
    {
        double ut = Planetarium.GetUniversalTime();
        Orbit next = n.nextPatch;   // KSP 예측(노드 뒤 궤도), 없을 수 있음
        double bt = EngineModel.BurnTimeStaged(v, n.DeltaV.magnitude, out bool shortDv);
        return new Dictionary<string, object>
        {
            ["ut"] = n.UT, ["time_to"] = n.UT - ut,
            ["prograde"] = n.DeltaV.z, ["normal"] = n.DeltaV.y, ["radial"] = n.DeltaV.x,
            ["dv"] = n.DeltaV.magnitude,
            ["burn_time"] = double.IsInfinity(bt) ? null : (object)bt,   // +∞ 는 직렬화하지 않는다(검토 R1-1-B-2)
            ["dv_short"] = shortDv,
            ["orbit_after"] = next == null ? null : OrbitSummary(next),
        };
    }

    public static object Add(Vessel v, RequestBody b)
    {
        ManualControl.RequireManual(v);
        PatchedConicSolver s = Solver(v);
        double ut = ResolveUt(v, v.orbit, b, "at", allowed: AllNonAuto, defaultAt: null);
        if (ut < Planetarium.GetUniversalTime())
            throw ApiException.BadRequest("ut is in the past");   // 검토 R1-3-B-17
        ManeuverNode n = s.AddManeuverNode(ut);
        n.DeltaV = new Vector3d(b.GetDouble("radial", 0), b.GetDouble("normal", 0), b.GetDouble("prograde", 0));
        s.UpdateFlightPlan();
        return List(v);
    }
    // Update: index 를 0..Count-1 로 검사(아니면 400) → ut 를 바꾸면 과거 ut 400(add 와 같음, 검토 R2-3-B-10)
    //         → 바꿀 필드만 반영 → n.OnGizmoUpdated(dv, ut) → UpdateFlightPlan

}
```

```csharp
/// <summary>at: apoapsis | periapsis | ascending_node | descending_node | ut (with "ut") | auto. Exactly one of at/ut.</summary>
public static double ResolveUt(Vessel v, Orbit o, RequestBody b, string name, string[] allowed, string defaultAt)
{
    string at = b.Has(name) ? b.GetString(name) : (b.Has("ut") ? "ut" : defaultAt ?? throw ApiException.BadRequest("at or ut required"));
    if (Array.IndexOf(allowed, at) < 0) throw ApiException.BadRequest("at " + at + " not allowed for this kind");
    double now = Planetarium.GetUniversalTime(), ut;
    switch (at)
    {
        case "ut": ut = b.GetDouble("ut"); break;
        case "apoapsis":
            if (o.eccentricity >= 1) throw ApiException.BadRequest("no apoapsis on a hyperbolic orbit");
            ut = now + o.timeToAp; break;
        case "periapsis": ut = now + o.timeToPe; break;
        case "ascending_node":
        case "descending_node":
            ut = NodeUt(o, ReferencePlaneNormal(v, o), at == "ascending_node", now);   // target 궤도면 또는 적도면
            break;
        default: throw ApiException.BadRequest("at " + at + " is resolved by the kind");   // auto
    }
    if (double.IsNaN(ut) || double.IsInfinity(ut)) throw new ApiException(409, "no_solution", "node time is undefined");
    return ut;
}

// NodeUt: 두 궤도면 사이 각(상대 경사)이 0.01° 미만이면 교점이 사실상 정의되지 않으므로 409 no_solution.
// 판정은 각도로 한다: Vector3d.Angle(n1, n2) < 0.01 (외적 크기 기준을 쓰지 않는다, 검토 R2-1-B-1, R2-3-B-8).
// 쌍곡선에서 교점 진근점이각이 점근선 밖이면 GetUTforTrueAnomaly 가 NaN → 위 NaN 검사(검토 R1-3-B-8).
```

- **검증**

  - 실측(기체 O, 80 km 궤도)

    - `remove`(`all`) 뒤에는 `maneuver_nodes`가 비어 있다.

  - 교점 미정의(상대 경사 < 0.01°)의 409 `no_solution`은 발사 궤도로 조건을 만들 수 없어 실측하지 않는다. 대신 경사가 있는 궤도에서 `at: ascending_node`·`descending_node` 노드의 `time_to` 차이가 반 주기 ± 1%인지 확인한다. 이 오류 경로는 미검증으로 보고한다.

## 단위 3: `execute_node` Mode

- **대상**: `ExecuteNodeMode`(새 파일), `Plan.cs`의 Mode 분기·`default` 메시지
- **코드 초안**:

```csharp
public sealed class ExecuteNodeMode : IFlightMode
{
    public static readonly string[] Args = { "tolerance", "late_limit_s" };

    public static Func<IFlightMode> Parse(Dictionary<string, object> o, string path)
    {
        double tol = Plan.OptionalNumber(o, "tolerance", path, 0.01, 100, 0.2);       // 설계 결정 18
        double late = Plan.OptionalNumber(o, "late_limit_s", path, 0, 3600, 10);      // 설계 결정 17
        return () => new ExecuteNodeMode(tol, late);
    }

    public void Begin(Vessel v, Telemetry t)
    {
        ManeuverNode n = FirstNode(v) ?? throw new ModeAbortException("no_node", "no maneuver node to execute");   // 설계 결정 6·33
        _nodeUt = n.UT;
        _dvPlanned = n.DeltaV.magnitude;
    }

    public ModeCommand Update(Vessel v, Telemetry t)
    {
        ManeuverNode n = FirstNode(v);
        if (n == null || Math.Abs(n.UT - _nodeUt) > 1)   // 지워졌거나 MCP 밖에서 바뀜(검토 R1-3-B-14)
            throw new ModeAbortException("no_node", n == null ? "maneuver node removed" : "maneuver node changed");
        double ut = Planetarium.GetUniversalTime();
        Vector3d burn = n.GetBurnVector(v.orbit);   // world, remaining Δv
        double dvMag = burn.magnitude;
        AttitudeTarget att = AttitudeTarget.Vector(burn.normalized);
        if (!_burning)
        {
            double bt = EngineModel.BurnTimeStaged(v, dvMag, out bool shortDv);
            if (double.IsInfinity(bt))
                return new ModeCommand(att, 1);   // 쏠 단이 없음: 추력 대기·no_thrust 는 Coordinator 판단(기존과 같음)
            double start = n.UT - bt / 2 - EngineModel.SpoolUpS(v) / 2;
            if (ut < start)
                return new ModeCommand(att, 0) { WarpToUt = start };
            double lateS = ut - start;
            if (lateS > _late)
                throw new ModeAbortException("node_missed", "burn start passed " + F(lateS, 1) + " s ago");
            _burning = true;
            _startInfo = (dvMag, bt, lateS, shortDv);
        }
        if (!_startReported && !FlightEvents.InTrial)   // trial 인스턴스가 그대로 실제 Mode 가 되므로 첫 실제 Tick 에 보고(검토 R2-2-B-3)
        {
            ReportStart(v, t, _startInfo);   // node_burn_start, shortDv 이면 insufficient_dv 1회(설계 결정 20)
            _startReported = true;
        }
        // 연소 중: CircularizeMode 83~145줄과 같은 규칙을 이 Mode 에 따로 쓴다(상수만 공유, 설계 결정 19)
        if (_lockedDir == null && dvMag < CircularizeMode.LockDv) _lockedDir = burn.normalized;
        bool fired = t.PrevCommandThrottle > 0 && t.Num["thrust"] > 0;
        string reason = dvMag <= EngineModel.TailoffDv(v) + _tolerance ? "dv"
            : _lockedDir != null && fired && dvMag > _prevDv ? "dv_growing" : null;
        if (_lockedDir != null && fired) _prevDv = dvMag;
        if (reason != null)
        {
            ReportDone(v, dvMag, ut, reason);   // node_burn_done
            n.RemoveSelf();
            Done = true;
            return new ModeCommand(null, 0);
        }
        AttitudeTarget a = AttitudeTarget.Vector(_lockedDir ?? burn.normalized);
        double accel = EngineModel.MaxAccel(v);
        double throttle = accel <= 0 ? 1 : Clamp01(dvMag / (accel * 2));   // 마지막 2 s 분량에서 감소
        double err = t.Num["attitude_error"];
        double gate = _lockedDir != null ? LockedGateDeg /* 1° */ : CircularizeMode.AttitudeGateDeg;   // 설계 결정 19, 단위 3 실측
        if (double.IsNaN(err) || err > gate) throttle = 0;
        return new ModeCommand(a, throttle);
    }

    public void End(Vessel v) { }
}
```

  - **trial 동작**: 제출 trial은 `BeginStep`과 첫 `Update`를 실제로 돌린다(Coordinator.cs 180~183줄). 그래서 첫 step의 `no_node`·`node_missed`는 409로 제출이 거부되고, 둘째 step 이후는 실행 시점에 `plan_aborted`가 된다(설계 결정 33). `ReportStart`는 `FlightEvents.InTrial`이면 기록하지 않는다(LandMode 패턴).
  - **`insufficient_dv`**: 필드는 `dv_node`, `dv_available`(`t.Num["total_dv"]`, NaN이면 null)이다. `shortDv`일 때 1회 남긴다. 1회 기록과 Save/Load는 `LandMode._infeasible` 패턴을 따른다.
  - **결과 이벤트**: `node_burn_done`에 `dv_planned`, `dv_executed`(계획 − 남은 Δv), `burn_s`, `dv_remaining`, `orbit_after`(`get_flight.orbit`과 같은 필드)를 남긴다.

```csharp
private static readonly string[] Apsides = { "apoapsis", "periapsis", "ut" };   // 설계 결정 13(ut 포함)

public static object Plan(Vessel v, RequestBody b)
{
    ManualControl.RequireManual(v);
    string kind = b.GetString("kind");
    bool append = b.GetBool("append", false);
    PatchedConicSolver s = ManeuverNodes.Solver(v);
    Orbit o = v.orbit;
    if (append && s.maneuverNodes.Count > 0)
        o = s.maneuverNodes[s.maneuverNodes.Count - 1].nextPatch
            ?? throw new ApiException(409, "no_solution", "last node has no orbit after it");   // 검토 R1-3-B-13
    List<NodeSpec> nodes;   // (ut, world Δv)
    switch (kind)
    {
        case "change_apoapsis":
            nodes = One(OrbitMath.ChangeApsis(o, ManeuverNodes.ResolveUt(v, o, b, "at", Apsides, "periapsis"), TargetR(o, b), raiseApo: true)); break;
        case "change_periapsis":
            nodes = One(OrbitMath.ChangeApsis(o, ManeuverNodes.ResolveUt(v, o, b, "at", Apsides, "apoapsis"), TargetR(o, b), raiseApo: false)); break;
        case "circularize":
            nodes = One(OrbitMath.Circularize(o, ManeuverNodes.ResolveUt(v, o, b, "at", Apsides, "apoapsis"))); break;
        case "change_inclination":
            nodes = One(OrbitMath.ChangeInclination(v, o, b.GetDouble("inclination"), b.Has("at") ? b.GetString("at") : "auto")); break;
        case "hohmann":
            nodes = OrbitMath.Hohmann(o, TargetR(o, b), HohmannUt(v, o, b)); break;   // 2개(설계 결정 11), auto = 지금 + 60 s
        // 단위 5: transfer_body, capture / 단위 7: rendezvous_phase, correct_approach
        default: throw ApiException.BadRequest("unknown kind " + kind);
    }
    OrbitMath.CheckFinite(nodes);                    // NaN·∞ → 409 no_solution
    if (!append) ManeuverNodes.Clear(s);
    foreach (NodeSpec n in nodes) ManeuverNodes.AddWorld(v, s, n.Ut, n.DvWorld);
    s.UpdateFlightPlan();
    return ManeuverPlanner.Result(v, s, kind);       // 설계 결정 16: 노드, 총 Δv, 연소 시간, dv_short, kind 별 예측
}

/// <summary>altitude → radius; 400 below the surface (or atmosphere top) or beyond the SOI (설계 결정 13).</summary>
private static double TargetR(Orbit o, RequestBody b)
{
    CelestialBody body = o.referenceBody;
    double alt = b.GetDouble("altitude");
    double floor = body.atmosphere ? body.atmosphereDepth : 0;
    if (alt <= floor || body.Radius + alt >= body.sphereOfInfluence)
        throw ApiException.BadRequest("altitude out of range (" + floor + " .. SOI)");
    return body.Radius + alt;
}
```

```csharp
// OrbitMath: 노드 시각의 속도 벡터를 바꾸는 해석식 (Orbit frame 벡터는 .xzy 로 world 변환 — CircularizeMode 와 같음)
public static NodeSpec ChangeApsis(Orbit o, double ut, double newR, bool raiseApo)
{
    Vector3d r = o.getRelativePositionAtUT(ut), vel = o.getOrbitalVelocityAtUT(ut);
    double rNow = r.magnitude;
    if (raiseApo ? newR < rNow : newR > rNow)   // 원지점은 노드 지점보다 낮을 수 없고, 근지점은 높을 수 없다(검토 R1-3-B-9)
        throw ApiException.BadRequest((raiseApo ? "apoapsis below" : "periapsis above") + " the node point radius " + F(rNow - o.referenceBody.Radius, 0));
    double a = (rNow + newR) / 2;   // 노드 지점이 반대쪽 장축 끝(근사: 노드 지점 속도를 수평으로 만든다)
    double vNew = Math.Sqrt(o.referenceBody.gravParameter * (2 / rNow - 1 / a));
    Vector3d horiz = Vector3d.Exclude(r, vel).normalized;
    return new NodeSpec(ut, (horiz * vNew - vel).xzy);
}

public static List<NodeSpec> Hohmann(Orbit o, double r2, double ut1)
{
    double mu = o.referenceBody.gravParameter, r1 = o.getRelativePositionAtUT(ut1).magnitude;
    double at = (r1 + r2) / 2;
    NodeSpec burn1 = ChangeApsis(o, ut1, r2, raiseApo: r2 > r1);
    double ut2 = ut1 + Math.PI * Math.Sqrt(at * at * at / mu);   // 반 주기 뒤
    return new List<NodeSpec> { burn1, Circularize(After(o, burn1), ut2) };   // After: Orbit.UpdateFromStateVectors
}

// ChangeInclination: at 이 auto 면 두 교점 중 반지름이 큰 쪽(Δv 작음). 교점에서 속도를 반지름축으로 Δi 회전 → Δv = vRot − v.
// 기준면과 상대 경사가 0.01° 미만이면 어느 지점에서나 같은 기동이므로 auto 는 지금 + 60 s 로 둔다(검토 R2-3-B-8).
// 기준면은 OrbitMath.ReferencePlaneNormal(target 궤도면 또는 적도면, 설계 결정 9, 검토 R3-1-B-6).
```

## 단위 5: 천체 전이(`transfer_body`, `capture`)와 `soi_changed`

- **대상**: `OrbitMath.TransferBody`·`Capture`, `FlightScene`의 `onVesselSOIChanged` 훅, `soi_changed` 이벤트

```csharp
/// <summary>Hohmann to the body's orbit radius at the phase window, then golden-section searches (Δv, then UT ±10% of the
/// period, 40 iterations each, 2 s budget) so the periapsis inside the body's SOI meets periapsis.</summary>
public static NodeSpec TransferBody(Orbit o, CelestialBody body, double periapsisAlt)
{
    if (o.eccentricity > 0.1)
        throw new ApiException(409, "no_solution", "circularize first (eccentricity " + F(o.eccentricity, 3) + ")");   // r1 근사 전제
    double mu = o.referenceBody.gravParameter, r1 = o.semiMajorAxis, r2 = body.orbit.semiMajorAxis;
    double tTransfer = Math.PI * Math.Sqrt(Math.Pow((r1 + r2) / 2, 3) / mu);
    double lead = Math.PI - tTransfer * 2 * Math.PI / body.orbit.period;    // 도착 때 천체가 있을 위상
    double ut0 = PhaseWindowUt(o, body.orbit, lead);
    NodeSpec best = ChangeApsis(o, ut0, r2, raiseApo: true);
    best = Refine(o, best, n => SoiPeriapsisError(o, n, body, periapsisAlt));
    if (double.IsInfinity(SoiPeriapsisError(o, best, body, periapsisAlt)))
        throw new ApiException(409, "no_encounter", "no SOI entry of " + body.bodyName + " found");   // 검토 R1-3-B-7
    return best;
}
```

```csharp
public sealed class ApproachMode : IFlightMode
{
    public static readonly string[] Args = { "distance", "max_speed" };
    private const double HoldS = 3;             // 완료 유지 시간(설계 결정 21, 초기값)
    private const double DoneSpeed = 0.2;
    private const double PortPointTolM = 2;     // 포트 목표점 허용 거리
    private const double EngineErr = 2;         // 주엔진/RCS 전환(설계 결정 23)
    private const double MaxClosestM = 10000;   // 예측 최근접 거리 상한 → target_too_far (설계 결정 22)
    private const double Margin = 0.6;          // 접근 속도 여유 계수(설계 결정 24)
    private const double KillLeadS = 10;        // 대기 종료 여유: 자세 정렬·Warp 해제(검토 R1-3-B-3)
    private const double ClearanceM = 5;        // 목표 기체 반경 밖 여유(포트 목표점 우회)

    public static Func<IFlightMode> Parse(Dictionary<string, object> o, string path)
    {
        double distance = Plan.OptionalNumber(o, "distance", path, 1, 1000, 50);
        double maxSpeed = Plan.OptionalNumber(o, "max_speed", path, 0.5, 100, 10);
        bool? stepRcs = o.TryGetValue("rcs", out object r) ? r as bool? : null;   // trial 뒤 적용될 step rcs(설계 결정 23)
        return () => new ApproachMode(distance, maxSpeed, stepRcs);
    }

    public void Begin(Vessel v, Telemetry t)   // 검사만, 게임 상태 변경 없음. 탐색 1회는 trial·실행 시작 모두에서 돈다
    {
        Vessel tv = TargetVessel(v) ?? throw new ModeAbortException("no_target", "no target vessel");
        if (tv.mainBody != v.mainBody)
            throw new ModeAbortException("no_target", "target orbits " + tv.mainBody.bodyName);   // 검토 R2-3-B-6
        _targetId = tv.persistentId;
        double ut = Planetarium.GetUniversalTime(), span = OrbitMath.SafeSpan(v.orbit);
        if (double.IsNaN(span) || span <= 0)
            throw new ModeAbortException("target_too_far", "closest approach undefined on this orbit");
        (_closestUt, _closestDist) = OrbitMath.ClosestApproach(v.orbit, tv.orbit, ut, span);   // 시작 때 1회
        if (_closestDist > MaxClosestM)
            throw new ModeAbortException("target_too_far", "closest approach " + F(_closestDist, 0) + " m > " + MaxClosestM);   // 검토 R1-1-B-1
    }
    // ClosestApproach 의 예산 초과는 ApiException 이 아니라 결과 NaN 으로 돌려받아 위 target_too_far 로 바꾼다.
    // 그래서 둘째 step 이후에도 plan_aborted 사유가 정의된 목록 안에 있다(검토 R2-1-B-6, R2-3-B-6).

    public ModeCommand Update(Vessel v, Telemetry t)
    {
        Vessel tv = TargetVessel(v);
        if (tv == null || tv.persistentId != _targetId)
            throw new ModeAbortException("no_target", "target cleared or changed");   // 검토 R1-3-B-14
        double ut = Planetarium.GetUniversalTime();
        Vector3d goal = GoalPoint(v, tv);                      // 기체: CoM / 포트: 포트 축 앞 distance (우회 경유점 포함)
        Vector3d rel = goal - v.CoMD;
        Vector3d relV = v.obt_velocity - tv.obt_velocity;     // 내 상대 속도
        double d = rel.magnitude;

        if (_phase == Phase.Wait)                             // wait → kill → close (설계 결정 22)
        {
            double kill = EngineModel.BurnTimeStaged(v, relV.magnitude, out _);
            double wake = _closestUt - kill - EngineModel.SpoolUpS(v) - KillLeadS;   // 정지 거리 여유: 연소 전체 + 정렬
            if (ut < wake && Vector3d.Dot(rel, relV) > 0)
                return new ModeCommand(AttitudeTarget.Vector(-relV.normalized), 0) { WarpToUt = wake };
            Enter(Phase.Kill);
        }
        double aEngine = EngineModel.MaxAccel(v), aRcs = RcsAccel(v, _stepRcs);   // 0 이면 rcs_unavailable 1회(trial 에서는 남기지 않음)
        if (_phase == Phase.Kill)
        {
            if (relV.magnitude <= EngineErr) Enter(Phase.Close);
            else if (aEngine > 0) return EngineBurn(-relV, t);                // 상대 속도 상쇄(주엔진)
            else if (aRcs > 0) return new ModeCommand(AttitudeTarget.Vector(rel.normalized), 0) { Translate = ToolAxes(v, -relV / aRcs) };   // 검토 R2-3-B-5
        }

        if (aEngine <= 0 && aRcs <= 0)
            return new ModeCommand(AttitudeTarget.Vector(rel.normalized), 1);   // 추력 없음: Coordinator 의 대기·no_thrust 판단
        double a = aRcs > 0 ? aRcs : aEngine;
        double vWant = Math.Min(_maxSpeed, Margin * Math.Sqrt(2 * a * Math.Max(0, d - StopDist())));
        Vector3d err = rel.normalized * vWant - relV;         // 원하는 상대 속도 − 지금
        if (err.magnitude > EngineErr || aRcs <= 0)
            return EngineBurn(err, t);                        // 자세 err 방향, 5° 게이트, 스로틀 = err / (aEngine·1 s)
        return new ModeCommand(AttitudeTarget.Vector(rel.normalized), 0) { Translate = ToolAxes(v, err / aRcs) };
        // 완료: 목표점 도달(기체 d ≤ distance, 포트 d ≤ PortPointTolM) && relV ≤ DoneSpeed 가 HoldS 동안 → approach_done, Done
    }

    public void End(Vessel v) { }
}
```

```csharp
/// <summary>World vector → tool frame (x = right, y = -forward (dorsal), z = up (nose)) of v.ReferenceTransform,
/// each clamped to [-1, 1]; the same frame as rcs_translate (ManualControl.cs 302줄, 검토 R1-2-B-7).</summary>
internal static Vector3 ToolAxes(Vessel v, Vector3d w)
{
    Transform r = v.ReferenceTransform;
    return new Vector3(Clamp1(Vector3d.Dot(w, r.right)), Clamp1(-Vector3d.Dot(w, r.forward)), Clamp1(Vector3d.Dot(w, r.up)));
}
```

  - **target 해석**: `TargetVessel`은 `v.targetObject`가 `Vessel`이면 그 기체, `ModuleDockingNode`이면 `part.vessel`이다(설계 결정 21).
  - **`GoalPoint`**(포트 target)
    - 목표점 P = 포트 위치 + 포트 축 × `distance`다.
    - 자기 위치가 목표 기체 중심에서 축 뒤쪽(`Dot(self − CoM, axis) < 0`)이면 먼저 측면 경유점 S로 간다. S = CoM + n·(R + `ClearanceM`)이고, n은 축에 수직인 성분의 방향, R은 목표 기체 경계 반경이다. S에 들어온 뒤 P로 간다(설계 결정 21, 검토 R1-3-B-2).
  - **`RcsAccel(v, stepRcs)`**
    - RCS가 켜졌는지는 `stepRcs ?? v.ActionGroups[KSPActionGroup.RCS]`로 본다. step `rcs`가 trial 뒤에 적용되므로 그룹 상태를 직접 보지 않는다(설계 결정 23, 검토 R2-2-B-2, R2-3-B-3).
    - 켜졌고 모노프로펠런트가 있으면 활성 `ModuleRCSFX` 추력 합 / 질량, 그 밖은 0이다.
    - 0이면 `rcs_unavailable`(`reason: rcs_off`·`no_propellant`·`no_thrusters`)을 남기고 주엔진만 쓴다. `FlightEvents.InTrial`이면 남기지 않는다(검토 R1-3-B-12).
    - `RcsAccel`은 계산만 하는 정적 도우미이고 이벤트를 남기지 않는다. 이벤트와 1회 플래그는 부르는 Mode 인스턴스(`ApproachMode`·`DockMode` 각자의 `reported` 목록, Save/Load 대상)가 맡는다. 그래서 step마다 한 번 나오고, `dock`은 `dock_failed` 직전에 반드시 남긴다(검토 R3-1-B-3).
  - **`StopDist()`**: 완료 거리다. 기체 target이면 `distance`, 포트 target이면 0(목표점 자체가 포트 앞 `distance` 지점)이다.
  - **`approach_done.distance`**: 기체 target이면 두 기체 무게중심 사이 거리, 포트 target이면 자기 무게중심과 포트 목표점 사이 거리다. 이벤트에 `goal: vessel|port`를 같이 남긴다(검토 R2-1-B-10).
  - **진전 없음**: Mode 자체 시간 상한은 두지 않는다. 계획의 step `timeout_s`로 지정한다(0005의 기존 수단). 실측 계획에는 `timeout_s: 1800`을 쓴다.
  - **Save/Load 키**: `phase`, `target_id`, `closest_ut`, `closest_dist`, `hold_start`, `waypoint_done`, `reported`. reload 뒤에는 `Begin`을 다시 부르지 않으므로 시작 검사도 다시 하지 않는다(검토 R1-1-B-11).
- **검증**
  - 단위 7에서 최근접 1 km 아래인 상태에서, 아직 멀리 있을 때 계획 `[{mode: approach, rcs: true, timeout_s: 1800}]`을 낸다.
    - `approach_phase`가 `wait` → `kill` → `close` 순으로 나온다.
    - `approach_done`의 거리 ≤ 50, 상대 속도 ≤ 0.2다(완료 조건 "랑데부").
  - 포트 target: `set_target`(포트)로 다시 실행하면 `approach_done`의 거리가 포트 목표점 2 m 안이다.
  - target 없이 제출하면 409 `no_target`이다.
  - 최근접 거리 > 10 km 상태에서 제출하면 409 `target_too_far`다.
  - RCS 그룹을 끈 상태에서 `[{mode: approach, rcs: true}]`를 제출하면 받아들여지고, `rcs_unavailable` 이벤트가 나오지 않는다.
  - 연소 중 `reload_mod`를 해도 인계 뒤 완료되고, `approach_phase`가 같은 단계를 두 번 내지 않는다.

```csharp
        }
        if (Coordinator.DockCoupled && Coordinator.TryAdoptDocked(v))   // target 역참조보다 먼저(검토 R1-2-B-6)
            return new ModeCommand(null, 0);
        if (_tgt == null || _tgt.part == null || _tgt.vessel == null)
            throw new ModeAbortException(Coordinator.DockCoupled ? "dock_failed" : "no_target", "target port lost");   // 검토 R1-3-B-4
        double aRcs = ApproachMode.RcsAccel(v, _stepRcs);   // step rcs 기준(설계 결정 23)
        if (aRcs <= 0)
            throw new ModeAbortException("dock_failed", "no RCS translation available");   // rcs_unavailable 이벤트 뒤(설계 결정 27)

        Vector3d axis = _tgt.nodeTransform.forward;                         // 목표 포트 바깥 방향
        Vector3d off = _own.nodeTransform.position - _tgt.nodeTransform.position;
        double along = Vector3d.Dot(off, axis);
        Vector3d lateral = off - axis * along;
        Vector3d relV = v.obt_velocity - _tgt.vessel.obt_velocity;
        Vector3d want = PhaseVelocity(along, lateral);   // BackOut → Align → Advance → Contact (아래)
        return new ModeCommand(AttitudeTarget.Vector(-axis), 0) { Translate = ApproachMode.ToolAxes(v, (want - relV) / aRcs) };
    }

    public void End(Vessel v)   // 여러 번·v null 이어도 안전
    {
        Coordinator.WatchDock(0, false);
        if (_armed && v != null) RestoreControlPart(v, _controlBefore);   // 그 부품이 v 에 없으면 그대로 둔다(설계 결정 25)
        _armed = false;
    }
}
```

  - **`PhaseVelocity`의 단계**: 판정 순서는 위에서 아래다.
    - `Route`: `along ≤ 0`(목표 포트 평면 뒤)이면 `ApproachMode`의 측면 경유점 계산(`Waypoints.Side`, 목표 기체 반경 + 5 m)을 같이 써서 경유점 → 대기점(포트 앞 `HoldOffM`) 순으로 간다. 직진하지 않는다(설계 결정 26, 검토 R2-3-B-1, R2-1-B-8).
    - `BackOut`: `0 < along < HoldOffM / 2`이고 수직 오차 > `LateralStopM`이면 +axis로 0.5 m/s 물러난다(포트 앞쪽이라 목표 기체를 지나지 않는다).
    - `Align`: 수직 오차를 0.2 m/s·m 비례로 줄이고, 축 방향 속도는 0이다.
    - `Advance`: 수직 오차 ≤ `LateralGoM`이면 −axis로 `min(MaxSpeed, max(ContactSpeed, 0.1·(along − 1)))`을 낸다. 수직 오차 > `LateralStopM`이면 `Align`으로 돌아간다.
    - `Contact`: `0 < along ≤ ContactM`이면 접촉으로 보고 `dock_attempt`(시도 횟수, 접촉 속도, 수직 오차)를 남긴다.
  - **재시도**: 접촉 뒤 결합되지 않은 채 `along`이 1 m보다 커지면 실패 1회다. 대기점까지 물러나 다시 하고, `retries`(기본 2)를 넘기면 `dock_failed`다(설계 결정 27).
  - **`debug_dock_reject`**(설계 결정 34)
    - `DebugState`에 `DockRejectLeft` 카운터를 두고, `DebugTools`에 `debug_dock_reject(count)`를 더한다. 기존 `debug_*` 도구 패턴을 따르고 `debug_clear`로 해제한다.
    - 카운터가 0보다 크면 `Advance`가 `along` 2 m에서 멈춘다. 포트 자력 포착 범위(Clamp-O-Tron 약 0.5 m)보다 충분히 멀게 둔다(검토 R3-1-B-1).
    - 정지 상태(상대 속도 ≤ 0.05 m/s)가 5 s 유지되면 그 시도를 실패로 세고(`dock_attempt {forced: true}`) 카운터를 1 줄인다. 실제 실패와 같은 경로로 대기점(`HoldOffM`)까지 물러난 뒤 다시 정렬한다.
    - 실측 때 KSP.log에 대상 포트의 `acquireRange`를 남겨 2 m가 포착 범위 밖인지 확인한다.

```csharp
// Coordinator
private static uint _dockPort;         // dock step 의 자기 포트 part persistentId (0 = 감시 안 함)
public static bool DockCoupled { get; private set; }

/// <summary>portId 0 = stop watching. coupled restores a coupling seen before a reload (DockMode.Load keeps it; the first Tick calls this).</summary>
public static void WatchDock(uint portId, bool coupled) { _dockPort = portId; DockCoupled = portId != 0 && coupled; }

public static void OnCouple(Part from, Part to)
{
    if (_dockPort != 0 && (from?.persistentId == _dockPort || to?.persistentId == _dockPort)) DockCoupled = true;
}

/// <summary>Adopts the merged vessel only for a dock step whose own port coupled and is on the active vessel.</summary>
public static bool TryAdoptDocked(Vessel v)
{
    if (!Running || !DockCoupled || !(_mode is DockMode d) || v == null || !Contains(v, _dockPort)) return false;
    if (d.Done) return v.persistentId == _plan.VesselId;   // 이미 채택: MarkDocked·docked 이벤트는 한 번만(검토 R2-2-B-5)
    _plan.VesselId = v.persistentId;   // Handover.CapturePlan 이 vessel_id 로 저장
    AttitudeController.Retarget(v);    // 계획 전 SAS·RCS 기록 유지(검토 R1-2-B-4)
    d.MarkDocked(v);                   // docked 이벤트(id, 이름, 부품 수), Done = true
    return true;                       // _dockPort 는 step 종료(End)에서 0 으로
}

/// <summary>Our vessel being absorbed into the target during a watched dock: not a loss (검토 R1-2-B-3).</summary>
public static bool IsDockAbsorb(Vessel dying) => Running && _dockPort != 0 && DockCoupled && dying != null && dying.persistentId == _plan.VesselId;

// Tick 44줄
if (v == null || (v.persistentId != _plan.VesselId && !TryAdoptDocked(v)))
{
    Abort("vessel_changed", null, null);
    return;
}
// FlightScene.OnVesselChange
if (Coordinator.Running && !Coordinator.TryAdoptDocked(v)) Coordinator.Abort("vessel_changed", null, null);
// FlightScene.OnVesselDestroy
if (Coordinator.Running && v != null && v.persistentId == Coordinator.TargetVesselId && !Coordinator.IsDockAbsorb(v))
    Coordinator.Abort("vessel_lost", null, null);
// FlightScene.OnPartCouple: 지금 펄스 처리 뒤 Coordinator.OnCouple(a.from, a.to)
```

  - **결합 순서**: `onPartCouple`은 기체 병합 전에 온다. 그래서 `OnVesselDestroy`·`OnVesselChange` 시점에는 이미 `DockCoupled`가 true다. 이 순서는 실측에서 로그로 확인한다.
  - **같은 id 유지**: 활성 기체가 지배 기체로 남아 `persistentId`가 그대로이면 `vessel_changed` 경로가 돌지 않는다. 그래서 `DockMode.Update` 맨 앞에서 `TryAdoptDocked`를 부른다.
  - **포트 상태**: `ModuleDockingNode.state` 문자열은 판정에 쓰지 않는다. 결합 직후 값만 로그로 남긴다.
  - **reload 인계**: `DockMode.Save` 키는 `own_port`, `target_port`, `attempts`, `phase`, `armed`, `control_before`, `coupled`(`Coordinator.DockCoupled`)다.
    - `Load`는 인스턴스 필드만 복원하고(`_coupledSaved`, `_watchRestored = false`) 정적 상태는 건드리지 않는다. Resume 1단계의 "정적 상태를 바꾸지 않는다" 계약 때문이다.
    - `Resume`이 인계를 확정한 2단계(`Adopt` 뒤, 정적 상태를 바꾸는 구간)에서 `(_mode as DockMode)?.RestoreWatch()`를 불러 `WatchDock(own_port, coupled)`를 복원한다. 그래야 다음 Tick의 기체 id 검사(Coordinator.cs 44~49줄, `mode.Update`보다 앞) 때 이미 `DockCoupled`가 맞다(검토 R3-1-B-4). `Update`의 복원은 안전망으로 남긴다.
    - 인계가 끝난 뒤 첫 Tick의 `Update`가 `WatchDock(own_port, coupled)`를 부른다. 인계가 실패하면 `Incompatible`의 `EndMode`가 `End`를 부른다(검토 R1-2-B-5, R2-2-B-8). 결합 후 reload에서는 `vessel_id`가 교체된 값이라 `Resume`이 맞는다. `Plan.Restore`도 활성 기체로 다시 `Parse`하므로 같은 값이 된다.
  - **정리**: 중단·실패 때 `Release()` 맨 앞의 `_mode.End(v)`가 감시를 풀고 제어 기준 부품을 되돌린다(단위 1).
- **검증**

  - `dock_failed`: 분리할 수 없으므로 결합 전 상태(새 두 기체 또는 결합 전 reload 저장)에서 `debug_dock_reject`(`count: 3`) 뒤 `dock`(`retries: 2`)을 낸다.

    - `plan_aborted {reason: dock_failed}`로 끝난다.

  - RCS 그룹을 끈 상태에서 첫 step `[{mode: dock, rcs: true}]`를 제출하면 받아들여진다(검토 R2-2-B-2).
  - 계획 교체: `dock` 진행 중 `submit_flight_plan`으로 교체하면 제어 기준 부품이 원래대로 돌아온다(`End`, 검토 R2-2-B-4).
