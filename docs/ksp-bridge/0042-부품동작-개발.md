**코드 초안**

```csharp
/// <summary>get_vessel_parts and invoke_part_event (design 0041): what the flight part action window shows and does. Main thread only.</summary>
public static class PartStates
{
    public static object List(Vessel v, string partIdText, string module)
    {
        IEnumerable<Part> parts = v.parts;
        if (partIdText != null)
        {
            uint id = ParsePartId(partIdText);   // 400, same text as DebugHandler.PartId
            parts = new[] { v.parts.FirstOrDefault(p => p.persistentId == id) ?? throw PartNotFound(id) };
        }
        List<object> rows = new List<object>();
        foreach (Part p in parts)
        {
            List<object> modules = new List<object>();
            if (module == null || module == "Part")
            {
                AddIfShown(modules, DescribePartLevel(p));
            }
            for (int i = 0; i < p.Modules.Count; i++)
            {
                PartModule m = p.Modules[i];
                if (m.isEnabled && (module == null || m.moduleName == module))
                {
                    AddIfShown(modules, DescribeModule(p, m, i));
                }
            }
            if (module != null && modules.Count == 0)
            {
                continue;
            }
            rows.Add(new Dictionary<string, object>
            {
                ["id"] = (double)p.persistentId, ["name"] = p.partInfo.name, ["title"] = p.partInfo.title,
                ["modules"] = modules, ["resources"] = Resources(p),
            });
        }
        return new Dictionary<string, object> { ["vessel"] = v.vesselName, ["id"] = (double)v.persistentId, ["parts"] = rows };
    }

    /// <summary>CanActivateEvent steps 1·2: the event is in the window's list at all.</summary>
    internal static bool Listed(BaseEvent e) => !(e.advancedTweakable && !GameSettings.ADVANCED_TWEAKABLES)
        && e.guiActive && e.active && !e.EventIsDisabledByVariant;

    /// <summary>CanActivateEvent steps 3·4 for a listed event: null = the button can be pressed now.</summary>
    internal static string Unavailable(BaseEvent e, Part p)
    {
        if (p.vessel == null || p.vessel != FlightGlobals.ActiveVessel) return "not_active_vessel";
        bool ships = InputLockManager.IsUnlocked(ControlTypes.ACTIONS_SHIP);
        bool full = !e.requireFullControl || InputLockManager.IsUnlocked(ControlTypes.TWEAKABLES_FULLONLY);
        if ((ships && p.vessel.IsControllable && full) || e.guiActiveUncommand) return null;
        return !ships || !full ? "input_locked" : "no_control";
    }

    private static List<object> Fields(BaseFieldList list, object host)
    {
        List<object> rows = new List<object>();
        foreach (BaseField f in list)
        {
            if (!f.guiActive || (f.advancedTweakable && !GameSettings.ADVANCED_TWEAKABLES)) continue;
            Dictionary<string, object> d = new Dictionary<string, object> { ["field"] = f.name, ["title"] = f.guiName, ["units"] = f.guiUnits };
            try { d["value"] = FieldValue(f.GetValue(host)); }
            catch (Exception ex) { d["value"] = null; d["error"] = ex.GetType().Name + ": " + ex.Message; }   // design 0041: one bad getter must not fail the list
            rows.Add(d);
        }
        return rows;
    }

    /// <summary>Main states that are not window fields (design 0041 decision 2).</summary>
    private static Dictionary<string, object> State(PartModule m)
    {
        switch (m)
        {
            case ModuleDeployablePart d: return new Dictionary<string, object> { ["deploy_state"] = d.deployState.ToString() };
            case ModuleDockingNode n: return new Dictionary<string, object>
            {
                ["state"] = n.state,
                ["docked_part_id"] = n.otherNode != null && n.otherNode.part != null ? (object)(double)n.otherNode.part.persistentId : null,
            };
            default: return null;
        }
    }

    private static object FieldValue(object value) => value == null || value is bool || value is string || value is double
        || value is float || value is int ? value : value.ToString();   // enum → name, others → ToString (Json handles float·int)
}
```

- `DescribeModule(p, m, index)`: `{index, module, fields: Fields(m.Fields, m), events, state: State(m)}`. events는 `m.Events`에서 `Listed`인 것 `{event: name, title: guiName, available: Unavailable(e, p) == null, reason: Unavailable(e, p)}`. `state`가 null이면 키를 넣지 않는다.
- `DescribePartLevel(p)`: `Fields(p.Fields, p)`·`p.Events`로 같은 형식, `module: "Part"`, `index: null`.
- `AddIfShown`: fields·events가 모두 비고 `state`도 없으면 넣지 않는다.
- `Resources(p)`: `{resource, amount, max, flow_enabled}`.
- `ParsePartId(string)`: `double.TryParse(InvariantCulture)` 뒤 `DebugHandler.PartId`와 같은 검사·문구(`part_id must be an integer part id (persistentId) 0..4294967295`).

**대상**

- `FlightHandler.Routes`: 라우트 추가

**코드 초안**

```csharp
public static object Invoke(Vessel v, RequestBody b)
{
    uint id = PartIdFromBody(b);                       // DebugHandler.PartId rule and text
    string module = b.GetString("module"), eventName = b.GetString("event");
    int? moduleIndex = b.Has("module_index") ? ModuleIndex(b) : (int?)null;   // integer >= 0, else 400; null = absent
    Part p = v.parts.FirstOrDefault(x => x.persistentId == id) ?? throw PartNotFound(id);
    BaseEventList events = FindTarget(p, module, moduleIndex, out PartModule target, out int? index);
    BaseEvent e = events[eventName] ?? throw new ApiException(404, "event_not_found",
        eventName + " is not an event of " + module + " (events: " + string.Join(", ", events.Select(x => x.name).ToArray()) + ")");
    string reason = !Listed(e) ? "inactive" : Unavailable(e, p);
    if (reason != null)
    {
        throw new ApiException(409, "event_unavailable", module + "." + eventName + " cannot be pressed in the part action window now: " + reason);
    }
    uint before = v.persistentId;
    List<object> counterparts = new List<object>();
    try
    {
        if (target != null)   // module event: UIPartActionButton.OnClick sends to the counterparts' part-level lists first
        {
            foreach (Part s in p.symmetryCounterparts)
            {
                counterparts.Add(new Dictionary<string, object> { ["part_id"] = (double)s.persistentId, ["sent"] = s.Events.Send(e.id) });
            }
        }
        e.Invoke();           // part-level event: Invoke only
    }
    catch (Exception ex)
    {
        Record(p, module, index, eventName, counterparts, false, false, ex.GetType().Name + ": " + ex.Message);
        throw new ApiException(500, "event_failed", ex.GetType().Name + ": " + ex.Message);
    }
    bool changed = FlightGlobals.ActiveVessel == null || FlightGlobals.ActiveVessel.persistentId != before;
    bool separated = p == null || p.vessel != v;
    Record(p, module, index, eventName, counterparts, changed, separated, null);
    Dictionary<string, object> r = new Dictionary<string, object>
    {
        ["part_id"] = (double)id, ["part"] = p.partInfo.name, ["module"] = module, ["index"] = index,
        ["event"] = eventName, ["title"] = e.guiName, ["counterparts"] = counterparts,
        ["vessel_changed"] = changed, ["separated"] = separated,
        ["part_vessel_id"] = p != null && p.vessel != null ? (object)(double)p.vessel.persistentId : null,
        ["state_error"] = null,
    };
    try
    {
        r["module_state"] = target == null ? DescribePartLevel(p) : DescribeModule(p, target, index.Value);
    }
    catch (Exception ex)   // the event already ran: answer 200 so a retry does not press a toggle twice
    {
        r["module_state"] = null;
        r["state_error"] = ex.GetType().Name + ": " + ex.Message;
    }
    return r;
}
```

- `FindTarget(p, module, moduleIndex, out target, out index)`
  - `module == "Part"`: `module_index`가 있으면 400. `target = null`, `index = null`, `p.Events`.
  - `moduleIndex`가 있으면 `p.Modules[i]`가 범위 안이고 `moduleName == module`이어야 한다. 아니면 404 `module_not_found`.
  - 없으면 이름이 같은 모듈 목록. 0개 → 404(메시지에 `p.Modules`의 이름 목록). 그중 `isEnabled`가 2개 이상 → 409 `ambiguous_module`(메시지에 활성 모듈 index 목록). 1개면 그것, 0개면 첫 모듈.
  - 고른 모듈이 `!isEnabled` → 409 `event_unavailable`(`module disabled`).
- `Record(p, module, index, ev, counterparts, changed, separated, error)`: `FlightEvents.Add("part_event", Coordinator.Running ? Coordinator.StepIndex : -1, {part_id, part, module, index, event, counterparts, vessel_changed, separated, error})`와 `UnityEngine.Debug.Log("[KspBridge] part_event <part> (<id>) <module>.<event>" + (error != null ? " failed: " + error : separated ? " separated" : ""))`.
- Undock 뒤 부품이 활성 기체 밖에 있으면 `module_state`의 event는 `Unavailable`이 `not_active_vessel`을 돌려준다(설계 결정: 활성 기체 부품만 다룸).

```csharp
yield return new Route("POST", "/api/flight/part_event", (m, b, q) => PartStates.Invoke(Guard.ActiveVessel(), b));
```

- Logic 빌드 성공

```csharp
/// <summary>get_commnet (design 0041 decisions 7·8). Main thread only.</summary>
public static class CommNetState
{
    private static readonly string[] Connection = { "connected", "connected_home", "control_state", "signal_strength", "signal", "signal_delay", "path" };

    public static object Describe(Vessel v)
    {
        CommNetVessel c = v.connection as CommNetVessel;
        CommNode node = c?.Comm;
        Dictionary<string, object> d = new Dictionary<string, object>
        {
            ["enabled"] = HighLogic.CurrentGame.Parameters.Difficulty.EnableCommNet,
            ["vessel"] = v.vesselName, ["id"] = (double)v.persistentId,
            ["antenna"] = node == null ? null : new Dictionary<string, object>
            {
                ["transmit_power"] = node.antennaTransmit.power, ["relay_power"] = node.antennaRelay.power,
            },
        };
        if (!(bool)d["enabled"] || c == null)
        {
            foreach (string k in Connection) d[k] = null;
            return d;
        }
        d["connected"] = c.IsConnected;
        d["connected_home"] = c.IsConnectedHome;
        d["control_state"] = c.ControlState.ToString();
        d["signal_strength"] = c.SignalStrength;
        d["signal"] = c.Signal.ToString();
        d["signal_delay"] = c.SignalDelay;
        d["path"] = c.ControlPath.Select(link => (object)Hop(link)).ToList();   // path links: a = start side, b = next node
        return d;
    }

    /// <summary>Path links carry only a, b, cost and the used strength (CommNetwork.FindPath). The antenna roles come from
    /// the network link from[to] (a node is a Dictionary of neighbour → link; bestLink is a pathing scratch value reset by
    /// every vessel's path search, so it is not read here) whose stored strength equals the used one.</summary>
    private static Dictionary<string, object> Hop(CommLink link)
    {
        CommNode from = link.a, to = link.b;
        double? range = null;
        string aRole = null, bRole = null;   // antenna used by o.a / o.b
        if (from.TryGetValue(to, out CommLink o) && o != null)
        {
            double s = link.signalStrength;
            if (o.strengthRR == s) { range = Range(o.a.antennaRelay, o.b.antennaRelay); aRole = bRole = "relay"; }
            else if (o.strengthAR == s) { range = Range(o.a.antennaRelay, o.b.antennaTransmit); aRole = "relay"; bRole = "transmit"; }
            else if (o.strengthBR == s) { range = Range(o.b.antennaRelay, o.a.antennaTransmit); aRole = "transmit"; bRole = "relay"; }
        }
        bool fromIsA = o != null && o.a == from;
        return new Dictionary<string, object>
        {
            ["from"] = Node(from), ["to"] = Node(to),
            ["distance"] = (from.precisePosition - to.precisePosition).magnitude,
            ["max_range"] = range, ["signal_strength"] = link.signalStrength,
            ["from_antenna"] = fromIsA ? aRole : bRole, ["to_antenna"] = fromIsA ? bRole : aRole,
        };
    }

    private static double Range(CommNode.AntennaInfo a, CommNode.AntennaInfo b) => CommNetScenario.RangeModel.GetMaximumRange(a.power, b.power);

    private static Dictionary<string, object> Node(CommNode n)
    {
        Vessel owner = FlightGlobals.Vessels.FirstOrDefault(x => x.connection != null && x.connection.Comm == n);
        return new Dictionary<string, object>
        {
            ["name"] = n.displayName, ["vessel_id"] = owner != null ? (object)(double)owner.persistentId : null, ["home"] = n.isHome,
        };
    }
}
```

- 디컴파일로 확인한 것: `CommNode`는 `Node<...> : Dictionary<CommNode, CommLink>`이다. 경로 링크는 `Set(commNode, b, ...)`로 `base.a`·`base.b`를 채우고 `path.Insert(0, ...)`로 쌓는다.
- 구현 시 확인할 것(결과로 순회·멤버 접근 방식만 바꾸며 응답 형식은 바꾸지 않는다): 기반 `Path<...>`가 `CommLink`를 열거할 수 있는지(아니면 그 목록 필드), 기반 `Link<...>`의 `a`·`b` 멤버 접근성.

```csharp
yield return new Route("GET", "/api/flight/commnet", (m, b, q) => CommNetState.Describe(Guard.ActiveVessel()));
```

**검증**
- Logic 빌드 성공
- `reload_mod` 뒤 비행 중 `curl /api/flight/commnet`: KSC 근처나 궤도 기체에서 `connected=true`, `path` 첫 구간 `from.vessel_id`가 활성 기체 id, 직결이면 그 구간 `to.home=true`. `antenna.transmit_power`가 0보다 크다. 각 구간 `max_range`·`from_antenna`·`to_antenna`가 null이 아니고 `max_range`가 `distance`보다 크다. 가능하면 다른 CommNet 기체가 로드된 상태(예: Undock 직후)에서도 같은 결과인지 본다.
