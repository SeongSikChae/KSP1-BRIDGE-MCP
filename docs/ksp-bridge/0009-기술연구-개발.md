```csharp
public static object List(NameValueCollection q)
{
    Career.Require();
    bool all = q["all"] == "true";
    foreach (ProtoRDNode n in AssetBase.RnDTechTree.GetTreeNodes())
    {
        string id = n.tech.techID;
        string state = Researched(id) ? "researched" : ParentsMet(n) ? "researchable" : "locked";
        if (!all && state == "locked") continue;
        // id, title=GetTechnologyTitle(id), cost=n.tech.scienceCost, state, parents, any_parent,
        // all ? part_count : parts[{name,title}] (PartLoader.LoadedPartsList.Where(TechRequired==id))
    }
}

public static object Research(string id)
{
    Career.Require();
    ProtoRDNode n = Find(id) ?? throw NotFound;
    if (Researched(id)) throw 409 already_researched;
    if (!ParentsMet(n)) throw 409 prerequisites_missing (부족 노드 나열);
    float limit = GameVariables.Instance.GetScienceCostLimit(ScenarioUpgradeableFacilities.GetFacilityLevel(SpaceCenterFacility.ResearchAndDevelopment));
    if (cost > limit) throw 409 cost_limit;
    if (R&D.Instance.Science < cost) throw 409 not_enough_science;
    R&D.Instance.AddScience(-cost, TransactionReasons.RnDTechResearch);
    R&D.Instance.UnlockProtoTechNode(n.tech);
    ResearchAndDevelopment.RefreshTechTreeUI();
    ContractDefs.Instance?.TechResearched(default);
}

static bool Researched(string id) => ResearchAndDevelopment.GetTechnologyState(id) == RDTech.State.Available;
static bool ParentsMet(ProtoRDNode n) => n.parents.Count == 0
    || (n.AnyParentToUnlock ? n.parents.Any(p => Researched(p.tech.techID)) : n.parents.All(p => Researched(p.tech.techID)));
```

## 구현 중 변경
- `ContractDefs`는 `FinePrint` 네임스페이스, `TechResearched`가 internal → 리플렉션 호출(인자 미사용 확인)
- 과학 판정은 `CurrencyModifierQuery.RunQuery(RnDTechResearch, 0, -cost, 0).CanAfford()` (독립 리뷰 반영, 전략 보정 포함)

## 알려진 차이 (독립 리뷰, 수정 안 함)
- `GameEvents.OnTechnologyResearched` 미발생: RDTech 인스턴스가 필요하고 스톡 치트도 쏘지 않음. 이 이벤트를 쓰는 모드를 설치하면 재검토
- 트리 노드는 정적 캐시(`GetTreeNodes`) 사용: 다른 기술 트리 세이브로 바꾸면 R&D 화면을 한 번 열어야 갱신될 수 있음
- hideEmpty 노드 미필터, Bypass 꺼진 세이브에서 parts_unlocked 빈 배열(구매 필요 미표시)
