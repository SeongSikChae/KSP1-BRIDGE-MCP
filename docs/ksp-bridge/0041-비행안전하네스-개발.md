### 코드 초안
```csharp
public static class AscentGuidance
{
    public static string State { get; private set; } = "idle";
    // idle | launching | vertical | kick | gravity_turn | coasting | correcting | done | aborted
    public static string Reason { get; private set; }
    public static bool Running => State != "idle" && State != "done" && State != "aborted";

    private static Vessel _v;
    private static double _target, _heading, _turnSpeed, _turnAngle, _maxAoa, _apTol;
    private static bool _autoStage;
    private static float _throttle;
    private static int _partsDied0;
    private static double _lastAp, _lastUt, _apRatePerThrottle; // throttle-down

    public static Dictionary<string, object> Start(Vessel v, AscentArgs a)
    {
        if (NodeExecutor.Running) throw ApiException.BadRequest("execute_node is running (cancel_node first)");
        if (v.packed || TimeWarp.CurrentRate > 1.01f) throw ApiException.BadRequest("stop time warp first");
        bool pad = v.situation == Vessel.Situations.PRELAUNCH;
        if (!pad && v.situation != Vessel.Situations.LANDED && v.situation != Vessel.Situations.FLYING)
            throw ApiException.BadRequest("ascent starts on the ground or in atmospheric flight (situation " + v.situation + ")");
        double atmo = v.mainBody.atmosphere ? v.mainBody.atmosphereDepth : 0;
        if (a.TargetApoapsis <= atmo) throw ApiException.BadRequest("target_apoapsis must be above the atmosphere (" + atmo + " m)");
        // ... store args, Cancel("restarted"), _v = v, _partsDied0 = AnomalyMonitor.PartsDied(v.id)
        SetThrottle(pad ? 0f : 1f);
        Hold(90);                                       // surface pitch 90 / heading
        SetState(pad ? "launching" : "vertical");
        // ... return plan: state, target_apoapsis, atmosphere_height, stages (StageDv per stage), warnings
    }

    private static void Hold(double pitch) =>
        AttitudeController.Set(_v, "own", "surface", pitch, _heading, null, 0, 0, null);
}
```

```csharp
public static void FixedUpdate()
{
    if (!Running) return;
    if (HighLogic.LoadedScene != GameScenes.FLIGHT || FlightGlobals.ActiveVessel != _v || _v.state == Vessel.State.DEAD) { Abort("vessel_changed"); return; }
    if (!AttitudeController.Active(_v) || !OwnsThrottle()) { Abort("overridden"); return; }
    if (AnomalyMonitor.ControlLost(_v)) { Abort("control_lost"); return; }
    if (AnomalyMonitor.PartsDied(_v.id) > _partsDied0) { Abort("part_lost"); return; }

    double ap = _v.orbit.ApA;
    switch (State)
    {
        case "launching":
            if (!TryStage("launch")) return;            // unit 2 checks warnings; unit 1: StageManager.ActivateNextStage()
            SetThrottle(1f); SetState("vertical"); break;
        case "vertical":
            if (_v.srfSpeed >= _turnSpeed) { Hold(90 - _turnAngle); SetState("kick"); }
            break;
        case "kick":
            if (SurfacePitch(_v) <= 90 - _turnAngle + 1)
            {
                AttitudeController.Set(_v, "own", "surface_prograde", null, null, null, 0, 0, _maxAoa);
                SetState("gravity_turn");
            }
            break;
        case "gravity_turn":
            if (ap >= _target) { SetThrottle(0f); SetState("coasting"); break; }   // unit 3 adds prograde hold
            SetThrottle(ThrottleFor(ap));
            break;
    }
}

/// <summary>Pitch of the surface velocity above the horizon (deg).</summary>
private static double SurfacePitch(Vessel v)
{
    Vector3d up = (v.CoMD - v.mainBody.position).normalized;
    return 90 - Vector3d.Angle(v.srf_velocity, up);
}

/// <summary>Full throttle until the apoapsis gets close, then scale by the measured apoapsis rise per unit throttle.</summary>
private static float ThrottleFor(double ap)
{
    double ut = Planetarium.GetUniversalTime();
    double dt = ut - _lastUt;
    if (_lastUt <= 0) { _lastAp = ap; _lastUt = ut; return 1f; }   // first sample (start, after staging)
    if (dt > 0.2 && _throttle > 0.05f && Prediction.Get(_v).ThrustKn > 0)      // measure only while thrusting
    {
        double rate = (ap - _lastAp) / dt / _throttle;
        _apRatePerThrottle = _apRatePerThrottle <= 0 ? rate : 0.8 * _apRatePerThrottle + 0.2 * rate;
        _lastAp = ap; _lastUt = ut;
    }
    if (_apRatePerThrottle <= 0) return 1f;
    double t = (_target - ap) / (_apRatePerThrottle * 5.0);  // reach the target in ~5 s at this throttle
    return (float)Math.Max(0.05, Math.Min(1.0, t));
}
```
- 중단 시 자세(검토 결정): 스로틀은 0으로 하고 잠금을 풉니다. 자세는 `overridden`·`control_lost`·`vessel_changed`면 풀고, 그 밖(`stage_warning`, `out_of_fuel`, `flameout`, `part_lost`, `cancelled`)은 `surface_prograde`(`max_aoa_deg`) 유지로 바꿔 둡니다. 결과·기록에 `attitude_kept`를 남깁니다.
  ```csharp
  private static void Abort(string reason, string detail = null)
  {
      Reason = reason;
      bool keep = reason != "overridden" && reason != "control_lost" && reason != "vessel_changed";
      ReleaseThrottle();
      if (keep) SetFrame("surface_prograde", _maxAoa); else ReleaseAttitude();
      SetState("aborted", detail, keep);
  }
  ```
- 소유권·종료는 `NodeExecutor`의 `SetThrottle`/`OwnsThrottle`/`StopEngines`와 같은 형태로 둡니다. 자세 소유 판정은 frame이 `surface`/`surface_prograde`/`prograde` 중 하나이고 `AttitudeController.Active`일 때로 합니다. 상태 전이 때 이 도구가 프레임을 바꾸므로 `Frame` 비교 대신 마지막으로 설정한 프레임을 `_frame`에 두고 비교합니다.
- `SetState`는 `FlightRecorder.AscentEvent(_v, {state, reason, altitude, apoapsis, srf_speed, stage})`를 남깁니다.

## 단위 2. 자동 분리

### 대상
`AscentGuidance.TryStage(string why)`, `FixedUpdate`의 분리 판정. `CraftAnalysis.Drops`, `CraftAnalysis.Control`, `CraftAnalysis.StageWarnings` 재사용(`StageHandler`와 같은 호출).

### 코드 초안
```csharp
private static double _lastStageUt = -1;

/// <summary>Booster drop or empty stage: true when the next stage should fire now.</summary>
private static bool NeedStage(out string why)
{
    why = null;
    List<ModuleEngines> lit = Engines(_v).Where(e => e.EngineIgnited).ToList();
    if (lit.All(e => e.flameout || !e.isOperational)) { why = "no_thrust"; return true; }
    int next = _v.currentStage - 1;
    if (next < 0) return false;
    HashSet<Part> drops = new HashSet<Part>(CraftAnalysis.Drops(_v, next, new HashSet<Part>(_v.parts)));
    List<ModuleEngines> dropped = lit.Where(e => drops.Contains(e.part)).ToList();
    if (dropped.Count > 0 && dropped.All(e => e.flameout)) { why = "boosters_empty"; return true; }
    return false;
}

private static bool TryStage(string why)
{
    int next = _v.currentStage - 1;
    if (next < 0) { Abort("out_of_fuel"); return false; }
    HashSet<Part> attached = new HashSet<Part>(_v.parts);
    List<Part> dropped = CraftAnalysis.Drops(_v, next, attached);
    attached.ExceptWith(dropped);
    List<Part> burning = attached.Where(p => p.inverseStage >= next && p.FindModuleImplementing<ModuleEngines>() != null).ToList();
    List<string> warnings = new List<string>();
    CraftAnalysis.StageWarnings(next, dropped, burning.Count > 0, (bool)CraftAnalysis.Control(attached, burning)["steerable"], warnings);
    if (warnings.Count > 0) { Abort("stage_warning", string.Join("; ", warnings)); return false; }
    // Never fire parachutes, and only stage when an engine with fuel will be left to burn (now or after a decoupler-only stage).
    if (attached.Any(p => p.inverseStage == next && p.FindModuleImplementing<ModuleParachute>() != null)
        || !attached.Any(p => p.inverseStage <= next && HasFuelledEngine(p)))
    { Abort("out_of_fuel"); return false; }
    StageManager.ActivateNextStage();
    _lastStageUt = Planetarium.GetUniversalTime();
    _apRatePerThrottle = 0; _lastUt = 0;                // re-measure the apoapsis rate after staging
    FlightRecorder.AscentEvent(_v, new Dictionary<string, object> { ["state"] = "staged", ["stage"] = next, ["why"] = why });
    return true;
}
```
- `FixedUpdate`의 vertical·kick·gravity_turn·correcting에서 `ut − _lastStageUt > 1 && NeedStage(out why)`이면 `_autoStage ? TryStage(why) : Abort("flameout")`.
- 분리 뒤 1 s가 지나도 추력이 없고 `currentStage == 0`이면 `out_of_fuel`(TryStage의 `next < 0`).
- `HasFuelledEngine(p)`: 엔진 모듈이 있고, 추진제(`propellants`)를 공급받을 수 있는 자원이 0보다 큼(`p.crossfeedPartSet`의 해당 자원 합, 또는 간단히 `GetConnectedResourceTotals`). 분리기만 있는 단은 그다음 단에 이런 엔진이 있으면 통과합니다(`inverseStage <= next`).
- 낙하산이 든 단은 상승 유도가 절대 분리하지 않습니다(`out_of_fuel`로 멈추고 사용자가 판단).
- 분리로 떨어진 부품은 `PartsDied`에 들어가지 않습니다(분리는 파괴가 아님). 떨어진 단이 나중에 파괴되면 다른 Vessel이라 `PartsDied(_v.id)`에 들어가지 않는지 단위 검증에서 확인합니다.
- `Abort(reason, detail)` 오버로드: `Reason = reason`, 기록에 `detail`.

### 검증
- 발사대 무인 기체(액체 2단 + 방사형 SRB 2개): `wait_until ascent_state` 로 SRB 고갈 → `staged boosters_empty`, 1단 고갈 → `staged no_thrust`를 비행 기록에서 확인.
- 경고 단: 열차폐를 떨어뜨리는 단이 있는 기체에서 그 단 차례에 `aborted stage_warning`, 분리 안 됨, 스로틀 0, `surface_prograde` 유지.
- 낙하산 단: 엔진 단 연료가 끝나고 다음 단이 낙하산이면 `aborted out_of_fuel`, 낙하산 펴지지 않음(발사대에서 연료를 줄인 기체 `resources` 0.05로 확인).

## 단위 3. 활공·원지점 보정·done

### 코드 초안
```csharp
case "gravity_turn":
    if (ap >= _target) { SetThrottle(0f); Prograde(); if (InAtmo()) SetState("coasting"); else Finish(); break; }
    SetThrottle(ThrottleFor(ap));
    break;
case "coasting":
    if (!InAtmo()) { Finish(); break; }
    if (ap < _target - _apTol) { SetState("correcting"); }
    break;
case "correcting":
    if (ap >= _target) { SetThrottle(0f); SetState("coasting"); break; }
    if (AttitudeController.ErrorDeg(_v) < 5) SetThrottle(ThrottleFor(ap)); // burn only along prograde
    break;

private static bool InAtmo() => _v.mainBody.atmosphere && _v.altitude < _v.mainBody.atmosphereDepth;
private static void Prograde() => SetFrame("prograde", _maxAoa);   // keep the AoA limit in the upper atmosphere
private static void Finish()
{
    // Throttle 0 and release the throttle lock; keep the prograde hold for warp_to ap / circularization.
    if (OwnsThrottle()) { ThrottleLock.Set(_v, 0f); FlightInputHandler.state.mainThrottle = 0f; ThrottleLock.Clear(); }
    SetState("done");
}
```
- 대기 없는 천체이거나 이미 대기권 밖이면 `gravity_turn`에서 원지점 도달 시 바로 `Finish()`.
- `correcting`의 `ThrottleFor`는 측정된 비율을 그대로 써서 짧게 끝납니다.

### 검증
- 단위 2와 같은 비행에서: 원지점 80 km 도달 → `coasting`, 항력으로 원지점 감소 시 `correcting` → `coasting`, 70 km 통과 → `done`. done 뒤 `get_vessel_state`에서 스로틀 0, 자세 prograde 유지, 원지점 ≥ 80 km − 200 m.
- 이어서 `add_maneuver_node`(원지점) + `execute_node`로 순환화해 근지점 > 70 km(묶음 B 도구와 연결 확인).
- 무인 기체로 시험. 회수는 궤도 이탈 연소 후 낙하산 트리거 먼저 걸고 진행.

### 대상

### 코드 초안
```csharp
List<object> wps = Waypoints(c);

private static List<object> Waypoints(Contract c)
{
    List<object> list = new List<object>();
    FinePrint.WaypointManager m = FinePrint.WaypointManager.Instance();
    if (m == null) return list;
    Vessel v = HighLogic.LoadedSceneIsFlight ? FlightGlobals.ActiveVessel : null;
    foreach (FinePrint.Waypoint w in m.Waypoints.Where(w => w.contractReference == c))
    {
        Dictionary<string, object> e = new Dictionary<string, object>
        {
            ["name"] = w.name, ["body"] = w.celestialName,
            ["latitude"] = w.latitude, ["longitude"] = w.longitude, ["altitude"] = w.altitude,
            ["on_surface"] = w.isOnSurface,
        };
        if (v != null && v.mainBody.bodyName == w.celestialName)
        {
            e["distance"] = GreatCircle(v.mainBody.Radius, v.latitude, v.longitude, w.latitude, w.longitude, out double bearing);
            e["bearing"] = bearing;
        }
        list.Add(e);
    }
    return list;
}
```

### 코드 초안
```csharp
public sealed class CraftReader
{
    private sealed class RawPart
    {
        public string Name, Id;
        public Vector3 Pos; public Quaternion Rot;
        public RawPart Parent; public string OwnNode, ParentNode; public bool Surface;
        public readonly List<RawPart> Children = new List<RawPart>();
        public readonly List<RawPart> Sym = new List<RawPart>();
        public int Istg; public ConfigNode Node;
    }

    public static Dictionary<string, object> Read(string path, List<string> warnings)
    {
        ConfigNode root = ConfigNode.Load(path) ?? throw ApiException.NotFound("craft file not found");
        List<RawPart> parts = root.GetNodes("PART").Select(Parse).ToList();
        Dictionary<string, RawPart> byId = parts.ToDictionary(p => p.Id);
        Link(parts, byId);                 // attN / srfN / sym
        PartSpec design = ToSpec(parts[0], warnings, "root");
        RoundTrip(design, parts, warnings);
        return new Dictionary<string, object> { ["design"] = ToJson(design), ["part_count"] = parts.Count };
    }

    private static RawPart Parse(ConfigNode n)
    {
        string full = n.GetValue("part");                 // e.g. "fuelTank_4294955554"
        int i = full.LastIndexOf('_');
        return new RawPart
        {
            Name = i > 0 ? full.Substring(0, i) : full, Id = i > 0 ? full.Substring(i + 1) : full,
            Pos = ConfigNode.ParseVector3(n.GetValue("pos")), Rot = ConfigNode.ParseQuaternion(n.GetValue("rot")),
            Istg = int.Parse(n.GetValue("istg") ?? "-1", CultureInfo.InvariantCulture), Node = n,
        };
    }
}
```

- `ToJson(PartSpec)`: `build_craft` 입력 키(`part`, `node`, `symmetry`, `angle`, `height`, `stage`, `thrust_limit`, `resources`, `children`, `radial`)로 변환. 기본값(symmetry 1, angle 0 등)은 생략하고 각도·높이는 소수 둘째 자리로 반올림합니다.

### 검증
- 묶음 A·B에서 `build_craft`로 만든 기체(HarnessTestB2, HarnessTestC): `read_craft` → `build_craft dryRun` 결과의 부품 수·질량·단별 Δv가 원본 `build_craft` 결과와 같고 warnings 없음.
- 게임 VAB에서 직접 만든 기체(스톡 예제 하나, 예: Ships/VAB 기본 기체): 트리 반환, 오프셋 편집 부품이 있으면 warnings에 위치 차이.
- 없는 이름 → 404, facility 잘못 → 400.

[McpServerTool(Name = "ascent_guidance", Destructive = true)]
[Description("Fly a gravity-turn ascent in-game: ignite on the pad (stages once), climb vertically, kick turnAngle deg at turnStartSpeed, " +
    "then follow surface prograde within maxAoaDeg; throttle down near the target and cut at targetApoapsis; coast to the atmosphere edge " +
    "re-burning if drag lowers the apoapsis by more than apTolerance. Auto staging: drops a stage when no engine thrusts or its boosters are empty; " +
    "a stage with get_stages warnings (protected part dropped, no steering) is not fired → aborted stage_warning. Returns the plan at once; " +
    "follow with wait_until field ascent_state (launching, vertical, kick, gravity_turn, coasting, correcting, done, aborted). " +
    "Do preflight (get_stages delta-v) before calling. After done the prograde hold stays on; circularize with add_maneuver_node + execute_node.")]
public Task<string> AscentGuidance(
    [Description("Target apoapsis (m above sea level), above the atmosphere")] double targetApoapsis,
    [Description("Heading (deg), default 90")] double heading = 90,
    [Description("Surface speed to start the turn (m/s), default 60")] double turnStartSpeed = 60,
    [Description("Kick angle from vertical (deg), default 10")] double turnAngle = 10,
    [Description("Angle-of-attack limit during the turn (deg), default 5")] double maxAoaDeg = 5,
    [Description("Auto staging, default true")] bool autoStage = true,
    [Description("Re-burn when the apoapsis falls this much below the target while coasting (m), default 200")] double apTolerance = 200,
    CancellationToken ct = default)
    => client.SendAsync(HttpMethod.Post, "api/ascent/start", new
    {
        target_apoapsis = targetApoapsis, heading, turn_start_speed = turnStartSpeed, turn_angle = turnAngle,
        max_aoa_deg = maxAoaDeg, auto_stage = autoStage, ap_tolerance = apTolerance,
    }, ct);

[McpServerTool(Name = "read_craft", ReadOnly = true)]
