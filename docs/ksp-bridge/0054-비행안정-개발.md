### 코드 초안

```csharp
// ControllerTrace.cs — internal throughout (Sample is internal; a public Record would be CS0051)
internal static class ControllerTrace
{
    internal const int Capacity = 1000;   // ~20 s at 50 Hz
    internal const int Kinds = 5;         // wheel, gimbal, surface, rcs, other
    internal static readonly string[] KindNames = { "wheel", "gimbal", "surface", "rcs", "other" };

    internal struct Sample
    {
        public double Ut, Throttle, Q;
        public Vector3 Err, Omega, Want, Input, Alpha, Moi;
        public Vector3[] Torque;   // per kind, Kinds long (new array per sample)
    }

    private static readonly Sample[] _buf = new Sample[Capacity];
    private static int _start, _count;
    private static uint _vessel;   // persistentId of the recorded vessel

    internal static void Record(uint vessel, in Sample s)
    {
        if (_count > 0 && vessel != _vessel) Clear();   // never mix vessels
        _vessel = vessel;
        _buf[(_start + _count) % Capacity] = s;
        if (_count < Capacity) _count++; else _start = (_start + 1) % Capacity;
    }

    internal static void Clear() { _start = 0; _count = 0; }
}
```

### 코드 초안

```csharp
internal static Dictionary<string, object> Describe(Vessel active, double seconds, int? step)
{
    List<Sample> w = active.persistentId == _vessel ? Window(seconds) : new List<Sample>();   // newest seconds, oldest first
    if (w.Count == 0)
    {
        return new Dictionary<string, object> { ["samples"] = 0.0 };
    }
    double tick = MedianDt(w);
    var axes = new Dictionary<string, object>();
    string[] names = { "pitch", "roll", "yaw" };
    for (int a = 0; a < 3; a++)
    {
        axes[names[a]] = Axis(w, a, tick);
    }
    var d = new Dictionary<string, object>
    {
        ["window_s"] = R(w[w.Count - 1].Ut - w[0].Ut, 2), ["samples"] = (double)w.Count, ["tick_s"] = R(tick, 4),
        ["throttle_mean"] = R(Mean(w, s => s.Throttle), 3), ["q_mean"] = R(Mean(w, s => s.Q), 3), ["axes"] = axes,
    };
    if (step is int k) d["series"] = Series(w, k);
    return d;
}

// One axis: pairs (i-1, i) with dt <= 1.5 * tick only.
private static Dictionary<string, object> Axis(List<Sample> w, int a, double tick)
{
    // err_rms_deg, err_max_deg
    // oscillation by spectrum: split the window at samples with |Want[a]| >= OmegaMax (slews at the rate limit) and at
    //   gaps (dt > 1.5 * tick); U = the longest contiguous run (runs are never joined)
    //   span = Ut_last - Ut_first of U; span < 5 s -> osc_hz, osc_amp_deg, osc_peak_ratio, err_offset_deg null
    //   err_offset_deg = mean err over U; r_i = err_i - (least-squares line over (t_i, err_i), i in U)
    //   Hann weight w_i = 0.5 - 0.5 cos(2π (t_i - t_first) / span)   (0 at both ends of U)
    //   for f = 0.20, 0.25, ..., 5.00 Hz: X(f) = Σ_U w_i r_i e^{-2πi f t_i}
    //   peak f* = argmax |X|; osc_hz = f*, osc_amp_deg = 2 |X(f*)| / Σ w_i (deg),
    //   osc_peak_ratio = |X(f*)| / median_f |X(f)|   (no oscillation threshold: calibrated against series in unit 4)
    //   cost: 97 frequencies × ≤ 1000 samples per axis, computed only on request
    // input_sat_ratio = |input| >= 0.99 share; alpha_model = mean Alpha[a] (reference only)
    // gain fit on one pair set: i with i-10..i all contiguous; for k in 1..10:
    //   y_i = (omega_i - omega_{i-1}) / dt_i, x_i = -Input_{i-k}[a] * Alpha_{i-k}[a]
    //   y = g_k x + c_k by least squares with intercept; r2_k = 1 - SSres / SStot
    //   best k by r2 -> alpha_gain = g_k, accel_bias = c_k, alpha_r2 = r2_k, lag_ticks = k - 1 (0 = no delay)
    //   lag_r2 = [r2_1 .. r2_10]; all null when < 10 pairs, std(x) < 0.01 * mean|x|,
    //   or (saturated share of the pairs >= 0.9 and Input[a] never changes sign): one-sided saturation mixes alpha change
    //   with q-driven disturbance; a biased but unsaturated input still fits (the intercept takes the steady disturbance)
    // want_slope_at_amp: e = max(|err_offset|, amp) in rad; KStop * sqrt(alpha_model / (2 * e)),
    //   0 if sqrt(2 * alpha_model * e) * KStop >= OmegaMax, null when osc_amp_deg is null
    // torque: per kind mean of Torque[k][a] (kN·m)
}
```

- 숫자는 기존 응답처럼 `double`, NaN·무한대는 null.
- `KStop`·`OmegaMax`는 `AttitudeController`의 상수를 `internal`로 열어 그대로 쓴다(값 복사 없음).

```csharp
[McpServerTool(Name = "get_controller_trace", ReadOnly = true)]
[Description("Attitude controller trace of the active vessel's flight plan: the last seconds (1..20, default 10) of the ticks it actually " +
    "steered (none without a plan or attitude target, while suspended; cleared on reload, scene change, vessel change and when a plan starts; " +
    "kept after a plan ends). Returns {window_s, samples, tick_s, throttle_mean, q_mean, axes {pitch, roll, yaw: {err_rms_deg, err_max_deg, " +
    "osc_hz, osc_amp_deg, osc_peak_ratio, err_offset_deg, input_sat_ratio, alpha_model, alpha_gain, alpha_r2, accel_bias, lag_ticks, lag_r2, " +
    "want_slope_at_amp, torque {wheel, gimbal, surface, rcs, other}}}, series?}. osc_*: strongest 0.2..5 Hz component of the detrended attitude " +
    "error over the longest run without rate-limited slews or gaps (null if under 5 s); osc_peak_ratio = peak / median spectrum; " +
    "err_offset_deg = the mean error it oscillates around. Limits: attitude error is low-frequency heavy, so a high peak ratio alone does not " +
    "prove an oscillation; a deceleration after a slew can show as a 0.2..0.5 Hz peak; a large oscillation that hits the rate limit is cut short " +
    "(null or smaller amplitude, with a high input_sat_ratio); periods over 5 s are not resolved. Check doubtful values against series. " +
    "alpha_model = available torque / MOI (rad/s²); alpha_gain = measured / modelled angular acceleration (1 = model right, <1 = torque overestimated), " +
    "fitted with an intercept accel_bias (rad/s², steady disturbance), alpha_r2 its fit; lag_ticks = extra physics ticks from input to response " +
    "(0 = none), lag_r2 the fit for 1..10 ticks (flat = lag not resolvable); want_slope_at_amp = target-rate gain (1/s) at the oscillation's " +
    "operating point. null where data is too short or flat, and the fit values also when the input stays saturated one way. step (1..50) adds series " +
    "[{t, throttle, q, pitch/roll/yaw {err_deg, omega, want, input}}] every step ticks (5 = about 10 Hz; 1 is about 1000 entries for 20 s). " +
    "{samples: 0} without data.")]
public Task<string> GetControllerTrace(
    [Description("Window in seconds, 1..20 (default 10)")] double seconds = 10,
    [Description("Series every step physics ticks, 1..50; omit for the summary only")] int? step = null,
    CancellationToken ct = default)
    => client.SendAsync(HttpMethod.Get, "api/flight/controller_trace?seconds=" + seconds.ToString(CultureInfo.InvariantCulture)
        + (step is int k ? "&step=" + k : ""), null, ct);
```

## 단위 4: 실측 (도구 검증)

### 확인 항목 (MCP 응답으로만, `get_status`로 장면 확인 후)

1. 리로드 직후 계획 시작 전 → `{samples: 0}`. 계획 뒤 다른 기체로 전환 → `{samples: 0}`.
2. StabT 궤도 `attitude` 단계 실행 중 `get_controller_trace` 요약: 세 축 값, `alpha_gain` 0.3~3, 종류별 토크(`wheel` > 0), 진동 값(`osc_hz`·`osc_amp_deg`·`osc_peak_ratio`)이 `series` 원시 오차와 맞음.
3. `step` 5 시계열 간격 약 0.1 s, 인자 범위 밖은 400, `seconds` 소수 허용.
4. 단위 1의 제어 출력 불변(diff 확인과 보조 비교) 결과를 기록한다.
5. 진동 요약 보정 자료: 진동이 있는 경우(감시기 `oscillation` 경고가 난 기체)와 없는 경우(궤도 정지 유지)에서 `osc_peak_ratio`·`osc_amp_deg`와 `step` 2 `series`를 함께 기록한다. 진동 판정 문턱과 과도·포화 제외 조건은 이 자료와 원인 판정 실측으로 정한다(0053 "진동 요약의 한계").
