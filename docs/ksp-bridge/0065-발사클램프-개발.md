`ClampHold.Detect` (판정은 `on_start` 요청이 아직 보내지지 않은 틱에만):

```csharp
public static ClampHold Detect(Vessel v)
{
    int n = v.currentStage - 1;
    int lo = int.MaxValue, hi = int.MinValue;
    foreach (Part p in v.parts)
    {
        if (p.isLaunchClamp())
        {
            lo = Math.Min(lo, p.inverseStage);
            hi = Math.Max(hi, p.inverseStage);
        }
    }
    if (lo != hi)
    {
        return null;                       // no clamp, or clamps spread over stages: as before
    }
    if (lo == n)
    {
        List<ModuleEngines> es = Liquid(v, n);
        return es.Count > 0 ? new ClampHold(split: false, clampStage: n, es) : null;   // SRBs only: as before
    }
    return lo == n - 1 && StageHasEngine(v, n) ? new ClampHold(split: true, clampStage: n - 1, null) : null;
}

/// <summary>Engines of stage n that can be shut down again: the active mode only (MultiModeEngine sets the other module's
/// isEnabled false), no SRB (throttleLocked), allowShutdown, with propellant.</summary>
private static List<ModuleEngines> Liquid(Vessel v, int n)
{
    List<ModuleEngines> list = new List<ModuleEngines>();
    foreach (ModuleEngines e in EngineModel.Engines(v))
    {
        if (e.part.inverseStage == n && e.isEnabled && e.allowShutdown && !e.throttleLocked && EngineModel.HasPropellantCached(e))
        {
            list.Add(e);
        }
    }
    return list;
}
```

`ClampHold.Tick` (`Result`는 `StageRequest.Result` 재사용, 실패는 예외):

```csharp
public StageRequest.Result Tick(Vessel v, double ut, bool send, StagingOptions o, Action<string> staged)
{
    if (!send)
    {
        return StageRequest.Result.Waiting;   // Submit's trial: detection only
    }
    switch (_phase)
    {
        case Phase.Ignite:
            if (!_split)
            {
                foreach (ModuleEngines e in _engines) e.Activate();   // OnActive returns early later: EngineIgnited
                Started(v, ut);
                break;
            }
            _request = _request ?? new StageRequest();
            StageRequest.Result r = _request.Tick(v, ut, true);
            if (r == StageRequest.Result.Failed) return r;            // AutoStager: stage_failed, as before
            if (r == StageRequest.Result.Done) { _request = null; staged("on_start"); Started(v, ut); }
            break;

        case Phase.Wait:
            double ratio = Ratio(v);
            double held = ut - _igniteUt;
            if (!EngineModel.AnyOperational(v))
            {
                Fail(v, ratio, held, "engine_out");                  // throws
            }
            if (ratio >= 0.9)
            {
                _phase = Phase.Release;
                _ratio = ratio;
                _heldS = held;
                break;
            }
            if (held > EngineModel.SpoolUpS(v) + o.ClampHoldS)
            {
                Fail(v, ratio, held, "timeout");                     // throws
            }
            break;

        case Phase.Release:
            _request = _request ?? new StageRequest();
            StageRequest.Result rr = _request.Tick(v, ut, true);
            if (rr == StageRequest.Result.Failed)
            {
                Fail(v, _ratio, _heldS, "release_failed");          // lit engines, clamps held: never carry on (R1-1-B-1)
            }
            if (rr == StageRequest.Result.Done)
            {
                Event("clamp_released", v, _ratio, _heldS, null);
                staged("on_start");
            }
            return rr;                                               // Done ends the on_start request
    }
    return StageRequest.Result.Waiting;
}

private static double Ratio(Vessel v)
{
    double throttle = v.ctrlState.mainThrottle;
    if (!(throttle > 0.01))
    {
        return 0;
    }
    double p = v.staticPressurekPa * 0.00986923;   // kPa → atm, as EngineModel.MaxAccel
    double max = EngineModel.MaxThrustSum(v, (float)p, v.atmosphericTemperature, v.atmDensity) * throttle;
    return max > 0 ? EngineModel.ThrustSum(v) / max : 0;
}

private void Fail(Vessel v, double ratio, double held, string reason)
{
    foreach (ModuleEngines e in LitEngines(v)) if (e.allowShutdown) e.Shutdown();   // same: ours; split: stage n's
    Event("clamp_hold_failed", v, ratio, held, reason);
    throw new ModeAbortException("clamp_hold_failed",
        reason + ": thrust " + (ratio * 100).ToString("F0") + "% of available after " + held.ToString("F1") + " s");
}

/// <summary>Stage activation already began (engines lit, or the split engine stage requested): a new on_start queues (R1-1-B-3).</summary>
public bool HasStarted => _phase != Phase.Ignite || (_request != null && _request.Sent);
```

`AutoStager.Update`의 `_onStart` 분기:

```csharp
if (_onStart != null)
{
    if (_hold == null && !_onStart.Sent)
    {
        _hold = ClampHold.Detect(v);   // every unsent tick: the stage cannot change before the request is sent
    }
    StageRequest.Result r = _hold != null
        ? _hold.Tick(v, ut, send, plan.StagingOptions, reason => Staged(v, reason))
        : _onStart.Tick(v, ut, send);
    switch (r)
    {
        case StageRequest.Result.Done:
            if (_hold == null) Staged(v, "on_start");   // the hold records its own staged events
            _hold = null;
            _onStart = TakeQueued();
            break;
        case StageRequest.Result.Failed:
            _hold = null;
            Event("stage_failed", v);
            _onStart = TakeQueued();
            break;
    }
    // ... existing: return while _onStart != null
}
```

- `StateName`: `_hold != null && _hold.Active` → `"clamp_hold"`(점화 뒤 해제 전). 그 밖은 기존 그대로.
- `Pending`, `BlocksWarp`: `_onStart != null`이 이미 참이므로 바뀌지 않는다(`_hold`는 `_onStart`가 있는 동안만 산다).
- `Save`: `["clamp_hold"] = _hold?.Save()`. `Load`: `_hold = ClampHold.Load(...)`(엔진은 `persistentId`로 다시 찾는다).
- `Reset`: `_hold?.RecordSent(v, Staged)`, 계획이 끝난 경우(`Reset(v)`)만 `_hold?.EndedWhileHolding(v)`(대기 중이면 켠 엔진 Shutdown + `clamp_hold_failed {plan_ended}`, 검토 R4-2-2), 그 뒤 `_hold = null`. `Clear`(리로드 인계)는 끄지 않는다. `RecordHandedRequests`도 `clamp_hold` 안의 요청을 같은 규칙으로 기록한다.
- `ModeAbortException`은 `ComputeCommand`를 거쳐 `Coordinator.Tick`의 기존 `catch (ModeAbortException e)`로 간다(`Abort(e.Reason, …)`).
