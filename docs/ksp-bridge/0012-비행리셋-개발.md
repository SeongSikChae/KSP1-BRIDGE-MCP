# 비행리셋 개발

이슈: [#8](https://github.com/SeongSikChae/KSP1-BRIDGE-MCP/issues/8) · 방안: [0010](0010-비행리셋-방안.md) · 설계: [0011](0011-비행리셋-설계.md)

## 공통

- 신규 파일: `src/KspBridge.Logic/Flight/FlightReset.cs` (static 진입점 + 게임 이벤트용 인스턴스). 나머지는 기존 파일에 추가한다.
- 게임 확인은 MCP 응답으로만 한다(화면·키 입력 없음). 각 단위 끝에 `dotnet build KspBridge.slnx`, `dotnet test tests/KspBridge.App.Tests`, `reload_mod`(Logic)·`reload_app`(App) 뒤 실측.
- 계획 중단 reason에 `revert`, `recover`, `leave`를 추가한다. reason 목록은 `ReadMe.md`의 "중단 reason"(202행 부근)에 있으므로 거기에 넣는다(`abort_flight` 설명은 `reason user`만 적음, 그대로 둠).
- 판정 순서(revert_flight·leave_flight 기준): 회수 대기(`recovery_pending`) → 씬 로드(`scene_loading`) → 씬·활성 기체(`Guard`) → 게임·모드(`no_game`, `not_supported`). 로드 중에는 `LoadedScene`·`ActiveVessel`을 믿을 수 없으므로 로드 판정을 먼저 한다. 라우트의 본문 파싱(`target` 누락 → 400)은 그보다 먼저일 수 있다.
  - 예외 `recover_vessel`: 게임 판정(`no_game`)을 씬 판정보다 먼저 한다(게임이 없으면 씬 판정과 회수가 의미 없음).
  - 예외 `launch_craft(clear_site)`: 기존 검사(`EditorScene.Require`, 발사 잠금, site, 구조) 뒤에 `scene_loading`만 본다. 회수 대기는 볼 필요가 없다: 대기 중 EDITOR 로드를 요청하면 `scene_changed`로 대기가 끝나므로 조립동에서는 대기가 있을 수 없다.
- `scene_ready`가 오래 false면(`loading_seconds`가 계속 늘어남) `reload_mod`로 해제한다. 씬 전환 도구 설명과 ReadMe에 이 문장을 넣는다.
- MCP만으로 재현할 수 없는 경로는 코드 리뷰로 대신하고 이 문서 "실측 결과"에 그렇게 적는다: 게임 설정 변경이 필요한 것(`revert_unavailable`, `leave_unavailable` 중 `CanLeaveToSpaceCenter`, `save_disabled`), 미션 게임, `scene_changed`·`timeout`·`vessel_not_found`.
- 씬을 바꾸는 도구(`revert_flight`, `leave_flight`, 비행 씬 `recover_vessel`, `launch_craft`) 뒤의 실측은 `get_status.reset.requested_scene`이 목적 씬이고 `scene_ready`가 true일 때까지 폴링한다(`scene` 이름은 요청 시점에 바뀜). `launch_craft`는 게임이 로드를 한 프레임 늦게 요청하므로 `requested_scene` 확인이 꼭 필요하다. EDITOR에 머물고 `input_locks`에 `EDITOR_LAUNCH` 잠금이 보이면 발사 전 검사 대화상자에서 멈춘 것이다.

```csharp
internal static bool IsStockGame() =>
    HighLogic.CurrentGame != null && HighLogic.CurrentGame.Mode != Game.Modes.MISSION
    && HighLogic.CurrentGame.Mode != Game.Modes.MISSION_BUILDER && HighLogic.CurrentGame.Mode != Game.Modes.SCENARIO_NON_RESUMABLE;

internal static void RequireStockGame(string tool)
{
    if (HighLogic.CurrentGame == null) throw new ApiException(409, "no_game", "load a save first");
    if (!IsStockGame())
        throw new ApiException(409, "not_supported", tool + " does not support " + HighLogic.CurrentGame.Mode + " games");
}

private static bool _sceneLoading;
internal static bool SceneReady => !_sceneLoading;

internal static void RequireSceneReady()
{
    if (_sceneLoading) throw new ApiException(409, "scene_loading", "a scene is still loading; poll get_status until reset.scene_ready");
}

private static void RequireNoPendingRecovery()
{
    if (_pendingId != null) throw new ApiException(409, "recovery_pending", "a vessel recovery is in progress (get_status reset.recovery)");

{
    if (!HighLogic.CurrentGame.Parameters.Flight.CanAutoSave)
        throw new ApiException(409, "save_disabled", "this game does not allow saving the persistent file");

// 스톡 게임에서는 시작 씬 인자가 쓰이지 않는다(Game.Updated의 미션 분기 전용, 미션 게임은 RequireStockGame이 막음)

// onVesselWillDestroy에만 반응하므로 DestroyImmediate·RecoverVesselFromFlight 뒤에는 직접 갱신해야 한다.
```

```csharp
public static object Revert(string target)
{
    RequireNoPendingRecovery();                           // 대기 중 재호출은 로드 중이어도 recovery_pending
    RequireSceneReady();                                  // 비행 씬은 onFlightReady(PostInit 백업) 뒤에야 ready
    Guard.ActiveVessel();                                 // 409 not_in_flight / no_active_vessel
    RequireStockGame("revert_flight");
    bool editor = target == "editor";
    if (!editor && target != "launch") throw ApiException.BadRequest("target must be launch or editor");
    List<string> why = RevertBlockers(editor, out EditorFacility facility);
    if (why.Count > 0)
        throw new ApiException(409, "revert_unavailable", "cannot revert to " + target + ": " + string.Join(", ", why));
    if (Coordinator.Running) Coordinator.Abort("revert", null, null);
    if (editor) FlightDriver.RevertToPrelaunch(facility); else FlightDriver.RevertToLaunch();
    var r = new Dictionary<string, object> { ["accepted"] = true, ["target"] = target };
    if (editor) r["facility"] = facility.ToString();
    return r;
}

// PauseMenu.drawStockRevertOptions와 같은 판정
internal static List<string> RevertBlockers(bool editor, out EditorFacility facility)
{
    GameParameters.FlightParams p = HighLogic.CurrentGame.Parameters.Flight;
    List<string> why = new List<string>();
    facility = EditorFacility.None;
    if (!editor)
    {
        if (!FlightDriver.CanRevertToPostInit) why.Add("no_post_init_state");
        if (!p.CanRestart) why.Add("restart_disabled");
        return why;
    }
    if (!p.CanLeaveToEditor) why.Add("leave_to_editor_disabled");
    if (!FlightDriver.CanRevertToPrelaunch || ShipConstruction.ShipConfig == null) why.Add("no_prelaunch_state");
    else
    {
        facility = ShipConstruction.ShipType;
        if (facility != EditorFacility.VAB && facility != EditorFacility.SPH) why.Add("unknown_facility");
    }
    return why;
}
```

**검증**
- App 테스트: `ToolsMapToHttpRequests`에 `POST /api/flight/revert {"target":"launch"}`, `ExpectedTools`에 `revert_flight` 추가 후 통과.
- 게임: 발사 → 계획 제출(`attitude` 단계, 스로틀 1로 상승) → 고도가 오른 뒤 `revert_flight launch` → `reset.scene_ready` true까지 폴링 → scene FLIGHT, 이벤트 `plan_aborted reason revert`, `get_flight` 고도가 발사대 높이·MET 0 부근. 로드 중 즉시 재호출 → 409 `scene_loading`. `revert_flight editor` → `scene_ready` 뒤 scene EDITOR, `editor.ready`. 조립동에서 호출 → 409 `not_in_flight`. 비행 씬(`scene_ready` true)에서 `target: x` → 400.
- 게임: 직전 비행은 다른 기체로 해 둔다. 조립동에서 `launch_craft` → `requested_scene` FLIGHT && `scene_ready` true가 된 즉시 활성 기체 id·이름 기록(`get_flight`) → `revert_flight launch` → accepted → `scene_ready` 뒤 활성 기체 id·이름이 같고 MET 0 부근(직전 비행이 아니라 이번 발사 직후). `scene_ready`가 true가 되기 전(`requested_scene` FLIGHT, `loading_seconds` 증가) `revert_flight` → 409 `scene_loading`.
- 이 단위의 `reset`은 최소형(`scene_ready`, `requested_scene`, `loading_seconds`)이고, 나머지 필드는 단위 5에서 더한다.
- 코드 리뷰로 대신: `revert_unavailable`(게임 설정 변경 필요), `not_supported`(미션 게임).

```csharp
public static object Leave()
{
    RequireNoPendingRecovery();                           // 대기 중 재호출은 로드 중이어도 recovery_pending
    RequireSceneReady();
    Guard.ActiveVessel();
    RequireStockGame("leave_flight");
    RequireLeave();                                       // CanAutoSave는 보지 않음: 게임 Space Center 버튼과 같음(설계 결정 10)
    if (Coordinator.Running) Coordinator.Abort("leave", null, null);
    Save();
    HighLogic.LoadScene(GameScenes.SPACECENTER);
    return new Dictionary<string, object> { ["accepted"] = true };
}

// 라우트는 body를 그대로 넘긴다: 비행 씬에서는 site를 읽지 않는다("" 포함)
public static object Recover(RequestBody b)
{
    RequireNoPendingRecovery();                           // 대기 중 재호출은 로드 중이어도 recovery_pending
    RequireSceneReady();
    RequireStockGame("recover_vessel");                   // no_game(메인 메뉴)이 씬 판정보다 먼저(LoadedScene MAINMENU면 CurrentGame null)
    GameScenes s = HighLogic.LoadedScene;
    // 단위 2: if (s == GameScenes.FLIGHT) throw ApiException.SceneNotAllowed("recover_vessel");  단위 4에서 아래로 바꾼다
    if (s == GameScenes.FLIGHT) return RecoverActive();
    if (s != GameScenes.EDITOR && s != GameScenes.SPACECENTER) throw ApiException.SceneNotAllowed("recover_vessel");
    string site = b.Has("site") ? b.GetString("site") : null;
    if (string.IsNullOrWhiteSpace(site)) throw ApiException.BadRequest("site is required outside the flight scene");
    return new Dictionary<string, object> { ["recovered"] = RecoverAtSite(site.Trim(), exact: true) };
}

// 발사 지점만: 시설 발사 지점(LaunchPad, Runway) 또는 추가 발사 지점. IsFacilityOrLaunchSite는 일반 시설도 받아서 쓰지 않는다.
private static void RequireLaunchSite(string site)
{
    PSystemSetup ps = PSystemSetup.Instance;
    bool ok = ps.SpaceCenterFacilityLaunchSites.Any(f => f.name == site || f.facilityName == site) || ps.IsLaunchSite(site, out _);
    if (!ok) throw ApiException.BadRequest("unknown launch site '" + site + "'");
}

// 회수 전후 차이(계약 보상 포함). 인스턴스가 없으면 null. crew는 회수 전에 읽는다(회수하면 명단으로 돌아감).
private static Dictionary<string, object> Measure(ProtoVessel pv, Action recover)
{
    double? f0 = Funding.Instance?.Funds;
    float? s0 = ResearchAndDevelopment.Instance?.Science, r0 = Reputation.Instance?.reputation;
    List<string> crew = pv.GetVesselCrew().Select(c => c.name).ToList();
    string name = pv.vesselName;
    uint id = pv.persistentId;               // get_flight·get_status와 같은 식별자
    recover();
    return new Dictionary<string, object>
    {
        ["vessel"] = name, ["id"] = id, ["crew"] = crew,
        ["funds"] = Funding.Instance?.Funds - f0,
        ["science"] = ResearchAndDevelopment.Instance?.Science - s0,
        ["reputation"] = Reputation.Instance?.reputation - r0,
    };
}

// exact: recover_vessel(site)는 landedAt 정확 일치, clear_site는 게임 LaunchSiteClear와 같은 부분 문자열(FindVesselsLandedAt 그대로)
internal static List<Dictionary<string, object>> RecoverAtSite(string site, bool exact)
{
    RequireLaunchSite(site);
    // 설계 결정 15: VesselRecovery가 SpaceCenter.Instance.cb를 읽다 실패하면 이벤트가 예외를 삼켜 자금·승무원 처리 없이 기체만 지워진다
    if (SpaceCenter.Instance == null)
        throw new ApiException(409, "not_supported", "vessel recovery needs the space center, which is not loaded in scene " + HighLogic.LoadedScene);
    RequireSave();
    FlightState state = HighLogic.CurrentGame.flightState;
    List<ProtoVessel> found = ShipConstruction.FindVesselsLandedAt(state, site);
    if (exact) found = found.Where(pv => pv.landedAt == site).ToList();
    List<Dictionary<string, object>> results = new List<Dictionary<string, object>>();
    bool attempted = false;                  // 이벤트(보상 지급)가 나갔을 수 있으면 저장한다
    try
    {
        foreach (ProtoVessel pv in found)
        {
            attempted = true;
            results.Add(Measure(pv, () => ShipConstruction.RecoverVesselFromFlight(pv, state, quick: true)));
        }
    }
    finally
    {
        if (attempted)
        {
            Save();                   // KSCVesselMarkers.RecoverVessel과 같은 순서. 발사는 persistent를 로드한다
            Safe(RefreshKscMarkers);  // 우주센터면 지운 기체의 마커가 남지 않게
        }
    }
    return results;
}
```

## 단위 4. recover_vessel (비행 씬 활성 기체)

**대상**: 회수 대기 필드(`_recovery`, `_loaded`, `_deadline`)와 `Fail`, `FlightReset.RecoverActive`·`Tick`·`CheckDeadline`, `Recover`의 비행 씬 분기, 단위 1의 `OnLoadRequested`·`OnLevelReady`에 회수 대기 처리 추가, `Stop`에서 대기 상태 정리, `BridgeLogic.FixedUpdate`, `Status`에 `recovery`·`can_recover` 추가(아래 단위 5 코드의 해당 부분; 이 단위 검증에 필요)

```csharp
private static Guid? _pendingId;            // 회수 대기 기체
private static bool _loaded;                // 우주센터 GUIReady를 받음
// 회수 시점은 렌더 프레임 차이(Time.frameCount - _readyFrame)로 판정한다. _readyFrame은 단위 1의 공통 필드
private static float _deadline;             // Time.realtimeSinceStartup 기준: 요청 → GUIReady 300초, GUIReady → 회수 30초
private static Dictionary<string, object> _recovery;   // get_status.reset.recovery

private static object RecoverActive()
{
    Vessel v = Guard.ActiveVessel();
    if (!v.IsRecoverable)
        throw new ApiException(409, "not_recoverable", "the active vessel must be landed or splashed on " + FlightGlobals.GetHomeBodyName() + " (situation " + v.situation + ")");
    RequireLeave();                                         // 게임 회수 버튼과 같은 판정
    RequireSave();
    if (Coordinator.Running) Coordinator.Abort("recover", null, null);
    Save();                                                 // VesselRetrieval과 같은 순서. 실패하면 예외로 끝나고 대기는 남지 않는다
    _pendingId = v.id; _loaded = false; _deadline = Time.realtimeSinceStartup + 300f;
    _recovery = new Dictionary<string, object> { ["state"] = "pending", ["vessel"] = v.vesselName, ["id"] = v.persistentId };   // 내부 추적은 _pendingId(Guid)
    HighLogic.LoadScene(GameScenes.SPACECENTER);
    return new Dictionary<string, object> { ["accepted"] = true, ["vessel"] = v.vesselName };
}

private static GameScenes? _requestedScene;  // 마지막으로 요청된 씬: 그 씬의 준비 신호만 로드 중 표시를 푼다(시작 뒤 요청 없으면 null)
private static float _loadStarted;            // 마지막 로드 요청 시각(Time.realtimeSinceStartup)
private static int _readyFrame;               // 요청한 씬이 준비된 프레임(Time.frameCount): 회수 시점·마커 갱신 지연에 씀

// GameEvents.onGameSceneLoadRequested: 모든 로드 요청에서 씬 로드 중으로 표시(단위 1에서 먼저 넣음).
// 회수 대기 중이면 우리가 요청한 SPACECENTER 외의 로드 요청은 대기를 끝낸다
private void OnLoadRequested(GameScenes scene)
{
    _sceneLoading = true;
    _requestedScene = scene;
    _loadStarted = Time.realtimeSinceStartup;   // reset.loading_seconds
    if (_pendingId == null) return;
    if (scene != GameScenes.SPACECENTER) { Fail("scene_changed"); return; }
    _loaded = false; _deadline = Time.realtimeSinceStartup + 300f;   // 우리 요청 또는 재요청: 로드 대기로(단위 4)
}

// GameEvents.onFlightReady: FlightDriver PostInit이 PostInitState를 만든 뒤에 옴. 비행 씬은 여기서 ready
private void OnFlightReady()
{
    if (_requestedScene == GameScenes.FLIGHT) { _sceneLoading = false; _readyFrame = Time.frameCount; }
}

// GameEvents.onLevelWasLoadedGUIReady (LOADINGBUFFER에서는 안 옴). 비행 씬은 onFlightReady가 푼다.
// 요청한 씬과 다른 씬의 GUIReady(예: FlightDriver.Start가 SPACECENTER를 다시 요청한 뒤 오는 FLIGHT GUIReady)는 무시
private void OnLevelReady(GameScenes scene)
{
    if (scene == _requestedScene && scene != GameScenes.FLIGHT) { _sceneLoading = false; _readyFrame = Time.frameCount; }
    if (_pendingId == null || scene != _requestedScene) return;
    if (scene == GameScenes.SPACECENTER)
    {
        _loaded = true;
        _deadline = Time.realtimeSinceStartup + 30f;        // 회수 제한은 로드가 끝난 뒤부터
    }
    else Fail("scene_changed");
}

// 시간 제한: FixedUpdate가 오지 않아도(timeScale 0) get_status가 판정하도록 Tick과 Status 양쪽에서 부른다
private static void CheckDeadline()
{
    if (_pendingId != null && Time.realtimeSinceStartup > _deadline) Fail("timeout");
}

// BridgeLogic.FixedUpdate에서 Coordinator.Tick 다음에 호출. GUIReady 뒤 렌더 프레임 8개(VesselRetrieval: sceneLoaded 뒤 yield return null × 8,
// GUIReady는 sceneLoaded+1이라 게임보다 1프레임 늦음)를 FixedUpdate 횟수가 아니라 프레임 차이로 판정한다
public static void Tick()
{
    CheckDeadline();
    // _sceneLoading: 대기 중 SPACECENTER가 다시 요청되면 새 로드가 끝날 때까지 기다린다
    if (_pendingId == null || !_loaded || _sceneLoading || Time.frameCount - _readyFrame < 8) return;
    Guid id = _pendingId.Value;
    bool fired = false;                       // 회수 이벤트(보상 지급)가 나갔으면 실패해도 저장한다
    try
    {
        Vessel v = FlightGlobals.Vessels.FirstOrDefault(x => x.id == id);
        if (v == null) { Fail("vessel_not_found"); return; }
        Dictionary<string, object> r = Measure(v.protoVessel, () =>
        {
            fired = true;
            GameEvents.onVesselRecovered.Fire(v.protoVessel, true);   // quick: VesselRecovery가 결과 창을 만들지 않음
            UnityEngine.Object.DestroyImmediate(v.gameObject);
        });
        Save();
        Safe(RefreshKscMarkers);              // FixedUpdate가 늦게 와 마커 초기 생성 뒤에 회수했어도 지운 기체의 마커가 남지 않게
        r["state"] = "done";
        _recovery = r;
        _pendingId = null;
    }
    catch (Exception e)
    {
        if (fired) Safe(Save);
        Fail(e.GetType().Name + ": " + e.Message);
    }
}

private static void Fail(string error)
{
    if (_recovery != null) { _recovery["state"] = "failed"; _recovery["error"] = error; }
    _pendingId = null; _loaded = false;
}
```

- `Safe(Action)`(공통 절)은 실패 뒤 저장에서 저장 실패가 원래 오류를 가리지 않게 쓴다.
- `BridgeLogic.StartCore`·`Stop`의 Hook/Unhook은 단위 1에서 넣었다: `StartCore`는 `FlightScene.Hook()`과 같은 try 안에서 `FlightReset.Hook()`(실패 시 둘 다 Unhook), `Stop`은 `FlightReset.Unhook()` 후 `_sceneLoading`·`_requestedScene`을 지운다. 이 단위에서 `Stop`에 대기 상태·`_recovery` 정리를 더한다(설계: Handover 대상 아님). 이 단위에서는 `FixedUpdate`만 더한다: `Coordinator.Tick(); FlightReset.Tick();`(Tick 예외는 Fail로 흡수되어 Coordinator에 영향 없음). `Hook`에 `onFlightReady`는 단위 1에서 이미 들어 있다.

**검증**
- 게임: 발사(발사대 착지, `IsRecoverable`) → `scene_ready` 뒤 `reset.can_recover` true까지 폴링(발사 직후 기체가 미세하게 움직여 `ClearToSave`가 잠시 CLEAR가 아닐 수 있음) → `recover_vessel` → `{accepted}` → 즉시 `recover_vessel` 재호출 → 409 `recovery_pending` → `reset.scene_ready` true와 `reset.recovery.state` done까지 폴링(커리어면 funds > 0, crew에 승무원 이름) → scene SPACECENTER. 이어서 `recover_vessel site LaunchPad` → `recovered []`(이미 회수됨). 결과 창이 없음은 디컴파일 근거(`VesselRecovery.OnVesselRecovered`의 `if (!quick)` 분기에서만 `CreateFullDialog`)로 판단하고, 보조 증거로 회수 뒤 `get_status.input_locks`에 새 대화상자 잠금이 없는지 본다.
- 계획 실행 중(`attitude`, 스로틀 0, 착지 상태) `recover_vessel` → 이벤트 `plan_aborted reason recover`.
- 비행 중(FLYING) → 409 `not_recoverable`.
- 코드 리뷰로 대신: `scene_changed`, `timeout`, `vessel_not_found`, 이벤트 뒤 예외 시 저장(#7 디버그 도구 대상), 우주센터 마커 갱신(화면 UI).

## 단위 5. get_status reset, ReadMe

**대상**: `FlightReset.Status`의 revert 필드(`can_revert_*`, `revert_editor_facility`, `revert_unavailable`; `scene_ready` 등은 단위 1, `recovery`·`can_recover`는 단위 4에서 이미 넣음), `FlightTools.GetStatus` 설명, `ReadMe.md`. 아래 코드는 완성형이다.

```csharp
public static Dictionary<string, object> Status()
{
    CheckDeadline();                                        // FixedUpdate가 멈춰도 timeout 판정
    // 로드가 끝난 비행 씬에서만 판정(LoadedScene은 요청 시점에 바뀜)
    bool flight = HighLogic.LoadedScene == GameScenes.FLIGHT && !_sceneLoading && FlightGlobals.ActiveVessel != null;
    bool stock = IsStockGame();
    EditorFacility f = EditorFacility.None;
    List<string> launch = new List<string>(), editor = new List<string>();
    if (flight && !stock) launch.Add("not_supported");
    else if (flight)
    {
        launch = RevertBlockers(false, out _);
        editor = RevertBlockers(true, out f);
    }
    bool canRevertLaunch = flight && stock && launch.Count == 0;
    bool canRevertEditor = flight && stock && editor.Count == 0;
    Vessel v = flight ? FlightGlobals.ActiveVessel : null;
    bool canRecover = flight && stock && _pendingId == null && v.IsRecoverable
        && FlightGlobals.ClearToSave(false) == ClearToSaveStatus.CLEAR          // 폴링마다 게임 로그가 쌓이지 않게 logMsg:false
        && HighLogic.CurrentGame.Parameters.Flight.CanLeaveToSpaceCenter
        && HighLogic.CurrentGame.Parameters.Flight.CanAutoSave;
    return new Dictionary<string, object>
    {
        ["scene_ready"] = !_sceneLoading,                                       // 이 세 필드는 단위 1의 최소형과 같음
        ["requested_scene"] = _requestedScene?.ToString(),
        ["loading_seconds"] = _sceneLoading ? (object)(double)(Time.realtimeSinceStartup - _loadStarted) : null,
        ["can_recover"] = canRecover,
        ["can_revert_launch"] = canRevertLaunch,
        ["can_revert_editor"] = canRevertEditor,
        ["revert_editor_facility"] = canRevertEditor ? f.ToString() : null,
        ["revert_unavailable"] = flight ? launch.Concat(editor).Distinct().ToList() : null,
        // 복사본: 응답은 HTTP 스레드에서 직렬화되고(HttpServer), Fail은 메인 스레드에서 _recovery를 고친다
        ["recovery"] = _recovery == null ? null : new Dictionary<string, object>(_recovery),
    };
}

// BridgeLogic.Status: input_locks와 같이 격리
try { status["reset"] = FlightReset.Status(); }
catch (Exception e) { status["reset"] = null; status["reset_error"] = e.GetType().Name + ": " + e.Message; }
```
