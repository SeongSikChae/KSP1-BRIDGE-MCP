## U1. Logic: 이벤트 버퍼 `Flight/BroadcastEvents.cs`
- **대상**: 새 파일 `src/KspBridge.Logic/Flight/BroadcastEvents.cs`. `BridgeLogic.Start`, `FixedUpdate`, `Stop`도 고칩니다.
- **코드 초안**:
```csharp
public static class BroadcastEvents
{
    private const int MaxEvents = 500;
    private static readonly List<Dictionary<string, object>> Events = new List<Dictionary<string, object>>();
    private static bool _armed;
    private static long _seq;
    private static int _separations, _atmosphereIndex = -1;
    private static bool _liftedOff;
    private static double[] _altitudes = new double[0];
    private static readonly HashSet<int> Crossed = new HashSet<int>();
    private static EventData<GameEvents.HostedFromToAction<Vessel, Vessel.Situations>>.OnEvent _onSituation;
    private static EventData<GameEvents.HostedFromToAction<Vessel, CelestialBody>>.OnEvent _onSoi;
    private static EventData<EventReport>.OnEvent _onSeparation;

    public static void Hook()
    {
        _onSituation = OnSituation; _onSoi = OnSoi; _onSeparation = OnSeparation;
        GameEvents.onVesselSituationChange.Add(_onSituation);
        GameEvents.onVesselSOIChanged.Add(_onSoi);
        GameEvents.onStageSeparation.Add(_onSeparation);
    }

    public static void Arm(double[] altitudes, double atmosphereDepth)
    {
        lock (Events) { Events.Clear(); }
        _separations = 0; Crossed.Clear(); _liftedOff = false; _altitudes = altitudes;
        _atmosphereIndex = Array.IndexOf(altitudes, atmosphereDepth);   // 대기권 없으면 -1
        _armed = true;
    }

    private static void OnSituation(GameEvents.HostedFromToAction<Vessel, Vessel.Situations> e)
    {
        if (!_armed || e.host != FlightGlobals.ActiveVessel) return;
        if (e.to == Vessel.Situations.FLYING && (e.from == Vessel.Situations.PRELAUNCH || e.from == Vessel.Situations.LANDED) && !_liftedOff)
        { _liftedOff = true; Add("liftoff", "vessel", e.host.vesselName); return; }
        if (e.to == Vessel.Situations.ORBITING || e.to == Vessel.Situations.LANDED || e.to == Vessel.Situations.SPLASHED)
            Add("situation", "situation", e.to.ToString(), "body", e.host.mainBody.bodyName);
    }

    public static void FixedUpdate()   // 고도 돌파
    {
        Vessel v = FlightGlobals.ActiveVessel;
        if (!_armed || v == null) return;
        for (int i = 0; i < _altitudes.Length; i++)
            if (!Crossed.Contains(i) && v.altitude >= _altitudes[i])
            { Crossed.Add(i); Add("altitude", "meters", _altitudes[i], "atmosphere", i == _atmosphereIndex); }
    }

    private static void Add(string type, params object[] kv)
    {
        var e = new Dictionary<string, object> { ["seq"] = ++_seq, ["type"] = type,
            ["rt"] = (double)Time.realtimeSinceStartup, ["ut"] = Planetarium.GetUniversalTime(),
            ["met"] = FlightGlobals.ActiveVessel?.missionTime ?? 0 };
        for (int i = 0; i + 1 < kv.Length; i += 2) e[(string)kv[i]] = kv[i + 1];
        lock (Events) { Events.Add(e); if (Events.Count > MaxEvents) Events.RemoveAt(0); }
    }
    // ... Since(long since) → ageMs 계산, Disarm, Unhook ...
}
```
  - 분리 이벤트(`onStageSeparation`)는 `EventReport.origin.vessel == ActiveVessel`일 때만 `++_separations`를 `index`로 기록합니다.
  - `BridgeLogic`에는 기존 FlightRecorder 블록과 같은 모양으로 넣습니다: `BroadcastEvents.Hook()`, try/catch로 감싼 `BroadcastEvents.FixedUpdate()`, `BroadcastEvents.Unhook()`.
- **검증**: `dotnet build KspBridge.slnx`가 성공합니다(KspDir 경로 설정 필요).

```csharp
public static class Ssml
{
    public static string Build(string voice, string text) =>
        $"<speak version=\"1.0\" xmlns=\"http://www.w3.org/2001/10/synthesis\" xml:lang=\"ko-KR\"><voice name=\"{SecurityElement.Escape(voice)}\">{SecurityElement.Escape(text)}</voice></speak>";

public sealed class SpeechClient(HttpClient http, SpeechSettings s)
{
    public async Task<string> SynthesizeAsync(string text, string cacheDir, CancellationToken ct)
    {
        string file = Path.Combine(cacheDir, Hash(s.Voice + "\n" + text) + ".wav");
        if (File.Exists(file)) return file;
        using HttpRequestMessage req = new(HttpMethod.Post, $"https://{s.Region}.tts.speech.microsoft.com/cognitiveservices/v1")
        { Content = new StringContent(Ssml.Build(s.Voice, text), Encoding.UTF8, "application/ssml+xml") };
        req.Headers.Add("Ocp-Apim-Subscription-Key", s.ApiKey);
        req.Headers.Add("X-Microsoft-OutputFormat", "riff-24khz-16bit-mono-pcm");
        req.Headers.UserAgent.ParseAdd("KspBridge");
        using HttpResponseMessage res = await http.SendAsync(req, ct);
        res.EnsureSuccessStatusCode();                 // 메시지에 키를 넣지 않음
        Directory.CreateDirectory(cacheDir);
        await File.WriteAllBytesAsync(file, await res.Content.ReadAsByteArrayAsync(ct), ct);
        return file;
    }
}
```

- **대상**: 새 파일 `Broadcast/Narration.cs`, `Cue.cs`
- **코드 초안**:
```csharp
public sealed record Cue(double At, string Text, string? Audio = null, double Beep = 0);

public static class Narration
{
    public static string Subject(string name)          // 이/가
    {
        char c = name.TrimEnd()[^1];
        bool batchim = c is >= '가' and <= '힣' ? (c - '가') % 28 != 0
                     : char.IsDigit(c) ? "0136780".Contains(c)   // 영,일,삼,육,칠,팔,공
                     : "LMNRlmnr".Contains(c);
        return name + (batchim ? "이" : "가");
    }

    public static IEnumerable<(int T, string Text, double Beep)> Countdown(string craft, int seconds)
    {
        if (seconds >= 20) yield return (20, $"{craft}, 발사 20초 전입니다.", 0);
        if (seconds >= 10) yield return (10, "발사 10초 전.", 0);
        for (int t = Math.Min(seconds, 9); t >= 1; t--) yield return (t, t.ToString(), 0.15);
        yield return (0, "발사!", 0.6);
    }

    public static string? ForEvent(JsonElement e) => e.GetProperty("type").GetString() switch
    {
        "liftoff" => $"{Subject(Str(e, "vessel"))} 발사되었습니다.",
        "separation" => $"{e.GetProperty("index").GetInt32()}단이 분리되었습니다.",
        "altitude" when e.GetProperty("atmosphere").GetBoolean() => "대기권을 벗어났습니다.",
        "altitude" => $"고도 {e.GetProperty("meters").GetDouble() / 1000:0.#}킬로미터를 돌파했습니다.",
        "situation" => Str(e, "situation") switch { "ORBITING" => $"{Str(e, "body")} 궤도에 진입했습니다.",
            "LANDED" => $"{Str(e, "body")}에 착륙했습니다.", "SPLASHED" => $"{Str(e, "body")}에 착수했습니다.", _ => null },
        "soi" => $"{Str(e, "to")} 영향권에 진입했습니다.",
        _ => null,
    };
}
```

```csharp
    public static (List<string> Args, string FilterScript) Compose(IReadOnlyList<Cue> cues)
    {

        {

            if (c.Beep > 0) { args.AddRange(["-f", "lavfi", "-t", c.Beep.ToString(CultureInfo.InvariantCulture), "-i", "sine=frequency=1000:sample_rate=24000"]);
                              filters.Add($"[{input}:a]adelay={ms}|{ms}[a{input}]"); labels.Add($"[a{input}]"); input++; }
        }
        filters.Add($"{string.Concat(labels)}amix=inputs={labels.Count}:normalize=0[aout]");
        filters.Add("[0:v]subtitles=subs.ass[vout]");
        args.AddRange(["-filter_complex_script", "filter.txt", "-map", "[vout]", "-map", "[aout]",
                       "-c:v", "libx264", "-preset", "veryfast", "-crf", "20", "-c:a", "aac", "-shortest", "final.mp4"]);
        return (args, string.Join(";\n", filters));
    }
}
```

  - 음성 cue가 없으면 오디오 없이 `-map [vout]`만 씁니다.
  - `-shortest`: amix가 영상보다 길어지지 않게 합니다.
- **검증**: `FfmpegArgsTests`
  - Record 인수 목록
  - Compose: 입력 순서, adelay 값(ms), 비프 입력, amix 입력 수, 오디오가 없을 때의 경우, 상대 경로만 쓰는지(검토 회귀 2)
  - `FfmpegRecorder`: 가짜 Runner로 `frame=15` 줄을 받으면 0점이 `받은 시각 − 15/fps`인지, 시간이 지나면 실패하는지 확인합니다.
  - 합성 프로세스의 작업 폴더가 출력 폴더인지 확인합니다.

- **대상**: 새 파일 `Broadcast/BroadcastSession.cs`, `BroadcastTools.cs`. `Program`에 DI 등록과 `.WithTools<BroadcastTools>()`를 추가합니다.
- **코드 초안**:
```csharp
public async Task<string> StartAsync(string? craft, int countdown, double[]? altitudes, CancellationToken ct)
{
    lock (_gate) { if (_state is not (State.Idle or State.Done or State.Failed)) return Error("busy"); _state = State.Countdown; }
    string armJson = await _client.SendAsync(HttpMethod.Post, "api/broadcast/arm", new { altitudes }, ct);
    if (armJson.StartsWith("error:")) { _state = State.Idle; return armJson; }
    craft ??= JsonDocument.Parse(armJson).RootElement.GetProperty("vessel").GetString()!;
    var lines = Narration.Countdown(craft, countdown).ToList();
    string[] audio = await Task.WhenAll(lines.Select(l => SynthesizeOrNull(l.Text, ct)));   // 미리 합성
    _dir = CreateDir(craft);
    _zero = await _recorder.StartAsync(Path.Combine(_dir, "raw.mkv"), ct);                  // Stopwatch 0점
    long t0 = _clock.ElapsedTicks + TimeSpan.FromSeconds(countdown).Ticks;
    string? stageResult = null;
    for (int i = 0; i < lines.Count; i++)
    {
        await DelayUntil(t0 - TimeSpan.FromSeconds(lines[i].T).Ticks, ct);                 // 절대 시각 기준
        AddCue(new Cue(Offset(), lines[i].Text, audio[i], lines[i].Beep));
        if (lines[i].T == 0) stageResult = await _client.SendAsync(HttpMethod.Post, "api/stage", null, ct);
    }
    _state = State.Recording; _poll = Task.Run(() => PollLoop(_cts.Token));
    return Json(new { dir = _dir, ignited = !stageResult!.StartsWith("error:"), stage = stageResult, cues = _cues.Count });
}

private async Task PollLoop(CancellationToken ct)
{
    while (!ct.IsCancellationRequested)
    {
        string json = await _client.SendAsync(HttpMethod.Get, $"api/broadcast/events?since={_last}", null, ct);
        double received = Offset();
        // 각 event: text = Narration.ForEvent(e); at = received - ageMs/1000; AddCue(new Cue(at, text)); _last = seq
        await Task.Delay(250, ct);
    }
}
```
  - 카운트다운이 시작되기 전(arm 실패, 녹화 실패)에 실패하면 상태를 `Idle`로 되돌리고 오류를 반환합니다. 녹화 프로세스가 떠 있으면 정리합니다.
  - Stop 흐름:
    1. `_cts.Cancel()`
    2. disarm을 보냅니다(실패해도 무시).
    3. `_recorder.StopAsync`
    4. `Task.Run(ComposeAsync)`를 시작하고 `{id, dir, state:"composing"}`을 반환합니다.
  - ComposeAsync 흐름:
    1. 음성이 없는 이벤트 cue를 합성합니다(실패하면 `ttsFailures`에 넣습니다).
    2. 각 cue의 길이를 구합니다.
    3. `AssWriter`로 `subs.ass`를 씁니다.
    4. filter 스크립트를 쓰고 ffmpeg를 실행합니다.
    5. 결과 상태를 `Done` 또는 `Failed`로 바꿉니다(stderr 마지막 20줄 포함).
  - 시간 소스(`Stopwatch`)와 지연 함수는 생성자에서 주입받습니다. 테스트에서 가짜 시계를 쓰기 위해서입니다.
  - `BroadcastTools`:
```csharp
[McpServerTool(Name = "start_broadcast", Destructive = true)]
[Description("Record the KSP window as a broadcast: starts ffmpeg, counts down with Korean narration (MAI-Voice) and ignites the first stage at T-0 (POST /api/stage). Check get_stages delta-v on the pad BEFORE calling this (launch preflight).     => session is null ? Task.FromResult("error: not_configured: Broadcast section is missing in the config") : session.StartAsync(craftName, Math.Clamp(countdownSeconds, 3, 30), altitudes, ct);
```
  - `Broadcast` 섹션이 없으면 `BroadcastSession`을 등록하지 않습니다. 도구는 `BroadcastSession?`를 받아 `not_configured`를 반환합니다.
- **검증**: `BroadcastToolsTests`
  - 가짜 Runner, 시계, Speech, 기존 `ToolTest.Client()`를 씁니다.

## U9. 문서·실게임 검증
- **대상**: `ReadMe.md`(도구 표, YAML 키 표, ffmpeg 요구사항, Speech 키 암호화 설명), `config/kspbridge.yaml`
  - 검토 주의점: KSP는 창 모드 또는 테두리 없는 창 모드로 실행해야 합니다(전용 전체화면이면 gdigrab이 검은 화면만 녹화할 수 있음). ffmpeg는 5.0 이상이어야 합니다(`amix normalize=0`).
- **코드 초안**(ReadMe 도구 표 행):
```markdown
| 중계 | start_broadcast, stop_broadcast, get_broadcast_status | KSP 창 녹화 + 카운트다운(T-0 점화) + 한국어 자막·음성(MAI-Voice) 합성 MP4 |
```
- **검증**(kspbridge MCP 연결, ffmpeg와 Azure 키 필요):
  1. `dotnet publish src/KspBridge.App -o .mcp-bin-next`를 실행하고, 로직은 `reload_mod`로 올린 뒤 logic 0.10.0인지 확인합니다.
  2. 발사대에서 `get_stages`로 Δv를 확인합니다(rule launch-preflight).
  3. `start_broadcast`의 반환이 `ignited:true`인지 확인합니다.
  4. 궤도에 진입할 때까지 비행합니다(기존 도구 사용).
  5. `stop_broadcast`를 부르고 `get_broadcast_status`가 `done`이 될 때까지 확인합니다.
  6. `ffprobe final.mp4`로 확인합니다: h264와 aac 스트림이 있고, 길이가 raw와 ±1초 이내입니다.
  7. 카운트다운, 이륙, 분리, 고도, 궤도 cue 시각의 프레임을 추출해 자막이 보이는지 확인합니다.
  8. 음성을 직접 들어서 T-0 "발사!"와 화면의 점화가 맞는지 확인합니다(사용자 확인).
