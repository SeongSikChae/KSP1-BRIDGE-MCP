# 정밀착륙 개발

0060-정밀착륙-방안.md, 0061-정밀착륙-설계.md를 구현 지침으로 푼다. 코드 초안은 핵심 부분만 적는다. KSP API 이름 중 "(구현 시 확인)" 표시는 Assembly-CSharp에서 시그니처·단위를 확인한 뒤 쓰고, 다르면 같은 의미의 API로 바꾼다.

```csharp
/// <summary>Landing target argument (#22, docs 0060~0062): latitude+longitude, or a vessel id.
/// Parsing checks the form only (submit and reload re-parse alike); the vessel is resolved when the land step starts.</summary>
public sealed class LandingTarget
{
    public static readonly string[] Args = { "latitude", "longitude", "vessel" };
    public double Latitude { get; }    // NaN for a vessel target
    public double Longitude { get; }
    public double VesselId { get; }    // NaN for a coordinate target
    public bool IsVessel => !double.IsNaN(VesselId);

    /// <summary>null = no target. Throws through invalid(path, reason) on a bad form.</summary>
    internal static LandingTarget Parse(Func<string, bool> has, Func<string, object> get, string path,
                                        Func<string, string, Exception> invalid)
    {
        bool lat = has("latitude"), lon = has("longitude"), ves = has("vessel");
        if (!lat && !lon && !ves) return null;
        if (ves && (lat || lon)) throw invalid(path, "give latitude+longitude or vessel, not both");
        if (ves)
        {
            double id = Number(get("vessel"), path + ".vessel", 0, uint.MaxValue, invalid);
            if (id != Math.Floor(id)) throw invalid(path + ".vessel", "must be an integer vessel id");
            return new LandingTarget(double.NaN, double.NaN, id);
        }
        if (lat != lon) throw invalid(path, "latitude and longitude go together");
        return new LandingTarget(Number(get("latitude"), path + ".latitude", -90, 90, invalid),
                                 Number(get("longitude"), path + ".longitude", -180, 180, invalid), double.NaN);
    }

    /// <summary>Present key with null, a non-number, NaN or out of range: invalid (null is not the default here).</summary>
    private static double Number(object o, string path, double min, double max, Func<string, string, Exception> invalid)
    {
        if (!(o is double d) || double.IsNaN(d) || d < min || d > max)
            throw invalid(path, "must be a number " + min + ".." + max);
        return d;
    }

    internal static double Wrap180(double lon) { lon %= 360; return lon > 180 ? lon - 360 : lon < -180 ? lon + 360 : lon; }
}
```

```csharp
public sealed class DragModel
{
    public double K { get; private set; }        // m², drag accel = K·ρ·v²/m (ρ kg/m³, m kg)
    public string Source { get; private set; }   // "initial" | "measured"
    private const double MeasureQ = 1000, Tau = 3;   // Pa, s

    /// <summary>Area with the vessel bottom (engine side, vessel axis) facing the flow, whatever the current attitude.</summary>
    public static DragModel Initial(Vessel v)
    {
        Vector3 bottom = -v.ReferenceTransform.up;   // world direction of the bottom side
        double area = 0;
        foreach (Part p in v.parts)
        {
            Vector3 local = p.transform.InverseTransformDirection(bottom);
            area += p.DragCubes.GetCubeAreaDir(local);   // (구현 시 확인: 면적 m², 방향 부호)
        }
        return new DragModel { K = 0.5 * area * PhysicsGlobals.DragCubeMultiplier * PhysicsGlobals.DragMultiplier, Source = "initial" };
    }

    /// <summary>Physics tick: KSP's own per-part drag forces (no velocity differencing, thrust model or frame correction).</summary>
    public void Measure(Vessel v, double dt)
    {
        if (v.dynamicPressurekPa * 1000 < MeasureQ || dt <= 0 || v.atmDensity <= 0) return;
        double force = 0;   // kN
        foreach (Part p in v.parts) force += p.dragScalar;   // (구현 시 확인: dragScalar 단위·의미, 바디 리프트 제외)
        double vr = v.srfSpeed;
        double kNow = force * 1000 / (v.atmDensity * vr * vr);
        K += (1 - Math.Exp(-dt / Tau)) * (kNow - K);   // blended from the first sample
        Source = "measured";
    }
}
```

```csharp
public sealed class Impact { public double Lat, Lon, Ut, Alt; }

/// <summary>Unpowered RK4 from (r0, v0): position relative to the body centre and inertial velocity, both in world axes at
/// "now". ut0 may be in the future (deorbit node): the body rotation from now to each sample time is applied.</summary>
public static Impact Predict(CelestialBody body, Vector3d r0, Vector3d v0, double ut0, double mass, DragModel drag,
                             Stopwatch budget = null, double budgetMs = double.PositiveInfinity)
{
    double now = Planetarium.GetUniversalTime();
    double maxTerrain = MaxTerrain(body);   // pqsController.mapMaxHeight, 0 without PQS (구현 시 확인)
    Vector3d r = r0, vel = v0, prevR = r0;
    double t = 0, prevAbove = double.NaN;
    for (int i = 0; i < MaxSteps && t < MaxS; i++)
    {
        if (budget != null && budget.Elapsed.TotalMilliseconds > budgetMs) return null;   // caller reports the timeout
        double alt = r.magnitude - body.Radius;
        double rot = ut0 - now + t;                                                       // rotation seconds from now
        double ground = alt < maxTerrain ? Ground(body, r, rot) : double.NegativeInfinity;   // terrain only below max terrain
        double above = alt - (body.ocean ? Math.Max(ground, 0) : ground);
        if (above <= 0)
        {
            return i == 0 ? Surface(body, r, rot, ut0 + t)                                // starts below: here
                          : Interpolate(body, prevR, prevAbove, r, above, rot, ut0 + t);
        }
        double dt = alt > Math.Max(body.atmosphereDepth, maxTerrain) ? 10
                  : alt > maxTerrain || above > SurfaceBand ? 1 : 0.2;   // terrain-relative fine band (SurfaceBand 2000 m)
        prevR = r; prevAbove = above;
        Rk4(body, mass, drag, ref r, ref vel, dt);
        t += dt;
    }
    return null;   // no impact within MaxS (3600 s) / MaxSteps (20000)
}

/// <summary>Latitude/longitude of an inertial position after rotSeconds of body rotation from now.</summary>
private static void LatLon(CelestialBody body, Vector3d r, double rotSeconds, out double lat, out double lon)
{
    Vector3d axis = body.angularVelocity.normalized;                   // (구현 시 확인: 부호)
    double angleDeg = -body.angularVelocity.magnitude * rotSeconds * 180 / Math.PI;
    Vector3d fixedNow = QuaternionD.AngleAxis(angleDeg, axis) * r;     // rotate back into today's surface frame
    Vector3d world = body.position + fixedNow;
    lat = body.GetLatitude(world);
    lon = LandingTarget.Wrap180(body.GetLongitude(world));
}

private static Vector3d Accel(CelestialBody body, double mass, DragModel drag, Vector3d r, Vector3d vel)
{
    double rm = r.magnitude;
    Vector3d a = -body.gravParameter / (rm * rm * rm) * r;
    double alt = rm - body.Radius;
    if (body.atmosphere && alt < body.atmosphereDepth)
    {
        double rho = body.GetDensity(body.GetPressure(alt), body.GetTemperature(alt));   // (구현 시 확인: kPa·K → kg/m³)
        Vector3d vr = vel - Vector3d.Cross(body.angularVelocity, r);                     // relative to the rotating air
        a -= drag.K * rho * vr.magnitude / mass * vr;                                     // mass kg
    }
    return a;
}
```

**대상**
- `LandMode.cs`: `Update`의 자세 계산(유도 분기), 연료 여유 판정, `landed` 이벤트, `Save`/`Load`

**코드 초안**: 유도 분기 밖은 기존 식 그대로다(설계 결정 6).

```csharp
double maxAccel = EngineModel.MaxAccel(v);   // moved before the attitude; the maxAccel <= 0 branch below is unchanged
double hV = Math.Max(VerticalFirstM, _finalAlt * 5);
bool guide = _burning && _track != null && _track.Guiding && maxAccel > 0;   // Guiding: fixed, kept, not ended
Vector3d vhVec = Vector3d.Exclude(up, v.srf_velocity);
double aV = Math.Max(double.IsNaN(aReq) ? g : aReq, g);              // vertical accel the throttle will make
bool vertical = h <= hV;
Vector3d want = guide && !vertical ? Want(v, up, h, hV, vv, g) : Vector3d.zero;
double capDeg = Math.Min(vertical ? VerticalTiltCap : GuidedTiltCap,  // 15° / 30°
    Math.Max(0, Math.Acos(Clamp(aV / maxAccel, 0, 1)) / Deg - VerticalMarginDeg));   // vertical decel first (−12°)
// enter the loop once the relative horizontal speed fits under the cap; stay in it afterwards
double entryLim = Math.Max(vv, 0) * Math.Tan(capDeg * Deg);
if (guide && !_inLoop && (want - vhVec).magnitude <= entryLim && vh <= entryLim)   // #17 tilt is under the cap too
{
    _inLoop = true;
    // seed with the tilt the #17 formula actually commands this tick (it stops at −srf_velocity): no jump
    double t17 = Math.Min(Vector3d.Angle(up, -v.srf_velocity), Clamp(vh * TiltPerMs, 0, 90));
    _aCmdPrev = vh > 1e-3 ? -vhVec.normalized * aV * Math.Tan(Math.Min(t17, capDeg) * Deg) : Vector3d.zero;
}
AttitudeTarget att;
double maxTilt;
if (!guide || !_inLoop)
{
    maxTilt = Clamp(vh * TiltPerMs, 0, 90);                          // #17 formula, unchanged (no target / free fall / pre-loop)
    att = AttitudeTarget.RotateTowards(up, -v.srf_velocity, maxTilt);
}
else
{
    Vector3d aCmd = ClampMag((want - vhVec) / Tau, aV * Math.Tan(capDeg * Deg));   // τ 4 s
    aCmd = RateLimitTilt(_aCmdPrev, aCmd, aV, TiltRateDeg * (ut - _aCmdUt));       // tilt and direction ≤ 10°/s
    aCmd = ClampMag(aCmd, aV * Math.Tan(capDeg * Deg));                             // the cap stays hard after the rate limit
    _aCmdPrev = aCmd;
    att = AttitudeTarget.Vector(up * aV + aCmd);                     // tilt = atan(|aCmd| / aV) toward aCmd
    maxTilt = Math.Atan2(aCmd.magnitude, aV) / Deg;
}
_aCmdUt = ut;                                                         // every tick: no large step after a thrust gap
_track?.SetGuidance(!_burning ? "free_fall" : !guide ? (_track.Ended ? "dropped" : "off") : !_inLoop ? "pre_loop" : vertical ? "vertical" : "guided");
```

- `aReq`(기존 `(vv² − final_speed²)/(2(h − final_alt)) + g`)는 자세 계산 앞에서 구한다. 값과 쓰임은 그대로다. `final_altitude` 아래(`aReq`가 NaN)에서는 `a_v = g`.
- 스로틀 식(`aReq / aMaxV`, `aMaxV = maxAccel × 실제 cos`)은 그대로다.
- 시간 간격은 `ut − _aCmdUt`(ut 차분)이다. 리로드 뒤 첫 틱도 저장한 `a_cmd_ut`로 이어진다.
- 상한이 1° 아래면(수직 여유 없음) 진입하지 않고 `_track.EndGuidance()` + `Unreachable("no_margin", "braking", …)`로 유도를 끝낸다.
- 진입 전(`pre_loop`)은 기존 식이라 큰 수평 속도를 90°까지 기울여 없앤다. 진입 조건을 만족하면 그 틱 기존 식 명령과 같은 방향·기울임으로 `_aCmdPrev`를 채워 명령이 뛰지 않는다.

```csharp
/// <summary>|want| = min(|d| / tGoV, √(2·a_hLim·|d|), |d| / (4τ)): tGoV to the vertical-first height, a_hLim = 0.5·g·tan 30°;
/// |d|/(4τ) with the τ velocity loop is a critically damped second-order system: no overshoot (decision 6).</summary>
private Vector3d Want(Vessel v, Vector3d up, double h, double hV, double vv, double g)
{
    Vector3d d = Vector3d.Exclude(up, _track.TargetWorld(v) - v.CoMD);
    double dist = d.magnitude;
    if (dist < 1) return Vector3d.zero;
    double tGoV = Math.Max(h - hV, 1) / Math.Max(vv, _finalSpeed);
    double aHLim = 0.5 * g * Math.Tan(GuidedTiltCap * Deg);
    double speed = Math.Min(Math.Min(dist / tGoV, Math.Sqrt(2 * aHLim * dist)), dist / (4 * Tau));
    if (v.mainBody.atmosphere && v.altitude < v.mainBody.atmosphereDepth)
    {
        speed = Math.Min(speed, Math.Max(vv, 0) * Math.Tan(AoaLimitDeg * Deg));   // keep the velocity within 15° of vertical
    }
    return d / dist * speed;
}

/// <summary>Moves the commanded horizontal accel so the commanded tilt and its direction change by at most maxDeg
/// (measured: 10°/s average turn rate below 600 m). Through zero when the direction reverses.</summary>
private static Vector3d RateLimitTilt(Vector3d prev, Vector3d next, double aV, double maxDeg)
{
    // tilt-space vectors: direction × tilt angle (deg); limit the step length to maxDeg, map back with aV·tan
    Vector3d p = TiltVec(prev, aV), n = TiltVec(next, aV);
    Vector3d step = n - p;
    Vector3d t = step.magnitude <= maxDeg ? n : p + step.normalized * maxDeg;
    return t.magnitude < 1e-6 ? Vector3d.zero : t.normalized * aV * Math.Tan(t.magnitude * Deg);
}
```

```csharp
/// <summary>Horizontal Δv (world) moving the predicted impact onto the target: 2x2 Jacobian by finite differences
/// (1 m/s downrange and crossrange), one Newton step. false: no impact or a singular Jacobian.</summary>
public static bool Solve(CelestialBody body, Vector3d r, Vector3d vel, double mass, DragModel drag,
                         double tLat, double tLon, out Vector3d dvWorld)
{
    Vector3d up = r.normalized;
    Vector3d down = Vector3d.Exclude(up, vel).normalized;
    Vector3d cross = Vector3d.Cross(up, down);
    double ut = Planetarium.GetUniversalTime();
    Impact i0 = ImpactPredictor.Predict(body, r, vel, ut, mass, drag);
    Impact iD = ImpactPredictor.Predict(body, r, vel + down, ut, mass, drag);
    Impact iC = ImpactPredictor.Predict(body, r, vel + cross, ut, mass, drag);
    dvWorld = Vector3d.zero;
    if (i0 == null || iD == null || iC == null) return false;
    // e = local north/east metres of (impact − target); J columns = (iD − i0), (iC − i0); dv = −J⁻¹·e
    // |det J| < 1e-6 → false
    return true;
}
```

```csharp
{
```

```csharp
    if (o.PeR <= peR) throw new ApiException(409, "no_solution", "the periapsis is already at or below the requested one");
    ResolveTarget(target, o.referenceBody, out double tLat, out double tLon);   // vessel: 409 no_target
    DragModel drag = DragModel.Initial(v);
    var sw = Stopwatch.StartNew();
    for (int i = 0; i < Samples; i++)   // 36 burn times over one period from start + lead

        double ut = start + Lead + o.period * i / Samples;
        OrbitMath.NodeSpec n = OrbitMath.ChangeApsis(o, ut, peR, raiseApo: false);
        OrbitMath.StateWorld(o, ut, out Vector3d r, out Vector3d vel);   // relative to the body, world axes at now
        if (Vector3d.Dot(n.DvWorld, vel) >= 0) throw new ApiException(409, "no_solution", "the burn is not retrograde");
        Impact imp = ImpactPredictor.Predict(o.referenceBody, r, vel + n.DvWorld, ut, v.totalMass * 1000, drag, sw, BudgetMs);
        if (imp == null && sw.Elapsed.TotalMilliseconds > BudgetMs) throw new ApiException(409, "no_solution", "deorbit search exceeded 2 s");
        // signed downrange error along the ground track; remember the first − → + sign change
    }
    // bisection on the bracket to 1 m (same budget); crossrange > 5000 m or no bracket → 409 no_solution
}
```

```csharp
// DragModel: K(M, ρv) = 0.5 · DragCubeMultiplier · DragMultiplier · PseudoRe(ρv) · Σ AreaDrag(dir, M) · Scale
public static DragModel Build(Vessel v)
{
    Vector3 flow = -v.ReferenceTransform.up;                 // falling bottom first: the velocity points to the vessel bottom
    var area = new double[MachSteps + 1];                   // M = 0 .. 25 step 0.1
    foreach (Part p in v.parts)
    {
        DragCubeList c = p?.DragCubes;
        if (c == null || c.None) continue;
        Vector3 d = p.transform.InverseTransformDirection(flow);   // = AddSurfaceDragDirection(-dragVectorDirLocal)
        for (int m = 0; m <= MachSteps; m++)
        {
            float mach = m * MachStep;
            for (int i = 0; i < 6; i++)                             // faces: right, left, up, down, forward, back
            {
                float dot = Vector3.Dot(d, Faces[i]);
                float curve = PhysicsGlobals.DragCurveValue(c.SurfaceCurves, (dot + 1f) * 0.5f, mach);
                float wd = c.WeightedDrag[i];
                float cd = wd < 1f ? Mathf.Pow(c.DragCurveCd.Evaluate(wd), c.DragCurveCdPower.Evaluate(mach)) : wd;
                area[m] += c.AreaOccluded[i] * curve * cd;
            }
        }
    }
    return new DragModel(area);
}

public double K(double mach, double rhoV) =>
    0.5 * PhysicsGlobals.DragCubeMultiplier * PhysicsGlobals.DragMultiplier
        * PhysicsGlobals.DragCurvePseudoReynolds.Evaluate((float)rhoV) * Area(mach) * Scale;   // Area: linear in the table, clamped

// Measure (q ≥ 1 kPa): Scale ← EMA(τ 3 s) of (Σ dragScalar · 1000 / (ρ v²)) / K₁(v.mach, ρ v) where K₁ is K with Scale = 1
```

```csharp
// ImpactPredictor (decision 12-2): Unity's semi-implicit Euler = leapfrog with the velocity half a step behind. The exact path's
// velocity is V = u + (h/2)·a, and velocity-dependent forces (drag, Coriolis) see the late u = V − (h/2)·a, so the exact path
// follows a_eff(r, V) = a(r, V − (h/2)·a(r, V)) (2nd order). Offset V − u = (h/2)·a: re-set at every jump of h or a.
public struct PhysStep
{
    public bool On;            // false (default) = plain RK4, today's result
    public double WarpEndUt;   // T = brake start − 30 s; +Infinity when not known (×2 in the whole atmosphere)
    public bool DiagBand;      // target land only (decision 12 diag, U6 removes): ×1 at 50–60 km
    public double H(CelestialBody body, double alt, double ut)
    {
        if (!On || !(body.atmosphere && alt < body.atmosphereDepth)) return 0;   // rails above the atmosphere: analytic
        bool x1 = ut >= WarpEndUt || (DiagBand && alt > 50000 && alt < 60000);
        return (x1 ? 1 : 2) * BaseDt;
    }
    public const double BaseDt = 0.02;   // KSP's ×1 physics step (TimeWarp.fixedDeltaTime default); never read at run time
}

// Unity's integrated acceleration (incl. thrust when the prediction burns). Frame per predicted point: alt below
// body.inverseRotThresholdAltitude (raised to the atmosphere depth on atmospheric bodies) = inverse rotation, rotating frame:
// gravity + drag + thrust − 2ω×v_rot − ω×(ω×r) (signs as KSP's getCoriolisAcc / getCentrifugalAcc); above it inertial.
private static Vector3d UnityAccel(CelestialBody body, double spin, double mass, DragModel drag, Vector3d thrust, Vector3d r, Vector3d v)
// RK4 derivative: the state stays inertial (r, V) as today; the inertial acceleration (Accel: gravity + drag on the air-relative
// velocity + thrust, no Coriolis / centrifugal) is taken at the late velocity V − (h/2)·a_U, a_U = UnityAccel (rotating frame
// under inverse rotation). The dropped h·ω×a_U is ~3e-4 m/s².
private static Vector3d EffAccel(..., double h, Vector3d thrust, Vector3d r, Vector3d V)
{
    if (h == 0) return Accel(..., thrust, r, V);   // rails: same as today
    Vector3d aU = UnityAccel(..., thrust, r, V);
    return Accel(..., thrust, r, V - aU * (h / 2));
}

// Predict / Propagate loop: RK4 on EffAccel with h held over the step; offset at the start and at each jump
double h = step.H(body, r.magnitude - body.Radius, ut0);
vel += UnityAccel(..., thrust, r, vel) * (h / 2);
while (...)
{
    Rk4(EffAccel with h and the step's thrust, ref r, ref vel, dt);   // was: Rk4 on Accel
    t += dt;
    double h2 = step.H(body, r.magnitude - body.Radius, ut0 + t);
    bool jump = h2 != h || ThrustChanged(t) || FrameChanged(r);      // flags, not a threshold on |Δa|
    if (jump)
    {
        // same time and place, before / after: old h, thrust, frame vs new ones (not the previous step's value)
        Vector3d a1 = UnityAccel(..., thrustBefore, frameBefore, r, vel);
        Vector3d a2 = UnityAccel(..., thrustAfter, frameAfter, r, vel);
        vel += a2 * (h2 / 2) - a1 * (h / 2);
        h = h2;
    }
}
```

- 범위 밖 발견: 상승(`gravity_turn`, `target_apoapsis` 76000)이 원지점 75,971 m(29 m 부족)에서 끝나지 않고 원지점을 지나 5% 스로틀로 계속 연소함. 하강으로 바뀐 뒤 계획을 끊고 `circularize`만 다시 내서 궤도에 올림.

**선행 단위**: U5c
