```csharp
private static bool _resumeUsed;     // one re-request per step after an unexpected end (design 0048 decision 2)
private static bool _resumePending;  // the next warp_started is that re-request

public static void Sync()
{
    bool active = Active;
    if (_requested && _wasActive && !active)
    {
        _requested = false;
        if (_resumeUsed)
        {
            _cancelled = true;
            Debug.Log("[KspBridge] warp ended without our Stop again: latched for this step (user cancel)");
        }
        else
        {
            _resumeUsed = true;
            _resumePending = true;
            Debug.Log("[KspBridge] warp ended without our Stop: one re-request allowed in this step");
        }
        Stopped("unexpected");
    }
    _wasActive = _requested && active;
}
```

```csharp
// CircularizeMode
internal const double WarpAlignRate = 0.02;   // rad/s: no warp request while still turning (design 0048 decision 1)

/// <summary>Wait steps request warp only once aligned; our running warp counts as aligned (attitude_error is NaN on rails,
/// and dropping WarpToUt would make the Coordinator stop it).</summary>
internal static bool WarpAligned(Vessel v, Telemetry t)
{
    if (Warp.Requested)
    {
        return true;
    }
    double err = t.Num["attitude_error"];
    return !double.IsNaN(err) && err <= AttitudeGateDeg && v.angularVelocity.magnitude <= WarpAlignRate;
}

internal const double AlignWaitS = 30;   // unaligned this long while warp is worth it: warp_unavailable align_wait

/// <summary>WarpToUt for a wait step: the target once aligned, else null; reports align_wait once per step after AlignWaitS
/// unaligned (only when a warp would still be requested, 15 s or more left).</summary>
internal static double? WarpGate(Vessel v, Telemetry t, double targetUt, ref double alignSince)
{
    double ut = Planetarium.GetUniversalTime();
    if (WarpAligned(v, t))
    {
        alignSince = double.NaN;
        return targetUt;
    }
    if (double.IsNaN(alignSince))
    {
        alignSince = ut;
    }
    else if (ut - alignSince > AlignWaitS && targetUt - ut >= 15)
    {
        Warp.AlignWait();
    }
    return null;
}

private double _alignSince = double.NaN;   // 각 모드 인스턴스 필드(단계마다 새 인스턴스)

// CircularizeMode.Update 대기 분기
if (!pastAp && ut < start)
{
    return new ModeCommand(AttitudeTarget.Vector(Horizontal(v)), 0) { WarpToUt = WarpGate(v, t, start, ref _alignSince) };
}
```

### 코드 초안

```csharp
private const double ResyncSlackS = 30;   // wake moved later than this at the kill check: wait again (design 0048 decision 4)
private const double GuardFactor = 3;
private const double GuardSlackM = 500;

private Vector3d _closestRelV;            // predicted relative velocity at the closest approach (world)

public void Begin(Vessel v, Telemetry t)
{
    Vessel tv = TargetVessel(v) ?? throw new ModeAbortException("no_target", "no target vessel");
    // ... existing body check ...
    _targetId = tv.persistentId;
    FindClosest(v, tv, Planetarium.GetUniversalTime());   // throws target_too_far
}

/// <summary>Closest approach from now (Begin and the kill check); the current state when already closer than any later pass.</summary>
private void FindClosest(Vessel v, Vessel tv, double now)
{
    OrbitMath.ClosestApproach(v.orbit, tv.orbit, now, OrbitMath.SafeSpan(v.orbit, now), out _closestUt, out _closestDist);
    _closestRelV = (v.orbit.getOrbitalVelocityAtUT(_closestUt) - tv.orbit.getOrbitalVelocityAtUT(_closestUt)).xzy;
    double here = Vector3d.Distance(v.CoMD, tv.CoMD);
    if ((here < _closestDist || !OrbitMath.Finite(_closestDist)) && here <= MaxClosestM)
    {
        _closestUt = now;
        _closestDist = here;
        _closestRelV = v.obt_velocity - tv.obt_velocity;
    }
    if (!OrbitMath.Finite(_closestDist) || _closestDist > MaxClosestM)
    {
        throw new ModeAbortException("target_too_far", "closest approach " + F(_closestDist, 0) + " m > " + F(MaxClosestM, 0) + " m: rendezvous first");
    }
}

/// <summary>Kill start at the predicted closest relative speed: half the kill burn before the closest approach (design 0032
/// decision 22), the whole burn when the pass is closer than the distance covered while killing (stop before it).</summary>
private double Wake(Vessel v)
{
    double speed = _closestRelV.magnitude;
    double kill = EngineModel.BurnTimeStaged(v, speed, out _);
    double lead = !OrbitMath.Finite(kill) ? 0 : _closestDist < speed * kill / 2 ? kill : kill / 2;
    return _closestUt - lead - EngineModel.SpoolUpS(v) - KillLeadS;
}

private double _alignSince = double.NaN;

private ModeCommand WaitCommand(Vessel v, Telemetry t, double wake) =>
    new ModeCommand(AttitudeTarget.Vector(-_closestRelV.normalized), 0) { WarpToUt = CircularizeMode.WarpGate(v, t, wake, ref _alignSince) };
```

## 단위 5: 게임 실측

### 대상

- 코드 변경 없음(실측에서 상수 조정이 필요하면 해당 단위 파일을 고치고 그 단위 검증부터 다시 한다).

### 확인 항목 (MCP 응답·이벤트로만 확인, `get_status`로 장면·남의 비행 확인 후 진행)

1. `execute_node`: 노드를 기수와 먼 방향으로 만든 뒤 실행 → 정렬(`attitude_error ≤ 5°`) 전에는 `warp_started`가 없고, 정렬 뒤 `warp_started` → 연소 시작 약 15 s 전 `warp_stopped(ended_by=plan)`.
2. 재요청: 워프 중 `set_time_warp`로 1배속으로 내려 우리 워프를 끊는다 → `warp_stopped(ended_by=unexpected)` 뒤 `warp_started(resumed=true)`. 한 번 더 끊으면 재요청 없음.
3. `circularize`: 상승 후 원궤도화 단계에서 정렬 뒤 워프.
4. `approach`: `rendezvous_phase` 노드를 `execute_node`로 실행한 직후 `approach` → `approach_phase` `wait` 뒤 최근접 근처에서 `kill`. kill 진입 때 거리가 예측 최근접 근처(수 km 이내)이고 원거리 kill이 없다.
5. 원거리 가드: approach 대기 중 수동 노드 연소 등으로 궤도를 바꿔 재계산 결과가 어긋나게 만든다 → 재대기 또는 `approach_off_course`/`target_too_far`.
