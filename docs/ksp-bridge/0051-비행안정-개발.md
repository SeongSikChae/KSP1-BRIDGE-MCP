/// <summary>Highest part temperature / limit, internal or skin (1 = at the limit; parts explode just above). NaN without parts.</summary>
internal static double MaxTempRatio(Vessel v)
{
    double max = double.NaN;
    foreach (Part p in v.parts)
    {
        double r = Math.Max(p.maxTemp > 0 ? p.temperature / p.maxTemp : 0, p.skinMaxTemp > 0 ? p.skinTemperature / p.skinMaxTemp : 0);
        if (double.IsNaN(max) || r > max)
        {
            max = r;
        }
    }
    return max;
}

public sealed class Settings
{
    private static readonly string[] Keys =
    {
        "enabled", "abort", "flip_aoa_deg", "flip_min_q_kpa", "flip_for_s", "lost_error_deg", "lost_for_s",
        "waste_error_deg", "waste_for_s", "osc_flips", "osc_min_rate", "temp_warn", "temp_abort",
    };

    // Partial: only the keys the JSON gave (null = inherit).
    public bool? Enabled, Abort;
    public double? FlipAoaDeg, FlipMinQ, FlipForS, LostErrorDeg, LostForS, WasteErrorDeg, WasteForS, OscFlips, OscMinRate, TempWarn, TempAbort;

    /// <summary>Partial settings from a plan or step "monitor" object; null when the key is absent.</summary>
    public static Settings Parse(object raw, string path)
    {
        if (raw == null)
        {
            return null;
        }
        Dictionary<string, object> o = raw as Dictionary<string, object> ?? throw Plan.Invalid(path, "must be an object");
        Plan.UnknownKeys(o, Keys, path);
        Settings s = new Settings
        {
            Enabled = o.ContainsKey("enabled") ? Plan.OptionalBool(o, "enabled", path, true) : (bool?)null,
            FlipAoaDeg = Num(o, "flip_aoa_deg", path, 1, 180),
            // ... 나머지 키도 같은 형태(범위: *_for_s 0.1~600, *_error_deg 1~180, flip_min_q_kpa 0~1000, osc_flips 2~100,
            //     osc_min_rate 0.001~10, temp_warn 0.1~1, temp_abort 0.1~2)
        };
        if (s.TempWarn > s.TempAbort)   // both given in this object
        {
            throw Plan.Invalid(path, "temp_warn must not exceed temp_abort");
        }
        return s;
    }

    private static double? Num(Dictionary<string, object> o, string name, string path, double min, double max) =>
        o.ContainsKey(name) ? Plan.OptionalNumber(o, name, path, min, max, 0) : (double?)null;
}

/// <summary>Effective settings for a tick: step → plan → defaults. TempWarn is capped at TempAbort.</summary>
public sealed class Effective
{
    public bool Enabled, Abort;
    public double FlipAoaDeg, FlipMinQ, FlipForS, LostErrorDeg, LostForS, WasteErrorDeg, WasteForS, OscFlips, OscMinRate, TempWarn, TempAbort;

    public static Effective Merge(Settings step, Settings plan)
    {
        Effective e = new Effective
        {
            Enabled = step?.Enabled ?? plan?.Enabled ?? true,
            Abort = step?.Abort ?? plan?.Abort ?? true,
            FlipAoaDeg = step?.FlipAoaDeg ?? plan?.FlipAoaDeg ?? 30,
            // ... 나머지(기본값 FlipMinQ 5, FlipForS 2, LostErrorDeg 30, LostForS 5, WasteErrorDeg 10, WasteForS 5,
            //     OscFlips 6, OscMinRate 0.05, TempWarn 0.8, TempAbort 0.95)
        };
        e.TempWarn = Math.Min(e.TempWarn, e.TempAbort);
        return e;
    }
}

```csharp
public static class FlightMonitor
{
    private const double WindowS = 10;

    private struct Sample { public double Ut, Err, Rate, Aoa, Temp; public Vector3 Omega; }
    private static readonly Queue<Sample> _window = new Queue<Sample>();
    private static readonly Dictionary<string, string> _level = new Dictionary<string, string>();   // kind → warn|abort
    private static double _flipFor, _lostFor, _wasteFor, _lastUt = double.NaN;
    private static double _oscMinRate = 0.05;
    private static bool _running;   // false after EndPlan: Summary shows no alerts

    /// <summary>Every real plan tick after the abort check (not in the trial). Returns the abort message, or null.</summary>
    public static string Tick(Telemetry t, Vessel v, double ut, ModeCommand cmd, Effective s)
    {
        _running = true;
        _oscMinRate = s.OscMinRate;   // before the Enabled check: the summary counts with it either way
        double dt = double.IsNaN(_lastUt) ? 0 : Math.Max(0, ut - _lastUt);
        _lastUt = ut;
        if (!v.packed)
        {
            Add(new Sample { Ut = ut, Err = t.Num["attitude_error"], Rate = t.Num["angular_rate"], Aoa = t.Num["aoa"],
                Temp = t.Num["temp_ratio"], Omega = v.angularVelocity });
        }
        if (!s.Enabled)
        {
            return null;
        }
        bool thrusting = t.Num["throttle"] > 0 && t.Num["thrust"] > 0;
        double err = t.Num["attitude_error"], q = t.Num["dynamic_pressure"], aoa = t.Num["aoa"], temp = t.Num["temp_ratio"];
        bool pauseErr = t.Str["staging"] != "none" || cmd.Aligning || double.IsNaN(err);   // keep the timer, add nothing

        // flipped: only while the commanded attitude itself flies along the surface velocity (land / retrograde descent excluded)
        bool inAtmo = v.mainBody.atmosphere && v.altitude < v.mainBody.atmosphereDepth;
        double cmdAoa = cmd.Attitude.HasValue && v.srfSpeed >= 1
            ? Vector3d.Angle(cmd.Attitude.Value.Direction(v), v.srf_velocity) : double.NaN;
        Accumulate(ref _flipFor, inAtmo && q >= s.FlipMinQ && cmdAoa <= s.FlipAoaDeg && aoa > s.FlipAoaDeg, false, dt);
        Accumulate(ref _lostFor, thrusting && err > s.LostErrorDeg, pauseErr, dt);
        Accumulate(ref _wasteFor, thrusting && err > s.WasteErrorDeg, pauseErr, dt);

        string abortLevel = s.Abort ? "abort" : "warn";
        string abort = null;
        abort = Set("flipped", _flipFor >= s.FlipForS ? abortLevel : null, aoa, s.FlipAoaDeg,
            "aoa " + F(aoa) + " deg at q " + F(q) + " kPa for " + F(_flipFor) + " s") ?? abort;
        abort = Set("control_lost", _lostFor >= s.LostForS ? abortLevel : null, err, s.LostErrorDeg,
            "attitude_error " + F(err) + " deg under thrust for " + F(_lostFor) + " s") ?? abort;
        abort = Set("overheat", temp >= s.TempAbort ? abortLevel : temp >= s.TempWarn ? "warn" : null, temp,
            temp >= s.TempAbort ? s.TempAbort : s.TempWarn, "temp_ratio " + F(temp)) ?? abort;
        Set("thrust_waste", _wasteFor >= s.WasteForS ? "warn" : null, err, s.WasteErrorDeg, null);
        int flips = MaxFlips(s.OscMinRate);
        bool oscOn = _level.ContainsKey("oscillation") ? flips >= s.OscFlips / 2 : flips >= s.OscFlips;   // hysteresis
        Set("oscillation", oscOn ? "warn" : null, flips, s.OscFlips, null);
        return abort;
    }

    /// <summary>Sign changes per local axis over the window, counting only samples with |ω axis| ≥ minRate
    /// (a near-zero axis does not flip back and forth on noise).</summary>
    private static int MaxFlips(double minRate) { /* 축마다 마지막 유효 부호를 기억하며 바뀐 횟수, 세 축 최댓값 */ }
}
```

string monitorAbort = FlightMonitor.Tick(_t, v, ut, Command, ms);
if (monitorAbort != null)

- `Set(kind, level, value, limit, detail)`: 항목의 수준을 한 틱에 한 번 정한다. `level`이 지금과 다르면 null이 아닐 때 `monitor_alert {kind, level, value, limit}`, null이면(해제) `monitor_clear {kind}`를 `FlightEvents.Add(..., Coordinator.StepIndex, ...)`로 낸다. 같으면 아무것도 내지 않는다. `level`이 `abort`면 `kind + ": " + detail`을 돌려준다.
- `Accumulate(ref acc, cond, pause, dt)`: `pause`면 그대로, 아니면 `cond`일 때 `acc += dt`, 거짓이면 `acc = 0`. 스테이징·정렬 중·오차 NaN이 누적을 리셋하지 않는다.
- 과열은 한 번의 `Set`으로 중단·경고·해제를 정하므로 0.8~0.95 구간에서 이벤트가 반복되지 않는다.
- 진동 해제 기준은 `osc_flips / 2` 미만(정수 나눗셈 아님, 6이면 3 미만).
- `ResetStep()`: 경고 중인 항목마다 `monitor_clear`를 내고 `_level`·누적 시간을 비운다(표본 창 유지). `NextStep`과 `ReplaceSteps` include_current의 성공 경로(`_mode = mode` 직후)에서 부른다.
- `EndPlan()`: `_level`·누적 시간을 이벤트 없이 비우고 `_running = false`. `Abort`·`Complete`·`Stop`에서 부른다(`plan_aborted`·`plan_completed`가 끝을 알린다). 중단 직전의 `monitor_alert abort`는 `Set`이 이미 냈다.
- `Reset()`: `EndPlan()` + 표본 창·`_lastUt` 비움. 새 계획 시작에서 부른다.

public static Dictionary<string, object> Summary()
{
    if (_window.Count == 0)
    {
        return null;
    }
    // Err·Aoa·Temp는 NaN 표본을 빼고 최대·평균, 하나도 없으면 null
    return new Dictionary<string, object>
    {
        ["window_s"] = R(last.Ut - first.Ut, 1),
        ["samples"] = (double)_window.Count,
        ["attitude_error_max"] = Max(x => x.Err), ["attitude_error_mean"] = Mean(x => x.Err),
        ["angular_rate_max"] = Max(x => x.Rate), ["sign_flips"] = (double)MaxFlips(_oscMinRate),
        ["aoa_max"] = Max(x => x.Aoa), ["temp_ratio_max"] = Max(x => x.Temp),
        ["alerts"] = _running ? new List<object>(_level.Keys) : new List<object>(),   // ended plan: no current alerts
    };
}

## 단위 6: 게임 실측

### 확인 항목 (MCP 응답·이벤트로만, `get_status`로 장면 확인 후)

1. 불안정 기체: `KspBridge-StabC` 구성에서 하단 날개를 뺀 기체(`KspBridge-StabU`)를 노즈콘을 단 채 `gravity_turn` 기본값으로 발사 → `monitor_alert` `flipped` 또는 `control_lost` `abort` → `plan_aborted` `monitor`, 연료를 다 쓰기 전(`stage_dv` > 0).
2. 과열: 노즈콘 없는 `KspBridge-StabT` 구성으로 `gravity_turn` 기본값 → `monitor_alert` `overheat` `warn`(0.8) → `abort`(0.95) → 파괴(`part_lost`) 전에 `plan_aborted` `monitor`. 중단 뒤 기체가 계속 가열될 수 있으므로 `part_lost` 여부를 함께 기록한다.
3. 상승 중 `get_flight` `plan.monitor`가 값을 내고, `abort` 조건 `{var: temp_ratio, op: ">", value: 0.5}` 같은 새 변수 조건이 파싱·동작.
4. `monitor: {enabled: false}`로 같은 불안정 기체 → 감시 이벤트 없음(끄기 확인, 시간이 되면).
5. 역방향 하강: 대기권 하강 중인 기체에 `land`(또는 `attitude` `retrograde`) 계획 → 동압 5 kPa 이상 구간에서 `flipped` 이벤트·중단 없음. `land` 강제 시작이 기운 자세에서 나면 `control_lost`도 없음(`Aligning` 동안 멈춤).
6. 과열 0.8~0.95 구간에서 `overheat` 이벤트가 `warn` 한 번, `abort` 한 번만 나오는지(`get_flight_events`로 개수 확인).

### 검증

- 위 1~3, 5, 6 관찰(4는 가능하면).
- 진동 경고(`oscillation`)의 이슈 완료 조건은 3단계에서 판정한다(0050 검증 5). 여기서는 불안정 기체 상승 중 `plan.monitor` `sign_flips`가 값을 내는지만 본다.
