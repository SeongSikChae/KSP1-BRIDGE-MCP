```csharp
// AttitudeController fields
private static bool _powered = true;   // power hysteresis state

// AvailableTorque, cache rebuild block: only when the vessel object changed (not on a part count change), _powered = true;
// AvailableTorque, before the loop: one power lookup per tick, with hysteresis
bool power = HasCharge(v);

// in the loop: a wheel that is not running gives no torque

{
    continue;   // sum and _kindSum both skip it
}

private static bool HasCharge(Vessel v)
{
    PartResourceDefinition ec = PartResourceLibrary.Instance.GetDefinition("ElectricCharge");
    if (ec == null) return true;   // no such resource in this game: keep the old behaviour
    v.GetConnectedResourceTotals(ec.id, out double amount, out double max);
    if (_powered && amount <= 1e-6) _powered = false;                 // empty: off
    else if (!_powered && amount >= 0.01 * max && max > 0) _powered = true;   // 1% back: on
    return _powered;
}
```

## 단위 2: 선형 구간과 목표 각속도 앞먹임 (설계 결정 1·1-2, g = 1)

### 대상

- `Coordinator.cs`: 계획 시작, 단계 시작(`NextStep`), 단계 교체(`ReplaceSteps`)에서 `AttitudeController.ResetFeedforward()`
- `ControllerTrace.cs`: `want_slope_at_amp`를 같은 곡선으로, 표본 `Ff`, 요약 `ff_rms`, 시계열 `ff`(설계 결정 4)
- `Routes.cs`: `0.2.60`

### 코드 초안

```csharp
internal const float KLin = Kp / 3f;   // outer-loop gain near zero error, a third of the inner rate loop

private const float FfTau = 0.3f;   // feedforward low-pass time constant (s)
private static Vector3d _prevDir;
private static bool _prevDirValid;
private static Vector3 _ff;         // filtered feedforward

internal static void ResetFeedforward() { _prevDirValid = false; _ff = Vector3.zero; }

// Steer, before the axis loop: target-rate feedforward (pitch, yaw), err sign convention
Vector3d dirNow = target.Direction(_v).normalized;
Vector3 raw = Vector3.zero;
// a change faster than the controller can follow (1.5 · OmegaMax per tick) is a step, e.g. a reference switch
if (_prevDirValid && Vector3d.Angle(_prevDir, dirNow) * Mathf.Deg2Rad <= 1.5 * OmegaMax * dt)
{
    // target rotation rate in world space (axis × rate for a small angle), then in vessel axes; no atan2, no singularity
    Vector3 wWorld = (Vector3)(Vector3d.Cross(_prevDir, dirNow) / dt);
    Vector3 wLocal = Quaternion.Inverse(_v.ReferenceTransform.rotation) * wWorld;
    raw = new Vector3(wLocal.x, 0, wLocal.z);   // x: pitch (+x turns the nose toward +z), z: yaw (+z toward -x); no roll
}
_ff += (raw - _ff) * (1 - Mathf.Exp(-dt / FfTau));   // low-pass: tick noise out, slow target rotation through
Vector3 ff = Vector3.ClampMagnitude(_ff, OmegaMax);
_prevDir = dirNow; _prevDirValid = true;

float g = 1f;   // unit 3: the estimated gain of this axis
float want = i == 1 && !target.HasRoll
    ? 0f
    : Mathf.Clamp(ff[i] + Mathf.Sign(err[i]) * Mathf.Min(Mathf.Min(g * KLin * Mathf.Abs(err[i]),
        Mathf.Sqrt(2 * g * alpha * Mathf.Abs(err[i])) * KStop), OmegaMax), -OmegaMax, OmegaMax);
// input stays -a / alpha (model alpha)
```

- 부호: `AxisErrors`의 주석대로 x 양의 각속도는 기수(+y)를 +z로, z 양의 각속도는 −x로 돌린다. 피치 오차 `atan2(dir.z, dir.y)`·요 오차 `atan2(−dir.x, dir.y)`가 목표를 따라 커지는 방향과 같아 부호 변환이 없다. 실측에서 앞먹임이 `err` 변화와 같은 부호인지 시계열로 확인한다.
- 표본 `Ff`에는 제한 뒤의 `ff`를 축별로 기록한다(롤 0).
- `ResetFeedforward()` 호출: `Attach`·`Retarget`·`Adopt`(제어 기체 변경), `Fly`가 `Steer`까지 가지 않는 틱(계획 없음, 자세 목표 없음, `Suspended`), `Coordinator`의 계획 시작·단계 시작·단계 교체.

```csharp
// ControllerTrace.Axis: slope of the curve actually used at the operating point e (rad), gain g (1 until unit 3)
double lin = g * AttitudeController.KLin * e, sq = Math.Sqrt(2 * g * alphaModel * e) * AttitudeController.KStop;
slope = Math.Min(lin, sq) >= AttitudeController.OmegaMax ? 0.0
      : lin <= sq ? g * AttitudeController.KLin
      : AttitudeController.KStop * Math.Sqrt(g * alphaModel / (2 * e));
```

### 코드 초안

```csharp
// GainEstimator.cs: per axis, exponentially forgotten least squares (weight = dt) of y = dω/dt on x = -input·alpha
// (model) k ticks earlier, k = 1..10 on the same samples, with an intercept; the best r² picks k. Main thread only;
// no allocation per tick.
internal static class GainEstimator
{
    internal const int MaxLag = 10;
    private const double Tau = 3.0, TauUse = 1.0, MinR2 = 0.5, MinWeightS = 1.0, GMin = 0.2, GMax = 1.0;

    private struct Acc { public double W, Sx, Sy, Sxx, Sxy, Syy; }
    private static readonly Acc[,] _acc = new Acc[3, MaxLag];
    private static readonly Vector3[] _x = new Vector3[MaxLag];   // ring: -input·alpha of the last MaxLag ticks
    private static int _head, _filled;
    private static double _lastUt = double.NaN;
    private static Vector3 _lastOmega;
    private static readonly double[] _satW = new double[3], _posW = new double[3], _negW = new double[3], _allW = new double[3];
    private static Vector3 _gUsed = Vector3.one;

    internal static Vector3 Used => _gUsed;

    internal static void Reset() { /* zero _acc, ring, sat/pos/neg/all, _lastUt = NaN, _gUsed = 1 */ }

    /// Start of Steer, before this tick's inputs: y from ω now, x from the ring. active[a] false (roll without a roll
    /// target) leaves that axis at 1 and does not accumulate it.
    internal static void Observe(double ut, Vector3 omega, double dt, bool rollActive)
    {
        bool contiguous = !double.IsNaN(_lastUt) && ut - _lastUt > 0 && ut - _lastUt <= 1.5 * dt;
        if (!contiguous) { _filled = 0; }          // a gap (suspended, no attitude target) restarts the lag history only
        else if (_filled >= MaxLag) { /* per active axis, per k: decay all sums by exp(-step/Tau), add (step, step·x_k,
                                          step·y, step·x_k², step·x_k·y, step·y²) with step = ut - _lastUt; pick the best r²;
                                          if r² >= MinR2, W >= MinWeightS, std(x) >= 1% mean|x| and not one-sided saturation
                                          (sat share >= 0.9 with one input sign): move _gUsed toward clamp(g, GMin, GMax)
                                          by 1 - exp(-step/TauUse) */ }
        _lastUt = ut; _lastOmega = omega;
    }

    /// End of Steer: this tick's input and model alpha enter the ring.
    internal static void Push(Vector3 input, Vector3 alphaModel, double dt) { /* ring write, sat/pos/neg/all decay + add, _filled++ */ }
}
```

```csharp
Vector3 gv = GainEstimator.Used;
for (int i = 0; i < 3; i++)
{
    float alpha = Mathf.Max(torque[i] / Mathf.Max(moi[i], 1e-3f), 1e-3f);   // model alpha: input normalization as now
    float g = gv[i];                                                        // outer loop only (unit 2 curve)
    ...
}
GainEstimator.Push(new Vector3(pitch, roll, yaw), alpha3, dt);
```

- 표본 `Alpha`는 지금처럼 모델 `alpha`(진단 `alpha_gain`이 모델 대비 실측을 그대로 낸다), 새 `GainUsed`를 더한다. `alpha_eff` = 축별 `GainUsed·Alpha` 평균.
- `GainEstimator.Reset()`: `AttitudeController.Attach`가 새 기체에 붙을 때(같은 기체 재부착은 지금처럼 건너뜀; 도킹 결합으로 기체가 바뀔 때 포함), `AvailableTorque` 캐시 재생성 블록(부품 수 변화), `Coordinator`에서 실행 중인 계획이 없을 때의 새 계획 시작.
